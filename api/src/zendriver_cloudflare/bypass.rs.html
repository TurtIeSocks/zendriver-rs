1//! Cloudflare Turnstile bypass driver.
2//!
3//! Public entry is [`CloudflareBypass`] — constructed via
4//! `Tab::cloudflare()` (zendriver crate, feature-gated). The driver runs a
5//! single CDP poll loop that, per tick:
6//!
7//! 1. Re-evaluates a unified shadow-DOM walker in the page's main world
8//!    (private `POLL_JS`) returning, in one round-trip:
9//!    - the `cf-turnstile-response` (or legacy `cf_challenge_response`) token
10//!      if present,
11//!    - the Turnstile challenge iframe's bounding box if mounted,
12//!    - whether *any* challenge marker exists on the page
13//!      (container, hidden input, or live iframe).
14//! 2. Resolves to [`ClearanceOutcome::TokenAcquired`] the first tick a
15//!    non-empty token is observed — including the **invisible Turnstile**
16//!    path where the iframe never mounts and the token is populated
17//!    directly.
18//! 3. If a *clickable* challenge iframe is mounted (non-zero size and
19//!    visible), scrolls it into view, re-reads its box, and dispatches a raw
20//!    left-click at the canonical
21//!    `bbox.x + bbox.width * 0.15, bbox.y + bbox.height * 0.50` offset
22//!    (15% from left, 50% from top — Python's `cloudflare.py` convention).
23//!    Up to `MAX_CLICK_ATTEMPTS` clicks are spent, spaced
24//!    `CLICK_RETRY_TICKS` ticks apart, so a swallowed first click is not
25//!    the end of the run.
26//! 4. Resolves to [`ClearanceOutcome::ChallengeGone`] when the challenge
27//!    stops being actionable without yielding a token — either the iframe we
28//!    clicked is no longer a valid click target, or every challenge marker
29//!    observed on an earlier tick has vanished (the JS-only interstitial
30//!    that clears itself). The first of those is weaker than it sounds; see
31//!    the variant's own docs before treating it as clearance.
32//! 5. Resolves to [`ClearanceOutcome::TimedOut`] on deadline, carrying
33//!    `saw_challenge` — `true` when challenge markers were seen but never
34//!    resolved, `false` when the entire timeout window elapsed without
35//!    observing any challenge markers (the caller likely invoked the bypass
36//!    on a page that has no Cloudflare gate at all).
37

48/// Result of a clearance attempt.
49#[derive(Debug, Clone)]
50pub enum ClearanceOutcome {
51    /// Turnstile produced a token (value of `cf-turnstile-response`).
52    TokenAcquired(String),
53    /// The challenge stopped being actionable without yielding a token — the
54    /// iframe we clicked is no longer a valid click target, or all challenge
55    /// markers seen earlier in the run have vanished (a JS-only interstitial
56    /// that cleared itself). Usually the clearance-cookie shortcut.
57    ///
58    /// **Not proof the gate was passed.** "No longer a valid click target"
59    /// is `clickableRect` returning null, which it also does for a widget
60    /// that is still mounted but zero-size, `visibility: hidden`,
61    /// `display: none` or `opacity: 0`. So once a click has landed, a tick
62    /// where the widget is merely between states reaches this terminal too.
63    /// Confirm the page is actually through the gate before treating what
64    /// you scrape next as clean.
65    ChallengeGone,
66    /// Deadline elapsed without a terminal clearance state. `saw_challenge`
67    /// is `true` if any challenge marker (container, hidden input, or live
68    /// iframe) was ever observed; `false` if none ever appeared — the caller
69    /// likely invoked the bypass on a page that has no Cloudflare gate at all.
70    /// Not a fault: a deadline in a bot-management flow is a normal "didn't
71    /// finish, retry or give up" terminal.
72    TimedOut { saw_challenge: bool },
73}
74
75/// Default poll interval for `wait_for_clearance`.
76const DEFAULT_POLL_INTERVAL: Duration = Duration::from_millis(500);
77
78/// How many clicks a single `wait_for_clearance` run may spend on the
79/// interactive widget. Cloudflare drops clicks that land while the widget is
80/// still booting, so one latched attempt strands the run until the deadline.
81const MAX_CLICK_ATTEMPTS: u32 = 3;
82
83/// Poll ticks to wait after a click before spending another attempt. At the
84/// default 500ms interval this leaves the widget ~2s to answer a click.
85const CLICK_RETRY_TICKS: u32 = 4;

96impl<'a> CloudflareBypass<'a> {
97    /// Create a new bypass driver bound to `session` with a default 500ms
98    /// poll interval.
99    pub fn new(session: &'a SessionHandle) -> Self {
100        Self {
101            session,
102            poll_interval: DEFAULT_POLL_INTERVAL,
103        }
104    }
105
106    /// Override the default 500ms poll interval used by
107    /// [`wait_for_clearance`](Self::wait_for_clearance).
108    #[must_use]
109    pub fn poll_interval(mut self, dur: Duration) -> Self {
110        self.poll_interval = dur;
111        self
112    }
113
114    /// Poll the page until a Turnstile clearance terminal state is reached or
115    /// `timeout` elapses.
116    ///
117    /// Handles both the **interactive** flow (challenge iframe mounts → we
118    /// click it → token appears) and the **invisible** flow (no iframe;
119    /// token is populated directly by Cloudflare's loader script). Resolution
120    /// rules per tick are documented at module level.
121    ///
122    /// # Returns
123    /// - `Ok(ClearanceOutcome::TokenAcquired(token))` — the
124    ///   `cf-turnstile-response` input picked up a non-empty value, either
125    ///   after we clicked the interactive iframe or because the page uses
126    ///   invisible Turnstile.
127    /// - `Ok(ClearanceOutcome::ChallengeGone)` — the challenge stopped being
128    ///   actionable without yielding a token (e.g. a clearance-cookie
129    ///   shortcut): either an iframe we clicked is no longer a valid click
130    ///   target, or challenge markers observed on an earlier tick are all
131    ///   gone. Read [`ClearanceOutcome::ChallengeGone`] before treating this
132    ///   as proof the gate was passed.
133    /// - `Ok(ClearanceOutcome::TimedOut { saw_challenge })` — `timeout`
134    ///   elapsed without a terminal clearance state. `saw_challenge` is
135    ///   `true` if challenge markers were observed (markers present but
136    ///   never resolved); `false` if none ever appeared (the caller likely
137    ///   invoked the bypass on a page that has no Cloudflare gate). Not a
138    ///   fault — a deadline here is a normal "retry or give up" signal.
139    ///
140    /// # Errors
141    /// - [`CloudflareError::Call`] / [`CloudflareError::JsError`] — CDP or
142    ///   in-page evaluator failure.
143    ///
144    /// ```no_run
145    /// # async fn ex(tab: &zendriver_transport::SessionHandle)
146    /// #   -> Result<(), zendriver_cloudflare::CloudflareError> {
147    /// use std::time::Duration;
148    /// use zendriver_cloudflare::CloudflareBypass;
149    ///
150    /// let bypass = CloudflareBypass::new(tab)
151    ///     .poll_interval(Duration::from_millis(250));
152    /// let outcome = bypass.wait_for_clearance(Duration::from_secs(15)).await?;
153    /// println!("{outcome:?}");
154    /// # Ok(()) }
155    /// ```
156    pub async fn wait_for_clearance(
157        self,
158        timeout: Duration,
159    ) -> Result<ClearanceOutcome, CloudflareError> {
160        let deadline = Instant::now() + timeout;
161        let mut clicks: u32 = 0;
162        let mut ticks_since_click: u32 = 0;
163        let mut ever_seen_markers = false;
164
165        let mut stall_ticks: u32 = 0;
166        let mut warned_stall = false;
167
168        loop {
169            let state = poll_state(self.session).await?;
170            // Snapshot the latch *before* folding in this tick: the
171            // markers-vanished terminal below needs "seen on an earlier
172            // tick", not "seen including right now".
173            let seen_markers_before = ever_seen_markers;
174            if state.has_markers {
175                ever_seen_markers = true;
176            }
177
178            if let Some(token) = state.token {
179                return Ok(ClearanceOutcome::TokenAcquired(token));
180            }
181
182            // Click cadence: the first clickable tick clicks straight away,
183            // every later attempt waits `CLICK_RETRY_TICKS` ticks after the
184            // click that landed, and a run spends at most
185            // `MAX_CLICK_ATTEMPTS` clicks. So at the default interval the
186            // second click lands on tick 5, not tick 4.
187            let may_click = clicks < MAX_CLICK_ATTEMPTS
188                && (clicks == 0 || ticks_since_click >= CLICK_RETRY_TICKS);
189
190            let mut clicked = false;
191            match state.bbox {
192                Some(_) if may_click => {
193                    // Raw `Input.dispatchMouseEvent` coordinates are viewport
194                    // relative, so a below-the-fold widget must be scrolled in
195                    // and re-measured before the click can land on it.
196                    if let Some(target) = scroll_into_view(self.session).await? {
197                        let click_x = target.x + target.width * 0.15;
198                        let click_y = target.y + target.height * 0.50;
199                        click_at(self.session, click_x, click_y).await?;
200                        clicks += 1;
201                        ticks_since_click = 0;
202                        clicked = true;
203                    }
204                }
205                // The challenge stopped being actionable without a token:
206                // either the iframe we clicked is no longer a valid click
207                // target, or markers seen on an earlier tick are all gone (a
208                // JS-only interstitial clearing itself). Both are usually the
209                // clearance-cookie shortcut.
210                //
211                // OPEN QUESTION: `bbox` is `None` for a widget that is merely
212                // hidden or zero-size, so after the first landed click any
213                // unclickable tick hits this *success* terminal. Prescribed
214                // fix: return `iframePresent` alongside `bbox` from `POLL_JS`
215                // and key the `clicks > 0` disjunct off presence, not
216                // clickability. Not applied here — changing which runs report
217                // clearance is a call for a human, not a doc fix.
218                None if clicks > 0 || (seen_markers_before && !state.has_markers) => {
219                    return Ok(ClearanceOutcome::ChallengeGone);
220                }
221                _ => {}
222            }
223
224            // Every tick that did not land a click is a tick with no progress,
225            // the scroll-returned-`None` path included. That path used to skip
226            // the counter, so a widget that stayed mounted but stopped being
227            // measurable — the run this hint exists to explain — was the one
228            // run that could never produce it.
229            if clicked {
230                stall_ticks = 0;
231            } else {
232                stall_ticks += 1;
233                if stall_ticks == 10 && !warned_stall {
234                    tracing::warn!(
235                        poll_interval_ms = self.poll_interval.as_millis() as u64,
236                        "cloudflare clearance stalled — is BrowserBuilder::stealth enabled?"
237                    );
238                    // `stall_ticks` resets on a landed click, so `== 10` is
239                    // reachable more than once per run; the latch is what
240                    // keeps the hint to a single line.
241                    warned_stall = true;
242                }
243            }
244
245            // Only meaningful once something has been clicked — unguarded, it
246            // counted ticks before the first click, which the `clicks == 0`
247            // arm of `may_click` then had to ignore.
248            if clicks > 0 {
249                ticks_since_click = ticks_since_click.saturating_add(1);
250            }
251
252            if Instant::now() >= deadline {
253                return Ok(ClearanceOutcome::TimedOut {
254                    saw_challenge: ever_seen_markers,
255                });
256            }
257
258            tokio::select! {
259                () = tokio::time::sleep(self.poll_interval) => {}
260                () = tokio::time::sleep_until(deadline) => {}
261            }
262        }
263    }
264}

266/// Decoded payload from the unified poll evaluator. Combines token check,
267/// iframe bbox, and a coarse "any challenge marker present" flag so each
268/// tick is a single CDP round-trip.
269#[derive(Debug, Deserialize)]
270struct PollState {
271    /// Non-empty `cf-turnstile-response` / `cf_challenge_response` value, if
272    /// any.
273    #[serde(default)]
274    token: Option<String>,
275    /// Bounding box of the Turnstile challenge iframe when it is a valid
276    /// click target — mounted, non-zero size, and visible. Walks shadow roots
277    /// so shadow-hosted widgets still surface. `None` for a hidden or 0×0
278    /// iframe, which must never be clicked.
279    #[serde(default)]
280    bbox: Option<BoundingBox>,
281    /// `true` when the page has any of: a `.cf-turnstile` / `.turnstile`
282    /// container, a `cf-turnstile-response` hidden input, or a live
283    /// challenge iframe. Used to distinguish "wrong page" (no markers ever
284    /// seen) from "real challenge that timed out".
285    #[serde(default, rename = "hasMarkers")]
286    has_markers: bool,
287}

289/// Shared in-page prelude for the two evaluators below. Declares:
290///
291/// - `findChallengeIframe(root)` — the first `challenges.cloudflare.com`
292///   iframe reachable from `root`, descending into open shadow roots
293///   (Cloudflare sometimes hosts the widget inside a shadow root), else null.
294/// - `clickableRect(el)` — `el`'s viewport rect *only if it is a real click
295///   target*: non-zero size and not hidden by `visibility` / `display` /
296///   `opacity`. A zero-size or hidden iframe is not a target — invisible
297///   Turnstile mounts a 0×0 iframe whose token is populated with no click,
298///   and clicking it would dispatch mouse events at a meaningless point.
299///
300/// Never evaluated on its own — [`main_world_expr`] wraps it, so nothing here
301/// reaches the page's global object. See that function for why.
302///
303/// # JavaScript style
304/// Injected scripts in this crate declare with `var` and iterate with
305/// indexed loops, never `for...of` — `detect.js` included. `for...of` over a
306/// `NodeList` enters through `NodeList.prototype[Symbol.iterator]`, a
307/// writable, configurable property the page can redefine to watch someone
308/// walk its DOM. That one is worth declining: it is a single chokepoint,
309/// entered once per loop and then driven by one `next()` per element, and a
310/// handler installed there can read `new Error().stack` and filter the
311/// page's own iteration out of what it collects. An indexed loop never
312/// reaches it.
313///
314/// It buys that one signal and no more. Indexed access is *not*
315/// unobservable: `NodeList.prototype.length` is a WebIDL accessor with
316/// `configurable: true`, so a page can wrap the getter, delegate to the
317/// original, and count every read with nothing of its own broken. The rest
318/// of the walk is louder. Each tick calls `root.querySelectorAll("iframe")`
319/// and then `root.querySelectorAll("*")`, both ordinary writable methods on
320/// `Document` / `Element` / `DocumentFragment`, and the iframe test below
321/// calls `String.prototype.includes` — equally patchable — with the literal
322/// `"challenges.cloudflare.com"`. A hook there is handed the hostname the
323/// walker is hunting for, which is a sharper tell than the iterator hook
324/// this convention avoids.
325///
326/// Measured on Chrome 151 with all four wrapped: the walk still returns the
327/// right iframe, while the page observes both `querySelectorAll` selectors,
328/// every `length` read and that hostname, against zero `Symbol.iterator`
329/// calls. So the convention removes one signal from a loop that stays
330/// observable to a page that looks for it; it is not cover. This is a rule
331/// about declarations and iteration only — modern methods are used freely,
332/// since declining them would buy nothing either.
333const WALKER_JS: &str = r#"
334function findChallengeIframe(root) {
335    var iframes = root.querySelectorAll ? root.querySelectorAll("iframe") : [];
336    for (var i = 0; i < iframes.length; i++) {
337        var f = iframes[i];
338        if (f.src && f.src.includes("challenges.cloudflare.com")) return f;
339    }
340    var all = root.querySelectorAll ? root.querySelectorAll("*") : [];
341    for (var j = 0; j < all.length; j++) {
342        if (all[j].shadowRoot) {
343            var sub = findChallengeIframe(all[j].shadowRoot);
344            if (sub) return sub;
345        }
346    }
347    return null;
348}
349function clickableRect(el) {
350    if (!el) return null;
351    var r = el.getBoundingClientRect();
352    if (!(r.width > 0 && r.height > 0)) return null;
353    var view = el.ownerDocument && el.ownerDocument.defaultView;
354    var style = view && view.getComputedStyle ? view.getComputedStyle(el) : null;
355    if (style && (style.visibility === "hidden" || style.display === "none" || style.opacity === "0")) {
356        return null;
357    }
358    return { x: r.left, y: r.top, width: r.width, height: r.height };
359}
360"#;
361
362/// Body of the unified poll evaluator — statements only, run inside
363/// [`main_world_expr`]'s wrapper alongside [`WALKER_JS`]. Returns:
364/// - `token` — non-empty `cf-turnstile-response` (or legacy
365///   `cf_challenge_response`) input value, else null.
366/// - `bbox` — the challenge iframe's rect when it is a valid click target,
367///   else null.
368/// - `hasMarkers` — true when *any* Cloudflare challenge marker is present
369///   (container, hidden input, or challenge iframe — visible or not).
370///
371/// The token input is in light DOM by design (page JS reads it to submit
372/// forms), so `document.querySelector` is sufficient there.
373const POLL_JS: &str = r#"
374    var iframe = findChallengeIframe(document);
375    var bbox = clickableRect(iframe);
376    var input =
377        document.querySelector('[name="cf-turnstile-response"]') ||
378        document.querySelector('[name="cf_challenge_response"]');
379    var token = (input && input.value) ? input.value : null;
380    var hasContainer = !!document.querySelector('.cf-turnstile, .turnstile, [data-sitekey]');
381    var hasMarkers = hasContainer || !!input || !!iframe;
382    return { token: token, bbox: bbox, hasMarkers: hasMarkers };
383"#;
384
385/// Body of the scroll-into-view evaluator — statements only, run inside
386/// [`main_world_expr`]'s wrapper alongside [`WALKER_JS`]. Brings the
387/// challenge iframe fully into the viewport when it isn't already, then
388/// returns its *post-scroll* rect — or null if the widget vanished or is not
389/// a valid click target.
390///
391/// `behavior: "instant"` is load-bearing: the default `"auto"` resolves to
392/// the element's computed `scroll-behavior`, so on a page setting
393/// `html { scroll-behavior: smooth }` the scroll animates and the
394/// `clickableRect` call on the next synchronous line reads the *pre*-scroll
395/// rect — the one thing this round-trip exists to avoid.
396const SCROLL_JS: &str = r#"
397    var iframe = findChallengeIframe(document);
398    if (!iframe) return null;
399    var r = iframe.getBoundingClientRect();
400    var vw = window.innerWidth || document.documentElement.clientWidth;
401    var vh = window.innerHeight || document.documentElement.clientHeight;
402    var fullyVisible = r.top >= 0 && r.left >= 0 && r.bottom <= vh && r.right <= vw;
403    if (!fullyVisible && iframe.scrollIntoView) {
404        iframe.scrollIntoView({ block: "center", inline: "center", behavior: "instant" });
405    }
406    return clickableRect(iframe);
407"#;

409/// Compose the in-page source for `body`: [`WALKER_JS`] and `body` together
410/// inside a single IIFE, whose completion value is whatever `body` returns.
411///
412/// The wrapper is a stealth requirement, not formatting. `Runtime.evaluate`
413/// with no `contextId` runs a *classic script* in the page's main world,
414/// where a top-level `function foo() {}` becomes a property of the global
415/// object. Evaluating `WALKER_JS` unwrapped therefore published
416/// `window.findChallengeIframe` and `window.clickableRect` to the page on
417/// every poll tick — names belonging to an automation library, sitting in
418/// the same realm as the challenge script, for anything enumerating `window`
419/// to find. Inside the IIFE they are ordinary locals and the page's globals
420/// are untouched.
421fn main_world_expr(body: &str) -> String {
422    format!("(function(){{{WALKER_JS}{body}}})()")
423}
424
425/// Run [`POLL_JS`] against `session`'s main world and decode the result.
426async fn poll_state(session: &SessionHandle) -> Result<PollState, CloudflareError> {
427    let value = eval_main_world(session, &main_world_expr(POLL_JS)).await?;
428    serde_json::from_value(value)
429        .map_err(|e| CloudflareError::JsError(format!("invalid poll payload: {e}")))
430}
431
432/// Run [`SCROLL_JS`] against `session`'s main world: scroll the challenge
433/// iframe into view if needed and return the rect to click, or `None` when
434/// the widget is gone or not clickable.
435async fn scroll_into_view(session: &SessionHandle) -> Result<Option<BoundingBox>, CloudflareError> {
436    let value = eval_main_world(session, &main_world_expr(SCROLL_JS)).await?;
437    if value.is_null() {
438        return Ok(None);
439    }
440    serde_json::from_value(value)
441        .map(Some)
442        .map_err(|e| CloudflareError::JsError(format!("invalid scroll payload: {e}")))
443}

514    /// Names every declaration in `src` at `{}` depth 0 that a classic
515    /// script publishes on the page's global object: `var` and *named*
516    /// `function`. Both become enumerable `window` properties (verified on
517    /// Chrome 151); `let` / `const` / `class` are script-scoped and never
518    /// do, so they are not leaks and are not counted. An anonymous
519    /// `function (` — the IIFE wrapper itself — is an expression and binds
520    /// nothing.
521    ///
522    /// Brace counting is only sound because no injected source puts a brace
523    /// inside a string literal. Keep it that way; the alternative is a JS
524    /// parser for a cookie-name-sized job. Comment text is scanned like any
525    /// other, which can only ever produce a loud false positive, never a
526    /// silent pass.
527    fn top_level_page_globals(src: &str) -> Vec<String> {
528        let is_ident = |c: char| c.is_alphanumeric() || c == '_' || c == '$';
529        let mut found = Vec::new();
530        let mut depth = 0i32;
531        for (i, ch) in src.char_indices() {
532            match ch {
533                '{' => depth += 1,
534                '}' => depth -= 1,
535                _ => {}
536            }
537            if depth != 0 {
538                continue;
539            }
540            // A keyword that is really the tail of a longer identifier
541            // declares nothing: `myvar`, `refunction`.
542            if src[..i].chars().next_back().is_some_and(is_ident) {
543                continue;
544            }
545            for keyword in ["function", "var"] {
546                let Some(rest) = src[i..].strip_prefix(keyword) else {
547                    continue;
548                };
549                // `var` needs a separator to be a declaration. `function`
550                // does not: `function(` is an expression, and the empty-name
551                // check below is what rejects it.
552                if keyword == "var" && !rest.starts_with(char::is_whitespace) {
553                    break;
554                }
555                let name: String = rest
556                    .trim_start()
557                    .chars()
558                    .take_while(|c| is_ident(*c))
559                    .collect();
560                if !name.is_empty() {
561                    found.push(name);
562                }
563                break;
564            }
565        }
566        found
567    }

660    /// that stays mounted (the poll evaluator keeps reporting a bbox) but
661    /// never becomes measurable (the scroll evaluator keeps returning null),
662    /// so no click ever lands. That tick used to skip the stall counter
663    /// entirely, which meant the one diagnostic explaining a stuck run could
664    /// never appear on the stuck run it explains.
665    #[tokio::test]
666    async fn stalled_unmeasurable_widget_emits_the_stealth_hint() {
667        /// Stalled ticks to serve before letting the run finish. The hint
668        /// fires on the tenth.
669        const STALLED_TICKS: usize = 12;
670
671        let (mut mock, conn) = MockConnection::pair();
672        let sess = SessionHandle::new(conn.clone(), "S1");
673        let capture = LogCapture::new();
674
675        let poll = poll_value(None, Some(rect(10.0, 20.0, 40.0, 40.0)), true);
676
677        // Not spawned: `LogCapture` follows the future it wraps, not a task
678        // spawned out of it, so the driver and the mock server share a task.
679        // The generous budget is never reached — the server ends the run — so
680        // the tick count is deterministic instead of clock-dependent.
681        let driver = capture.capture(async {
682            CloudflareBypass::new(&sess)
683                .poll_interval(Duration::from_millis(1))
684                .wait_for_clearance(Duration::from_secs(30))
685                .await
686        });
687
688        let serving = async {
689            let mut polls = 0usize;
690            while let Some((method, id)) = mock.recv_cmd_timeout(DRAIN_BUDGET).await {
691                assert_eq!(method, "Runtime.evaluate");
692                let is_scroll = mock.last_sent()["params"]["expression"]
693                    .as_str()
694                    .unwrap()
695                    .contains("scrollIntoView");
696                let value = if is_scroll {
697                    // Mounted on every poll, measurable on none: no click can
698                    // land, so every tick is a tick without progress.
699                    Value::Null
700                } else {
701                    polls += 1;
702                    if polls > STALLED_TICKS {
703                        poll_value(Some("LATE_TOKEN"), None, true)
704                    } else {
705                        poll.clone()
706                    }
707                };
708                mock.reply(id, eval_reply(value)).await;
709            }
710        };
711
712        let (outcome, ()) = tokio::join!(driver, serving);
713
714        match outcome.unwrap() {
715            ClearanceOutcome::TokenAcquired(t) => assert_eq!(t, "LATE_TOKEN"),
716            other => panic!("expected TokenAcquired, got {other:?}"),
717        }
718        assert!(
719            capture.contains("clearance stalled"),
720            "a mounted-but-unmeasurable widget is a stalled run; captured {:?}",
721            capture.events()
722        );
723        assert_eq!(
724            capture.count("clearance stalled"),
725            1,
726            "the hint is latched to one line per run"
727        );
728        conn.shutdown();
729    }
