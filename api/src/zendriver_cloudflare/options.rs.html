1//! Caller-supplied inputs to the bypass: which markers identify the widget,
2//! and how to click it.
3//!
4//! Cloudflare owns both. The class names, the hidden input's `name`, the
5//! iframe host and the checkbox's position inside the widget are all theirs
6//! to change, on their schedule, with no notice. A crate that bakes them in
7//! is a crate that needs a release every time they move, and every user is
8//! stuck on the old literals until it ships.
9//!
10//! So they are data here, not literals: [`TurnstileSelectors`] is what the
11//! injected evaluators are built from, and [`ClickPolicy`] decides whether
12//! and where to click. Both are plain structs with public fields — no
13//! setters, no closures — so they survive a trip through JSON and can be
14//! filled in by an agent, a config file, or a hotfix in the calling program.
15//! For the part that cannot be expressed as data at all, there is
16//! [`CloudflareBypass::on_click`](crate::CloudflareBypass::on_click).
17

20/// The page markers that identify a Cloudflare Turnstile widget.
21///
22/// Every injected evaluator is templated from these, so overriding one here
23/// changes the detector, the poll loop and the scroll-and-measure step
24/// together — there is no second copy to keep in sync.
25///
26/// ```no_run
27/// # async fn ex(tab: &zendriver_transport::SessionHandle)
28/// #   -> Result<(), zendriver_cloudflare::CloudflareError> {
29/// use std::time::Duration;
30/// use zendriver_cloudflare::{CloudflareBypass, TurnstileSelectors};
31///
32/// // Cloudflare moved the widget behind a new host: change the marker,
33/// // not the crate.
34/// let outcome = CloudflareBypass::new(tab)
35///     .selectors(TurnstileSelectors {
36///         iframe_src_contains: "challenges.example-cdn.com".into(),
37///         ..TurnstileSelectors::default()
38///     })
39///     .wait_for_clearance(Duration::from_secs(30))
40///     .await?;
41/// # let _ = outcome;
42/// # Ok(()) }
43/// ```
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub struct TurnstileSelectors {
46    /// Substring an `<iframe>`'s `src` must contain for that iframe to be
47    /// the challenge widget. Matched with `String.prototype.includes`, so a
48    /// bare hostname is enough. Defaults to `"challenges.cloudflare.com"`.
49    /// An empty string disables the iframe signal — it never matches — since
50    /// `includes("")` would otherwise match every iframe on the page and
51    /// send the click at whichever one came first.
52    pub iframe_src_contains: String,
53    /// CSS selector for the challenge *container* — the element a site
54    /// author drops into their own HTML for Turnstile to mount into. Used
55    /// only as a presence signal, which is what separates "a challenge was
56    /// on this page" from "this page has no gate at all". Defaults to
57    /// `".cf-turnstile, .turnstile, [data-sitekey]"`. An empty string
58    /// disables the container signal rather than raising in-page.
59    pub container: String,
60    /// CSS selectors for the hidden input that receives the clearance
61    /// token, in **preference order**: the first one that matches an
62    /// element wins, so a page carrying both the modern and the legacy
63    /// input is read the modern way. Defaults to
64    /// `["[name=\"cf-turnstile-response\"]", "[name=\"cf_challenge_response\"]"]`.
65    /// Empty entries are skipped; an empty list disables token detection,
66    /// leaving only the marker-vanished and deadline terminals.
67    pub token_inputs: Vec<String>,
68}
69
70impl Default for TurnstileSelectors {
71    fn default() -> Self {
72        Self {
73            iframe_src_contains: "challenges.cloudflare.com".to_string(),
74            container: ".cf-turnstile, .turnstile, [data-sitekey]".to_string(),
75            token_inputs: vec![
76                "[name=\"cf-turnstile-response\"]".to_string(),
77                "[name=\"cf_challenge_response\"]".to_string(),
78            ],
79        }
80    }
81}

83/// Whether, how often and where to click the interactive widget.
84///
85/// ```no_run
86/// # async fn ex(tab: &zendriver_transport::SessionHandle)
87/// #   -> Result<(), zendriver_cloudflare::CloudflareError> {
88/// use std::time::Duration;
89/// use zendriver_cloudflare::{ClickPolicy, CloudflareBypass};
90///
91/// // Watch, do not touch: the page's own script drives the widget.
92/// let outcome = CloudflareBypass::new(tab)
93///     .click_policy(ClickPolicy {
94///         max_attempts: 0,
95///         ..ClickPolicy::default()
96///     })
97///     .wait_for_clearance(Duration::from_secs(30))
98///     .await?;
99/// # let _ = outcome;
100/// # Ok(()) }
101/// ```
102#[derive(Debug, Clone, PartialEq)]
103pub struct ClickPolicy {
104    /// How many clicks one `wait_for_clearance` run may spend on the
105    /// widget. Cloudflare drops clicks that land while the widget is still
106    /// booting, so a single attempt strands the run until the deadline.
107    /// `0` disables clicking entirely — the run then watches only for a
108    /// token or for the markers to vanish. Defaults to `3`.
109    pub max_attempts: u32,
110    /// Poll ticks to wait after a click before spending another attempt.
111    /// Defaults to `4`, which is about two seconds at the default 500ms
112    /// poll interval.
113    pub retry_ticks: u32,
114    /// Where to click horizontally inside the widget's box, as a fraction
115    /// of its width. Defaults to `0.15` — 15% from the left edge, which is
116    /// where the Turnstile checkbox sits.
117    pub x_fraction: f64,
118    /// Where to click vertically inside the widget's box, as a fraction of
119    /// its height. Defaults to `0.50`.
120    pub y_fraction: f64,
121}
122
123impl Default for ClickPolicy {
124    fn default() -> Self {
125        Self {
126            max_attempts: 3,
127            retry_ticks: 4,
128            x_fraction: 0.15,
129            y_fraction: 0.50,
130        }
131    }
132}

134/// What a click is aimed at: the widget's post-scroll box, and the point
135/// inside it that the [`ClickPolicy`] chose.
136///
137/// Handed to a caller-supplied
138/// [`on_click`](crate::CloudflareBypass::on_click) handler, which may click
139/// [`x`](Self::x) / [`y`](Self::y) however it likes, or ignore them and work
140/// from [`bbox`](Self::bbox).
141#[derive(Debug, Clone, Copy, PartialEq)]
142pub struct ClickTarget {
143    /// The widget's bounding box in viewport CSS pixels, measured *after*
144    /// it was scrolled into view.
145    pub bbox: BoundingBox,
146    /// Viewport x of the chosen click point:
147    /// `bbox.x + bbox.width * policy.x_fraction`.
148    pub x: f64,
149    /// Viewport y of the chosen click point:
150    /// `bbox.y + bbox.height * policy.y_fraction`.
151    pub y: f64,
152}

154#[cfg(test)]
155#[allow(clippy::panic, clippy::unwrap_used)]
156mod tests {
157    use super::*;
158
159    /// Every default is stated in prose on its own field, repeated in the
160    /// book's Cloudflare chapter, and read by callers who never set it. This
161    /// is the one place that holds those numbers to what the docs claim —
162    /// the driver's own tests deliberately assert against literals rather
163    /// than against `Default`, so that changing a default here cannot
164    /// quietly move an expectation there too.
165    #[test]
166    fn defaults_match_the_documented_prose() {
167        let s = TurnstileSelectors::default();
168        assert_eq!(s.iframe_src_contains, "challenges.cloudflare.com");
169        assert_eq!(s.container, ".cf-turnstile, .turnstile, [data-sitekey]");
170        assert_eq!(
171            s.token_inputs,
172            vec![
173                "[name=\"cf-turnstile-response\"]".to_string(),
174                "[name=\"cf_challenge_response\"]".to_string(),
175            ],
176            "the modern input is preferred over the legacy one, so order matters"
177        );
178
179        let p = ClickPolicy::default();
180        assert_eq!(p.max_attempts, 3);
181        assert_eq!(p.retry_ticks, 4);
182        assert_eq!(p.x_fraction, 0.15);
183        assert_eq!(p.y_fraction, 0.50);
184    }
