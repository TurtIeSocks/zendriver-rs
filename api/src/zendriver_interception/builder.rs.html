1//! [`InterceptBuilder`] — fluent rule + pattern registration.
2//!
3//! Two-phase API:
4//! - **Configure**: chain [`block`], [`block_hosts`], [`redirect`], [`respond`],
5//!   [`modify_request`], [`modify_response`] for declarative rules, plus
6//!   [`pattern`] / [`at_request`] / [`at_response`] / [`resource`] to control
7//!   which CDP `Fetch.RequestPattern` entries are sent on `Fetch.enable`.
8//! - **Activate**: [`start`](InterceptBuilder::start) spawns the actor task
9//!   (T6) with the registered rules + patterns, returning an
10//!   [`InterceptHandle`] for RAII teardown. Alternatively,
11//!   [`subscribe`](InterceptBuilder::subscribe) returns a
12//!   `Stream<Item = PausedRequest>` for the manual escape-hatch path —
13//!   callers drive Chrome's interception loop themselves.
14//!
15//! The `tab` field is a borrow of [`SessionHandle`] (not the full `Tab` from
16//! `zendriver` core) — this crate must not depend on `zendriver` (cycle).
17//! `Tab::intercept()` in `zendriver` constructs the builder via
18//! `InterceptBuilder::new(self.session())`.
19//!
20//! [`block`]: InterceptBuilder::block
21//! [`block_hosts`]: InterceptBuilder::block_hosts
22//! [`redirect`]: InterceptBuilder::redirect
23//! [`respond`]: InterceptBuilder::respond
24//! [`modify_request`]: InterceptBuilder::modify_request
25//! [`modify_response`]: InterceptBuilder::modify_response
26//! [`pattern`]: InterceptBuilder::pattern
27//! [`at_request`]: InterceptBuilder::at_request
28//! [`at_response`]: InterceptBuilder::at_response
29//! [`resource`]: InterceptBuilder::resource
30

53/// A pending `Fetch.RequestPattern` entry to send on `Fetch.enable`.
54///
55/// CDP's [`Fetch.RequestPattern`] takes an optional `urlPattern`,
56/// `resourceType`, and `requestStage`. We mirror it 1:1 here. The builder
57/// accumulates these via [`InterceptBuilder::pattern`] / `at_request` /
58/// `at_response` / `resource`, mutating the last-pushed entry per chain — so
59/// `builder.pattern("*").at_response().resource(Image)` produces a single
60/// `RequestPattern` with all three fields set.
61///
62/// [`Fetch.RequestPattern`]: https://chromedevtools.github.io/devtools-protocol/tot/Fetch/#type-RequestPattern
63#[derive(Debug, Clone, Default)]
64pub struct RequestPattern {
65    /// URL pattern in CDP wildcard syntax. `None` means "match any URL"
66    /// (CDP default).
67    pub url_pattern: Option<String>,
68    /// Resource type filter (e.g. `Image`, `XHR`). `None` means "all types".
69    pub resource_type: Option<ResourceType>,
70    /// Lifecycle stage at which to pause. `None` means CDP's default
71    /// (`Request`).
72    pub request_stage: Option<RequestStage>,
73}

75/// Fluent builder for rule-based interception against a single tab session.
76///
77/// Construct via `Tab::intercept()` (gated `feature = "interception"`, wired
78/// in Task 7). Chain configuration methods to register rules and declare CDP
79/// `Fetch.enable` patterns, then call [`start`](Self::start) (Task 7) to
80/// activate the background actor or [`subscribe`](Self::subscribe) (Task 7)
81/// for the stream-driven escape hatch.
82///
83/// `'tab` ties the builder's lifetime to the tab's session — the borrow lasts
84/// only until `start()` / `subscribe()` consumes the builder.
85//
86// `Debug` works because `Rule` has a hand-written `Debug` impl that renders
87// the closure variant's body as `<closure>`. Inner `Vec<Rule>` derives via
88// that.
89#[derive(Debug)]
90pub struct InterceptBuilder<'tab> {
91    tab: &'tab SessionHandle,
92    patterns: Vec<RequestPattern>,
93    rules: Vec<Rule>,
94    /// Optional proxy/server credentials. When set, `Fetch.enable` is sent
95    /// with `handleAuthRequests: true` and the actor responds to each
96    /// `Fetch.authRequired` event with `Fetch.continueWithAuth` carrying
97    /// these credentials. See cdpdriver/zendriver#208.
98    auth: Option<(String, String)>,
99}
100
101impl<'tab> InterceptBuilder<'tab> {
102    /// Construct a fresh builder bound to `tab`'s session.
103    ///
104    /// `pub` so adapter crates (e.g. `zendriver` core's `Tab::intercept()`
105    /// shim) can construct it from a `&SessionHandle` without going through
106    /// a trait. End users go through `Tab::intercept()` rather than calling
107    /// this directly.
108    ///
109    /// ```no_run
110    /// # async fn ex(tab: &zendriver_transport::SessionHandle)
111    /// #   -> Result<(), zendriver_interception::InterceptionError> {
112    /// use zendriver_interception::InterceptBuilder;
113    ///
114    /// let _handle = InterceptBuilder::new(tab)
115    ///     .block("*/tracker.js")?
116    ///     .start();
117    /// # Ok(()) }
118    /// ```
119    #[must_use]
120    pub fn new(tab: &'tab SessionHandle) -> Self {
121        Self {
122            tab,
123            patterns: Vec::new(),
124            rules: Vec::new(),
125            auth: None,
126        }
127    }
128
129    /// Auto-respond to `Fetch.authRequired` challenges with the given
130    /// credentials.
131    ///
132    /// This is the proxy-auth (and HTTP basic-auth) path: `Fetch.enable` is
133    /// sent with `handleAuthRequests: true` and every `Fetch.authRequired`
134    /// event is answered with `Fetch.continueWithAuth { authChallengeResponse:
135    /// { response: "ProvideCredentials", username, password } }`.
136    ///
137    /// Compose with rules: an `InterceptBuilder` configured with `handle_auth`
138    /// and `block` / `redirect` / `respond` rules handles both paths from the
139    /// same actor. Combine with [`BrowserBuilder::proxy_auth`] in the
140    /// `zendriver` crate if you want the wiring installed automatically on
141    /// every tab.
142    ///
143    /// **Only [`start`](Self::start) honors this.** The
144    /// [`subscribe`](Self::subscribe) escape hatch has no auth path — it never
145    /// subscribes to `Fetch.authRequired`, so it leaves
146    /// `handleAuthRequests: false` and logs an error rather than pausing auth
147    /// challenges nothing would answer. Use `start()` when you need auth.
148    ///
149    /// See cdpdriver/zendriver#208.
150    #[must_use]
151    pub fn handle_auth(mut self, user: impl Into<String>, pass: impl Into<String>) -> Self {
152        self.auth = Some((user.into(), pass.into()));
153        self
154    }
155
156    /// Auto-respond to `Fetch.authRequired` challenges with the given
157    /// credentials, scoped to `Document`-type requests only.
158    ///
159    /// Equivalent to [`handle_auth`](Self::handle_auth) plus
160    /// `.pattern("*").resource(ResourceType::Document)`, packaged as its own
161    /// preset so a proxy/HTTP-auth-only caller doesn't have to discover — by
162    /// reading [`start`](Self::start)'s source — that leaving patterns unset
163    /// pauses *every* request on the page, not just the one carrying the auth
164    /// challenge.
165    ///
166    /// Narrowing to `Document` means only the main-frame navigation pauses and
167    /// round-trips through the interception actor; other requests (XHR,
168    /// subresources, and any parallel/pipelined connection that also needs
169    /// credentials) are not paused and will not have their auth challenge
170    /// answered by this builder. Use plain [`handle_auth`](Self::handle_auth)
171    /// (which falls back to a match-all pattern in [`start`](Self::start)) if
172    /// you need every connection's 407 covered.
173    #[must_use]
174    pub fn handle_auth_scoped_to_document(
175        self,
176        user: impl Into<String>,
177        pass: impl Into<String>,
178    ) -> Self {
179        self.handle_auth(user, pass)
180            .pattern("*")
181            .resource(ResourceType::Document)
182    }
183
184    /// Push a new pattern entry with the given URL pattern string.
185    ///
186    /// Subsequent [`at_request`](Self::at_request) /
187    /// [`at_response`](Self::at_response) / [`resource`](Self::resource) calls
188    /// mutate this newest entry, so a chain like
189    /// `.pattern("*").at_response().resource(ResourceType::XHR)` produces one
190    /// `RequestPattern` with all three fields populated.
191    #[must_use]
192    pub fn pattern(mut self, pattern: impl Into<String>) -> Self {
193        self.patterns.push(RequestPattern {
194            url_pattern: Some(pattern.into()),
195            ..RequestPattern::default()
196        });
197        self
198    }
199
200    /// Pause matching requests at the `Request` stage on the most-recently
201    /// pushed pattern.
202    ///
203    /// If no pattern has been pushed yet, this creates an empty one (matches
204    /// every URL by CDP default) and sets the stage on it.
205    #[must_use]
206    pub fn at_request(mut self) -> Self {
207        self.ensure_pattern().request_stage = Some(RequestStage::Request);
208        self
209    }
210
211    /// Pause matching requests at the `Response` stage on the most-recently
212    /// pushed pattern.
213    #[must_use]
214    pub fn at_response(mut self) -> Self {
215        self.ensure_pattern().request_stage = Some(RequestStage::Response);
216        self
217    }
218
219    /// Restrict the most-recently pushed pattern to a single resource type.
220    #[must_use]
221    pub fn resource(mut self, kind: ResourceType) -> Self {
222        self.ensure_pattern().resource_type = Some(kind);
223        self
224    }
225
226    /// Register a [`Rule::Block`] for `pattern`.
227    ///
228    /// Compiles `pattern` eagerly; an invalid pattern fails the builder chain
229    /// with [`InterceptionError::InvalidPattern`] returned as `Err(Self)` via
230    /// the `Result` wrapper.
231    pub fn block(mut self, pattern: impl Into<String>) -> Result<Self, InterceptionError> {
232        self.rules.push(Rule::Block {
233            pattern: UrlPattern::new(pattern)?,
234        });
235        Ok(self)
236    }
237
238    /// Register a [`Rule::BlockHosts`] backed by `matcher`.
239    ///
240    /// Every request whose host is in `matcher` (exact, or a parent domain on
241    /// a dot boundary) is failed with `BlockedByClient`. Composes with other
242    /// rules in registration order. `zendriver` core's tracker-blocklist
243    /// wiring uses this; most callers reach it via `BrowserBuilder::block_trackers`.
244    #[must_use]
245    pub fn block_hosts(mut self, matcher: Arc<HostMatcher>) -> Self {
246        self.rules.push(Rule::BlockHosts { matcher });
247        self
248    }
249
250    /// Register a [`Rule::Redirect`] that rewrites `from` → `to`.
251    pub fn redirect(
252        mut self,
253        from: impl Into<String>,
254        to: impl Into<String>,
255    ) -> Result<Self, InterceptionError> {
256        self.rules.push(Rule::Redirect {
257            from: UrlPattern::new(from)?,
258            to: to.into(),
259        });
260        Ok(self)
261    }
262
263    /// Register a [`Rule::Respond`] serving a synthesized response.
264    pub fn respond(
265        mut self,
266        pattern: impl Into<String>,
267        status: u16,
268        headers: Vec<(String, String)>,
269        body: Vec<u8>,
270    ) -> Result<Self, InterceptionError> {
271        self.rules.push(Rule::Respond {
272            pattern: UrlPattern::new(pattern)?,
273            status,
274            headers,
275            body,
276        });
277        Ok(self)
278    }

280    /// Register a [`Rule::Modify`] driven by a user closure.
281    ///
282    /// The closure runs on the actor task per matching request — it must be
283    /// `Send + Sync` and `'static`. Wrap shared state in `Arc` if needed.
284    pub fn modify_request<F>(
285        mut self,
286        pattern: impl Into<String>,
287        modify: F,
288    ) -> Result<Self, InterceptionError>
289    where
290        F: Fn(&RequestInfo) -> RequestOverrides + Send + Sync + 'static,
291    {
292        self.rules.push(Rule::Modify {
293            pattern: UrlPattern::new(pattern)?,
294            modify: Arc::new(modify),
295        });
296        Ok(self)
297    }
298
299    /// Register a [`Rule::ModifyResponse`] driven by a user closure.
300    ///
301    /// The closure rewrites an upstream response's status/headers (keeping
302    /// Chrome's body) and only fires at the `Response` stage — pair this with
303    /// [`at_response`](Self::at_response) so Chrome actually pauses there.
304    /// Header overrides are *replacement*, not merge (CDP semantics): return
305    /// every header you want forwarded.
306    ///
307    /// Like [`modify_request`](Self::modify_request), the closure runs on the
308    /// actor task per matching response, so it must be `Send + Sync` and
309    /// `'static`. Wrap shared state in `Arc` if needed.
310    pub fn modify_response<F>(
311        mut self,
312        pattern: impl Into<String>,
313        modify: F,
314    ) -> Result<Self, InterceptionError>
315    where
316        F: Fn(&ResponseInfo) -> ResponseOverrides + Send + Sync + 'static,
317    {
318        self.rules.push(Rule::ModifyResponse {
319            pattern: UrlPattern::new(pattern)?,
320            modify: Arc::new(modify),
321        });
322        Ok(self)
323    }

325    /// Activate the rule-based interception loop.
326    ///
327    /// Spawns the background actor task with the registered rules and CDP
328    /// `RequestPattern` list, and returns an [`InterceptHandle`] whose
329    /// [`Drop`] (or explicit [`stop`](InterceptHandle::stop)) tears the
330    /// actor down.
331    ///
332    /// If no [`pattern`](Self::pattern) entries were added, a single
333    /// match-all (`"*"`) pattern is sent so Chrome actually pauses requests
334    /// — without it, `Fetch.enable` would attach to nothing and the rule
335    /// list would never fire.
336    #[must_use = "interception stops when the handle is dropped — bind the returned InterceptHandle to keep it alive"]
337    pub fn start(mut self) -> InterceptHandle {
338        if self.patterns.is_empty() && (!self.rules.is_empty() || self.auth.is_some()) {
339            // A match-all pattern is required in two cases:
340            //
341            // 1. Rules need Chrome to actually pause requests — without a
342            //    pattern `Fetch.enable` attaches to nothing and every rule is a
343            //    silent no-op.
344            // 2. `handleAuthRequests: true` (the proxy / HTTP-auth path):
345            //    Chrome (~M132+) REJECTS `Fetch.enable { patterns: [],
346            //    handleAuthRequests: true }` with CDP error -32602 "Can't
347            //    specify empty patterns with handleAuth set". Older Chrome
348            //    tolerated empty patterns — auth challenges still surfaced while
349            //    pausing zero requests — so we used to leave them empty on the
350            //    auth-ONLY path to avoid distorting the page's XHR cadence
351            //    (reese84 / Incapsula read it). Modern Chrome no longer permits
352            //    that, so a pattern is mandatory whenever auth handling is on.
353            //
354            // On the auth-only path this pauses (and immediately auto-continues
355            // — see `handle_paused`'s no-match arm) every request. If that
356            // per-request CDP round-trip regresses a timing-sensitive bot
357            // sensor, a caller can register an explicit narrower pattern (e.g.
358            // `resource_type: Document`, so only the main navigation — which
359            // carries the first proxy 407 — is paused), at the risk of missing
360            // 407s on parallel proxy connections.
361            self.patterns.push(RequestPattern {
362                url_pattern: Some("*".into()),
363                ..RequestPattern::default()
364            });
365        }
366        let cancel = CancellationToken::new();
367        let (done_tx, done_rx) = oneshot::channel();
368        let actor_session = self.tab.clone();
369        let actor_cancel = cancel.clone();
370        let actor_rules = self.rules;
371        let actor_patterns = self.patterns;
372        let actor_auth = self.auth;
373        tokio::spawn(async move {
374            run_actor(
375                actor_session,
376                actor_rules,
377                actor_patterns,
378                actor_auth,
379                actor_cancel,
380                done_tx,
381            )
382            .await;
383        });
384        InterceptHandle::new(cancel, done_rx)
385    }

387    /// Manual escape-hatch: subscribe to raw [`PausedRequest`] events.
388    ///
389    /// Enables `Fetch` interception with the declared patterns (defaulting
390    /// to a single match-all `"*"` pattern when none were added) and returns
391    /// a [`Stream`] that yields one [`PausedRequest`] per `Fetch.requestPaused`
392    /// CDP event. Callers must dispatch one of `PausedRequest`'s terminal
393    /// methods (`continue_` / `abort` / `respond` / `modify_and_continue`)
394    /// to release each pause — Chrome holds the request open otherwise.
395    ///
396    /// Rules registered via `block` / `redirect` / `respond` / `modify_request`
397    /// are ignored on this path: stream consumers drive every paused request
398    /// themselves. Use [`start`](Self::start) when you want the actor to
399    /// apply rules automatically.
400    ///
401    /// [`handle_auth`](Self::handle_auth) credentials are **also ignored**
402    /// here, and calling both logs an error. This path never subscribes to
403    /// `Fetch.authRequired`, so `Fetch.enable` is deliberately sent with
404    /// `handleAuthRequests: false`: flipping it on would make Chrome pause
405    /// every auth challenge with nobody to answer it, turning a visible 407
406    /// into a hang. Behind a proxy that requires credentials, use
407    /// [`start`](Self::start) instead.
408    ///
409    /// The returned stream owns the underlying CDP subscription. Dropping
410    /// the stream tears the subscription down — Chrome's interception stays
411    /// active until the session is closed, but no further pauses surface to
412    /// the caller.
413    #[must_use = "the returned stream is the only handle on the subscription"]
414    pub fn subscribe(mut self) -> impl Stream<Item = PausedRequest> + Send + use<> {
415        if self.auth.is_some() {
416            // Loud, because the alternative failure is invisible: behind a
417            // proxy every navigation 407s and the caller sees only empty
418            // pages. We cannot honor the credentials here — this path has no
419            // `Fetch.authRequired` subscription, and enabling
420            // `handleAuthRequests` without one converts the 407 into a hang.
421            tracing::error!(
422                "interception: handle_auth() credentials are ignored by subscribe(); \
423                 auth challenges will NOT be answered — use start() for proxy / HTTP auth"
424            );
425        }
426        if self.patterns.is_empty() {
427            self.patterns.push(RequestPattern {
428                url_pattern: Some("*".into()),
429                ..RequestPattern::default()
430            });
431        }
432        // Same ordering as the actor: subscribe BEFORE the (fire-and-forget)
433        // enable so we don't drop events Chrome emits between the enable
434        // round-trip and the subscription registration.
435        let raw = self.tab.subscribe::<Value>("Fetch.requestPaused");
436        let session = self.tab.clone();
437        let enable_session = session.clone();
438        let enable_patterns: Vec<Value> = self.patterns.iter().map(serialize_pattern).collect();
439        tokio::spawn(async move {
440            if let Err(e) = enable_session
441                .call(
442                    "Fetch.enable",
443                    json!({
444                        "patterns": enable_patterns,
445                        // Always false on this path — see the doc comment: no
446                        // `Fetch.authRequired` subscription exists here, so
447                        // `true` would pause challenges forever.
448                        "handleAuthRequests": false,
449                    }),
450                )
451                .await
452            {
453                warn!(error = %e, "interception: Fetch.enable failed; subscribe() stream will be empty");
454            }
455        });
456        raw.filter_map(move |ev_value| {
457            let session = session.clone();
458            async move {
459                let ev: RequestPausedEvent = match serde_json::from_value(ev_value) {
460                    Ok(ev) => ev,
461                    Err(e) => {
462                        warn!(error = %e, "interception: skipping malformed Fetch.requestPaused event");
463                        return None;
464                    }
465                };
466                let info = build_request_info(&ev);
467                let response = build_response_info(&ev);
468                Some(PausedRequest::new(ev.request_id, info, response, session))
469            }
470        })
471    }

473    /// Lazily push an empty pattern if none exists, so the stage/resource
474    /// setters always have a target. Mirrors CDP's "missing fields default to
475    /// match-all" semantics.
476    fn ensure_pattern(&mut self) -> &mut RequestPattern {
477        if self.patterns.is_empty() {
478            self.patterns.push(RequestPattern::default());
479        }
480        self.patterns
481            .last_mut()
482            .expect("ensure_pattern pushed if empty")
483    }

526    /// End-to-end on the rule-driven `start()` path: register a Block rule,
527    /// spawn the actor via `start()`, observe `Fetch.enable`, emit a matching
528    /// `Fetch.requestPaused`, and assert `Fetch.failRequest` is dispatched.
529    ///
530    /// This is the actor test from T6 reframed through the `start()` entry
531    /// point — proves the builder properly forwards rules + patterns to
532    /// `run_actor` (and that `start()` actually spawns the task).
533    #[tokio::test]
534    async fn start_spawns_actor_with_rules() {
535        let (mut mock, conn) = MockConnection::pair();
536        let sess = SessionHandle::new(conn.clone(), "S1");
537
538        let handle = InterceptBuilder::new(&sess)
539            .block("*/blocked/*")
540            .unwrap()
541            .pattern("*")
542            .start();
543
544        // The actor's `Fetch.enable` side-task fires fire-and-forget; wait
545        // for it to land so the `Fetch.requestPaused` subscription is in
546        // place before we emit an event.
547        let enable_id =
548            tokio::time::timeout(Duration::from_secs(2), mock.expect_cmd("Fetch.enable"))
549                .await
550                .expect("actor did not send Fetch.enable within 2s");
551        let enable_params = mock.last_sent()["params"].clone();
552        assert_eq!(enable_params["handleAuthRequests"], false);
553        assert_eq!(enable_params["patterns"][0]["urlPattern"], "*");
554        mock.reply(enable_id, json!({})).await;
555
556        // Emit a paused-event whose URL matches the Block rule.
557        mock.emit_event_for_session(
558            "Fetch.requestPaused",
559            json!({
560                "requestId": "REQ-1",
561                "request": {
562                    "url": "https://example.test/blocked/banner.png",
563                    "method": "GET",
564                    "headers": {},
565                },
566                "resourceType": "Image",
567            }),
568            "S1",
569        )
570        .await;
571
572        // Actor should dispatch Fetch.failRequest with BlockedByClient.
573        let fail_id =
574            tokio::time::timeout(Duration::from_secs(2), mock.expect_cmd("Fetch.failRequest"))
575                .await
576                .expect("actor did not send Fetch.failRequest within 2s");
577        let fail_params = mock.last_sent()["params"].clone();
578        assert_eq!(fail_params["requestId"], "REQ-1");
579        assert_eq!(fail_params["errorReason"], "BlockedByClient");
580        mock.reply(fail_id, json!({})).await;
581
582        // Teardown via the handle: stop() cancels + awaits the oneshot the
583        // actor signals after Fetch.disable lands.
584        let stop_fut = tokio::spawn(handle.stop());
585        let disable_id =
586            tokio::time::timeout(Duration::from_secs(2), mock.expect_cmd("Fetch.disable"))
587                .await
588                .expect("actor did not send Fetch.disable on stop()");
589        mock.reply(disable_id, json!({})).await;
590        stop_fut
591            .await
592            .expect("stop() task panicked")
593            .expect("stop() returned Err");
594        conn.shutdown();

731    /// On the `subscribe()` path: each `Fetch.requestPaused` event becomes
732    /// a `PausedRequest` yielded from the stream, with the request payload
733    /// decoded into [`RequestInfo`].
734    #[tokio::test]
735    async fn subscribe_yields_paused_request_per_event() {
736        let (mut mock, conn) = MockConnection::pair();
737        let sess = SessionHandle::new(conn.clone(), "S1");
738
739        let mut stream = Box::pin(InterceptBuilder::new(&sess).subscribe());
740
741        // Wait for the side-task's Fetch.enable to land so the subscription
742        // is in place before we emit.
743        let enable_id =
744            tokio::time::timeout(Duration::from_secs(2), mock.expect_cmd("Fetch.enable"))
745                .await
746                .expect("subscribe() did not send Fetch.enable within 2s");
747        mock.reply(enable_id, json!({})).await;
748
749        mock.emit_event_for_session(
750            "Fetch.requestPaused",
751            json!({
752                "requestId": "REQ-1",
753                "request": {
754                    "url": "https://example.test/widget.json",
755                    "method": "GET",
756                    "headers": {"accept": "application/json"},
757                },
758                "resourceType": "XHR",
759            }),
760            "S1",
761        )
762        .await;
763
764        let paused = tokio::time::timeout(Duration::from_secs(2), stream.next())
765            .await
766            .expect("subscribe() stream did not yield within 2s")
767            .expect("subscribe() stream closed before yielding");

772            paused

777                .map(|(_, v)| v.as_str()),
778            Some("application/json"),
779        );
780        assert!(
781            paused.response.is_none(),
782            "request-stage event has no response"
783        );
784
785        drop(stream);
786        conn.shutdown();
