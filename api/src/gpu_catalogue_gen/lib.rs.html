1//! Generator for the GPU device catalogue.
2//!
3//! Emits `crates/zendriver-stealth/src/gpu/catalogue.rs` from two pinned
4//! sources: driver-reported model names out of the fingerprint corpus, and PCI
5//! device IDs out of `pci.ids`. Nothing here invents a value.
6//!
7//! The catalogue supplies *identity* only — a renderer string, a device ID, an
8//! architecture token. Every capability number still comes from the measured
9//! tier tables that `gpu-tier-gen` emits, which is what keeps this generator
10//! from being able to fabricate a capability at all.
11
12pub mod sources;

14/// One `<vendor id>:<device id>` pair and the name `pci.ids` gives it.
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub struct PciDevice {
17    pub vendor_id: u32,
18    pub device_id: u32,
19    pub name: String,
20}
21
22/// Parse the `pci.ids` format.
23///
24/// The shape is three levels deep, distinguished only by leading tabs:
25///
26/// ```text
27/// 10de  NVIDIA Corporation            <- vendor
28/// \t2684  AD102 [GeForce RTX 4090]    <- device
29/// \t\t1043 87b3  TUF Gaming           <- subsystem
30/// ```
31///
32/// Only the middle level carries a device ID. Subsystem lines are a
33/// `(subvendor, subdevice)` pair, so reading one as a device invents a card
34/// that does not exist.
35///
36/// The file then ends with device-*class* sections, whose headings are
37/// unindented like vendors and whose subclass lines are indented exactly like
38/// devices:
39///
40/// ```text
41/// C 03  Display controller
42/// \t00  VGA compatible controller
43/// ```
44///
45/// `C 03` is not hex, so the current vendor is **cleared** rather than left
46/// alone. Leaving it alone would file every subclass in the file under
47/// whichever vendor happened to come last.
48#[must_use]
49pub fn parse_pci_ids(raw: &str) -> Vec<PciDevice> {
50    let mut out = Vec::new();
51    let mut vendor_id: Option<u32> = None;
52
53    for line in raw.lines() {
54        if line.trim().is_empty() || line.trim_start().starts_with('#') {
55            continue;
56        }
57
58        if let Some(rest) = line.strip_prefix('\t') {
59            // Two tabs: a subsystem pair, which has no device ID of its own.
60            if rest.starts_with('\t') {
61                continue;
62            }
63            let (Some(vendor), Some((id, name))) = (vendor_id, rest.split_once("  ")) else {
64                continue;
65            };
66            if let Ok(device_id) = u32::from_str_radix(id.trim(), 16) {
67                out.push(PciDevice {
68                    vendor_id: vendor,
69                    device_id,
70                    name: name.trim().to_string(),
71                });
72            }
73            continue;
74        }
75
76        // Unindented: a vendor heading, or a class heading that is not one.
77        vendor_id = line
78            .split_once("  ")
79            .and_then(|(id, _)| u32::from_str_radix(id.trim(), 16).ok());
80    }
81    out
82}

87pub enum Backend {
88    D3d11,
89    Metal,
90}

92/// Build the renderer string ANGLE would report for a device.
93///
94/// Reconstructed from the source that composes it rather than copied from
95/// samples, because the format has changed under the corpus: strings collected
96/// on older Chrome omit the device ID current ANGLE always appends. Composing
97/// means a future format change is a small, detectable fix here instead of
98/// silent rot spread across every row of the data.
99///
100/// D3D11, `Renderer11.cpp:2308-2319`:
101///
102/// ```text
103/// mDescription << " (" << FmtHex(DeviceId) << ")" << " Direct3D11"
104///              << " vs_" << major << "_" << minor << " ps_" << major << "_" << minor
105/// ```
106///
107/// Metal, `DisplayMtl.mm:188-201`:
108///
109/// ```text
110/// "ANGLE Metal Renderer" + ": " + MTLDevice.name
111/// ```
112///
113/// The trailing Metal field is the literal `getVersionString` returns for WebGL
114/// contexts (`:216`) because Chrome requires something there. It is not a
115/// version and must never be synthesized as one.
116///
117/// `device_id` is ignored on Metal: Apple silicon exposes no PCI ID, and the
118/// string has nowhere to put one.
119#[must_use]
120pub fn compose_renderer(
121    backend: Backend,
122    vendor: &str,
123    model: &str,
124    device_id: Option<u32>,
125) -> String {
126    match backend {
127        Backend::D3d11 => {
128            let id = device_id.unwrap_or(0);
129            format!("ANGLE ({vendor}, {model} (0x{id:08X}) Direct3D11 vs_5_0 ps_5_0, D3D11)")
130        }
131        Backend::Metal => {
132            format!("ANGLE ({vendor}, ANGLE Metal Renderer: {model}, Unspecified Version)")
133        }
134    }
135}

137/// One device the corpus reports. The renderer string itself is rebuilt by
138/// [`compose_renderer`].
139#[derive(Debug, Clone, PartialEq)]
140pub struct CorpusModel {
141    pub backend: Backend,
142    pub vendor: String,
143    pub model: String,
144    /// The PCI device id the corpus observed **this name paired with**, when
145    /// the string carried one.
146    ///
147    /// This is an observation, not a lookup: 467 such pairs cover 308 of the
148    /// 310 D3D11 names, and 77 names appear with several ids because one
149    /// marketing name really does span several SKUs. Preferring these over a
150    /// `pci.ids` name match is what turns a deterministic guess among 30
151    /// candidates into a pairing some real machine actually reported.
152    pub device_id: Option<u32>,
153    /// Marginal probability of this device across the whole corpus.
154    ///
155    /// `videoCard` is conditioned on `userAgent`, so a bucket's numbers are
156    /// `P(device | ua)` and summing them naively would over-weight anything
157    /// that appears under many user agents. `userAgent` is parentless and its
158    /// prior sums to 1, so the honest weight is
159    /// `Σ_ua P(ua) · P(device | ua)`.
160    pub weight: f64,
161}
162
163/// Key that orders [`CorpusModel`]s deterministically. `weight` is an `f64`,
164/// so the struct cannot derive `Ord`.
165fn model_key(m: &CorpusModel) -> (Backend, &str, &str, Option<u32>) {
166    (m.backend, m.vendor.as_str(), m.model.as_str(), m.device_id)
167}

169/// Reduce a model name to the form both sides of the `pci.ids` join agree on.
170///
171/// Two mistakes are already paid for here, so do not simplify them away.
172/// Writing the `(tm)`/`(r)` strip as `\b\(tm\)\b` matches nothing, because a
173/// word boundary before a parenthesis is not where it looks like it is; that
174/// alone dropped Intel's join rate to zero. The vendor words have to go too,
175/// because the driver leads with them (`NVIDIA GeForce RTX 3070`) and
176/// `pci.ids` does not (`GA104 [GeForce RTX 3070]`).
177#[must_use]
178pub fn normalize_model(s: &str) -> String {
179    const DROP_WORDS: &[&str] = &[
180        "nvidia",
181        "amd",
182        "ati",
183        "intel",
184        "corporation",
185        "inc",
186        "series",
187    ];
188    let lower = s
189        .to_ascii_lowercase()
190        .replace("(tm)", "")
191        .replace("(r)", "");
192    let mut out = String::new();
193    for word in lower
194        .split(|c: char| !c.is_ascii_alphanumeric())
195        .filter(|w| !w.is_empty())
196    {
197        if DROP_WORDS.contains(&word) {
198            continue;
199        }
200        if !out.is_empty() {
201            out.push(' ');
202        }
203        out.push_str(word);
204    }
205    out
206}

208/// Every marketing name one `pci.ids` entry covers.
209///
210/// A single entry routinely stands for several cards:
211/// `Navi 22 [Radeon RX 6700/6700 XT/6750 XT]`. Only the first slash segment
212/// carries the `Radeon RX` prefix, so each later segment is tried both bare and
213/// with that prefix prepended. Skipping this costs about a third of AMD.
214fn pci_aliases(name: &str) -> Vec<String> {
215    let mut out = vec![normalize_model(name)];
216    let mut rest = name;
217    while let Some(open) = rest.find('[') {
218        let Some(close) = rest[open..].find(']') else {
219            break;
220        };
221        let bracket = &rest[open + 1..open + close];
222        rest = &rest[open + close..];
223
224        out.push(normalize_model(bracket));
225        let parts: Vec<&str> = bracket
226            .split('/')
227            .map(str::trim)
228            .filter(|p| !p.is_empty())
229            .collect();
230        if parts.len() > 1 {
231            // "Radeon RX 6700" lends "Radeon RX" to "6700 XT" and "6750 XT".
232            let head: Vec<&str> = parts[0].split_whitespace().collect();
233            let prefix = head[..head.len().saturating_sub(1)].join(" ");
234            for part in &parts {
235                out.push(normalize_model(part));
236                if !prefix.is_empty() {
237                    out.push(normalize_model(&format!("{prefix} {part}")));
238                }
239            }
240        }
241    }
242    out.retain(|a| !a.is_empty());
243    out.sort();
244    out.dedup();
245    out
246}

251/// A catalogue row on its way to the generated file.
252///
253/// Strings rather than the stealth crate's enums, because this generator does
254/// not depend on it: `tier` is rendered as `Tier::{tier}`.
255#[derive(Debug, Clone, PartialEq)]
256pub struct CatalogueRow {
257    pub model: String,
258    pub vendor: String,
259    pub device_id: Option<u32>,
260    pub tier: String,
261    /// Marginal share of this device in the corpus population, for the
262    /// share-weighted draw. Browser population rather than gamer population,
263    /// which is the one a browser-automation tool wants.
264    pub weight: f64,
265    /// Where the device id came from. Nothing is emitted without one on
266    /// D3D11, so this records which of the two grounded sources supplied it.
267    pub id_source: IdSource,
268}
269
270/// Provenance of a row's device id.
271#[derive(Debug, Clone, Copy, PartialEq, Eq)]
272pub enum IdSource {
273    /// The corpus reported this exact `(name, id)` pair. Preferred: it is an
274    /// observation of a real machine rather than a reconstruction.
275    Observed,
276    /// No corpus entry for this name carried an id, so it was resolved by
277    /// name against `pci.ids`. Rare — 2 of 310 names.
278    PciIds,
279    /// Metal, which has no PCI id at all.
280    None,
281}

283/// What a `build_catalogue` run kept and what it had to drop.

285pub struct BuildReport {
286    pub rows: Vec<CatalogueRow>,
287    /// D3D11 models with no `pci.ids` match, so no device ID, so no composable
288    /// renderer string. Reported rather than silently swallowed.
289    pub unmatched: Vec<String>,

292/// Which measured capability tier a corpus model belongs to.
293///
294/// Two of the three D3D11 answers are *measured splits*, each found by probing
295/// a second device and diffing rather than predicted from ANGLE's source:
296///
297/// - **NVIDIA.** `ANGLE_FEATURE_CONDITION(features, skipVSConstantRegisterZero,
298///   isNvidia)` docks `maxVertexUniformVectors` by one, so NVIDIA parts report
299///   4095 where every other vendor reports 4096. The condition is the vendor
300///   and nothing else, which is why this arm can be a plain vendor test.
301/// - **Intel Gen9.** Two D3D11 values are read off the device rather than off
302///   the feature level — `MAX_SAMPLES`, which ANGLE fills by asking
303///   `CheckMultisampleQualityLevels` per renderable format, and the WebGPU
304///   feature list, which is Dawn's answer for the physical adapter. An Intel HD
305///   Graphics 520 measured 16 and 16 entries against the RDNA2 capture's 8 and
306///   19.
307///
308/// **Only Gen9 moves, and that is deliberate.** [`intel_generation`] recognises
309/// Gen12 as well, and Gen12 is emphatically *not* routed here: nobody has
310/// probed it, both of the values that moved are device-derived, and Iris Xe is
311/// the single heaviest entry in this catalogue — so guessing would be wrong at
312/// the largest scale available. Gen11 and Gen12 stay on the catch-all tier
313/// until someone captures them.
314fn tier_for(backend: Backend, vendor: &str, model: &str) -> &'static str {
315    match (backend, vendor) {
316        (Backend::Metal, _) => "MetalMacos",
317        (Backend::D3d11, "NVIDIA") => "D3d11Fl11Nvidia",
318        (Backend::D3d11, _) if intel_generation(model) == "gen-9" => "D3d11Fl11IntelGen9",
319        (Backend::D3d11, _) => "D3d11Fl11",
320    }
321}

323/// Intel's integrated-graphics generation for a driver-reported model name, as
324/// Dawn's `gpu_info.json` names it. `""` for anything it cannot place.
325///
326/// **A copy of `zendriver-stealth`'s `intel_architecture`, and it must stay
327/// one.** That function is private to a crate this generator does not depend on
328/// — the dependency runs the other way, since this emits that crate's
329/// `catalogue.rs` — so the predicate cannot literally be shared. What keeps the
330/// two from drifting is a test rather than discipline:
331/// `every_catalogued_device_round_trips_to_its_own_tier` composes every emitted
332/// row's renderer string and asserts the stealth crate resolves the tier this
333/// function assigned it. Classify a model differently on either side and the
334/// committed catalogue stops round-tripping.
335///
336/// It reads a model name where the stealth side reads a whole renderer string.
337/// That is the same test: the model is a substring of the renderer, and every
338/// token here names part of a model.
339fn intel_generation(model: &str) -> &'static str {
340    let r = model.to_ascii_lowercase();
341    if r.contains("iris xe") || r.contains("xe graphics") {
342        return "gen-12-lp";
343    }
344    match hd_family_number(&r) {
345        Some(500..=699) => "gen-9",
346        _ => "",
347    }
348}

350/// The `HD`/`UHD Graphics` family number, but only when it is three digits.
351///
352/// Mirrors the stealth crate's helper of the same name. Gen9 spelled its
353/// families with three digits where Broadwell (Gen8) used four, so counting the
354/// digit run is what keeps `HD Graphics 5500` off a tier measured on Skylake.
355fn hd_family_number(model_lower: &str) -> Option<u16> {
356    const TOKEN: &str = "hd graphics ";
357    let start = model_lower.find(TOKEN)? + TOKEN.len();
358    let digits: String = model_lower[start..]
359        .chars()
360        .take_while(char::is_ascii_digit)
361        .collect();
362    (digits.len() == 3).then(|| digits.parse().ok())?
363}

365/// Join corpus models against `pci.ids` and produce the catalogue rows.
366///
367/// A D3D11 renderer string carries its device ID by construction, so a model
368/// that fails to join **cannot be composed at all** and is dropped. Inventing
369/// the number is the one thing this design forbids, and a placeholder would be
370/// a fabricated fingerprint value wearing a real card's name.
371///
372/// Metal needs no join: Apple silicon exposes no PCI ID and the string has
373/// nowhere to put one.
374#[must_use]
375pub fn build_catalogue(network_json: &str, pci_ids: &str) -> BuildReport {
376    let devices = parse_pci_ids(pci_ids);
377    let mut by_vendor: std::collections::BTreeMap<u32, Vec<(&PciDevice, Vec<String>)>> =
378        std::collections::BTreeMap::new();
379    for d in &devices {
380        if VENDOR_IDS.iter().any(|(_, id)| *id == d.vendor_id) {
381            by_vendor
382                .entry(d.vendor_id)
383                .or_default()
384                .push((d, pci_aliases(&d.name)));
385        }
386    }
387
388    let mut report = BuildReport::default();
389    for m in extract_models(network_json) {
390        let tier = tier_for(m.backend, &m.vendor, &m.model);
391
392        if m.backend == Backend::Metal {
393            report.rows.push(CatalogueRow {
394                model: m.model,
395                vendor: m.vendor,
396                device_id: None,
397                tier: tier.to_string(),
398                weight: m.weight,
399                id_source: IdSource::None,
400            });
401            continue;
402        }
403
404        // Prefer what the corpus observed. A pci.ids name match can only
405        // reconstruct a plausible id -- `Intel(R) Graphics` alone matches 30
406        // candidates -- whereas the corpus reports the pair a real machine
407        // actually emitted, and reports every SKU a name spans rather than
408        // one representative of them.
409        if let Some(device_id) = m.device_id {
410            report.rows.push(CatalogueRow {
411                model: m.model,
412                vendor: m.vendor,
413                device_id: Some(device_id),
414                tier: tier.to_string(),
415                weight: m.weight,
416                id_source: IdSource::Observed,
417            });
418            continue;
419        }
420
421        // Fallback for the few names no corpus entry ever paired with an id.
422        // Lowest id, so bumping the pinned pci.ids cannot reshuffle it.
423        let vendor_id = VENDOR_IDS
424            .iter()
425            .find(|(v, _)| *v == m.vendor)
426            .map(|(_, id)| id);
427        let target = normalize_model(&m.model);
428        let hit = vendor_id
429            .and_then(|id| by_vendor.get(id))
430            .into_iter()
431            .flatten()
432            .filter(|(_, aliases)| !target.is_empty() && aliases.contains(&target))
433            .min_by_key(|(device, _)| device.device_id);
434        match hit {
435            Some((device, _)) => report.rows.push(CatalogueRow {
436                model: m.model,
437                vendor: m.vendor,
438                device_id: Some(device.device_id),
439                tier: tier.to_string(),
440                weight: m.weight,
441                id_source: IdSource::PciIds,
442            }),
443            // A D3D11 renderer string carries its device id by construction,
444            // so a row without one cannot be composed at all. Dropped, named.
445            None => report.unmatched.push(m.model),
446        }
447    }
448    report
449        .rows
450        .sort_by(|a, b| (&a.tier, &a.model, a.device_id).cmp(&(&b.tier, &b.model, b.device_id)));
451    report.unmatched.sort();
452    report.unmatched.dedup();
453    report
454}

456/// Pull the single JSON member out of the corpus zip.
457///
458/// # Errors
459/// If the archive cannot be read or holds no `.json` member.
460pub fn unzip_network_json(zipped: &[u8]) -> Result<String, Box<dyn std::error::Error>> {
461    let mut archive = zip::ZipArchive::new(std::io::Cursor::new(zipped))?;
462    let name = (0..archive.len())
463        .filter_map(|i| archive.by_index(i).ok().map(|f| f.name().to_string()))
464        .find(|n| n.ends_with(".json"))
465        .ok_or("corpus archive holds no .json member")?;
466    let mut out = String::new();
467    std::io::Read::read_to_string(&mut archive.by_name(&name)?, &mut out)?;
468    Ok(out)
469}

471/// Render the committed catalogue module.
472///
473/// Data only: `CatalogueEntry` itself is hand-written in `gpu::types`, the same
474/// split `gpu-tier-gen` uses for `tiers.rs`. A generated file that also defines
475/// its own types cannot be reviewed for shape independently of its contents.
476///
477/// Renderer strings are **not** stored. They are composed at read time from
478/// ANGLE's format, so a Chrome format change is one fix in one function rather
479/// than a rewrite of every row.
480#[must_use]
481pub fn emit_rust(corpus_commit: &str, pci_ids_commit: &str, report: &BuildReport) -> String {
482    let mut out = String::new();
483    out.push_str("// GENERATED by `cargo run -p gpu-catalogue-gen`. DO NOT EDIT.\n");
484    out.push_str("//\n");
485    out.push_str("// Identity only. Every capability value still comes from the measured\n");
486    out.push_str("// tier tables in `tiers.rs`; nothing here can fabricate one.\n");
487    out.push_str("//\n");
488    out.push_str(&format!(
489        "// Model names, device ids and population weights: fingerprint corpus\n\
490         // pinned at {corpus_commit}.\n\
491         // Device ids the corpus never reports: pciutils/pciids pinned at\n\
492         // {pci_ids_commit}.\n"
493    ));
494    out.push_str("//\n");
495    out.push_str(&format!("// {} entries.\n", report.rows.len()));
496    if !report.unmatched.is_empty() {
497        out.push_str(&format!(
498            "// {} models were dropped for having no device id in either source,\n\
499             // which a D3D11 renderer string cannot be composed without:\n",
500            report.unmatched.len()
501        ));
502        for model in &report.unmatched {
503            out.push_str(&format!("//   {model}\n"));
504        }
505    }
506    out.push_str("\nuse super::types::{CatalogueEntry, Tier};\n\n");
507    // Selection lands in the next task and is this table's only consumer, so
508    // until then the crate has no non-test reader for it.
509    out.push_str("pub(crate) const CATALOGUE: &[CatalogueEntry] = &[\n");
510    for row in &report.rows {
511        let device_id = row
512            .device_id
513            .map_or_else(|| "None".to_string(), |id| format!("Some(0x{id:04X})"));
514        out.push_str(&format!(
515            "    CatalogueEntry {{ model: {:?}, vendor: {:?}, device_id: {device_id}, \
516             tier: Tier::{}, weight: {:.9} }},\n",
517            row.model, row.vendor, row.tier, row.weight,
518        ));
519    }
520    out.push_str("];\n");
521    out
522}

552/// Pull every in-scope `(backend, vendor, model)` out of the corpus's
553/// `videoCard` node.
554///
555/// Only the model name is taken. The renderer string is rebuilt by
556/// [`compose_renderer`], because the corpus predates the device ID current
557/// ANGLE appends and a copied string would carry whichever Chrome collected it.
558///
559/// Results are sorted and deduplicated, so the emitted catalogue does not
560/// change when the corpus reorders its buckets.
561#[must_use]
562pub fn extract_models(network_json: &str) -> Vec<CorpusModel> {
563    let Ok(root) = serde_json::from_str::<serde_json::Value>(network_json) else {
564        return Vec::new();
565    };
566    let nodes = root["nodes"].as_array().map(Vec::as_slice).unwrap_or(&[]);
567
568    // `userAgent` is parentless, so its conditional table is a plain prior.
569    let mut prior: std::collections::BTreeMap<&str, f64> = std::collections::BTreeMap::new();
570    for node in nodes {
571        if node["name"] != "userAgent" {
572            continue;
573        }
574        for (ua, p) in node["conditionalProbabilities"]
575            .as_object()
576            .into_iter()
577            .flatten()
578        {
579            if let Some(p) = p.as_f64() {
580                prior.insert(ua.as_str(), p);
581            }
582        }
583    }
584
585    let mut totals: std::collections::BTreeMap<(Backend, String, String, Option<u32>), f64> =
586        std::collections::BTreeMap::new();
587
588    for node in nodes {
589        if node["name"] != "videoCard" {
590            continue;
591        }
592        let deeper = &node["conditionalProbabilities"]["deeper"];
593        for (ua, bucket) in deeper.as_object().into_iter().flatten() {
594            // A user agent the prior does not mention contributes nothing: its
595            // marginal weight is zero, so the device is still catalogued but
596            // never drawn by share.
597            let p_ua = prior.get(ua.as_str()).copied().unwrap_or(0.0);
598            for (key, p) in bucket.as_object().into_iter().flatten() {
599                let json = key.strip_prefix("*STRINGIFIED*").unwrap_or(key);
600                let Ok(card) = serde_json::from_str::<serde_json::Value>(json) else {
601                    continue;
602                };
603                let Some(renderer) = card["renderer"].as_str() else {
604                    continue;
605                };
606                let Some(m) = split_renderer(renderer) else {
607                    continue;
608                };
609                *totals
610                    .entry((m.backend, m.vendor, m.model, m.device_id))
611                    .or_default() += p_ua * p.as_f64().unwrap_or(0.0);
612            }
613        }
614    }
615
616    let mut out: Vec<CorpusModel> = totals
617        .into_iter()
618        .map(
619            |((backend, vendor, model, device_id), weight)| CorpusModel {
620                backend,
621                vendor,
622                model,
623                device_id,
624                weight,
625            },
626        )
627        .collect();
628    out.sort_by(|a, b| model_key(a).cmp(&model_key(b)));
629    out
630}

632/// Take an ANGLE renderer string apart, or answer `None` when it describes
633/// something v1 does not model.
634fn split_renderer(renderer: &str) -> Option<CorpusModel> {
635    let inner = renderer.strip_prefix("ANGLE (")?.strip_suffix(')')?;
636    let (vendor, rest) = inner.split_once(", ")?;
637
638    // A vendor ANGLE could not name, so it printed the raw PCI id — a VM
639    // display adapter in every corpus instance. There is no real device here.
640    if vendor.starts_with("0x") {
641        return None;
642    }
643
644    if let Some(model) = rest.strip_prefix("ANGLE Metal Renderer: ") {
645        // Everything before the trailing ", Unspecified Version" field.
646        let model = model.split_once(", ").map_or(model, |(before, _)| before);
647        if NON_MAC_APPLE_PREFIXES.iter().any(|p| model.starts_with(p)) {
648            return None;
649        }
650        // Intel and AMD Macs are deliberately out of scope for v1: Dawn
651        // resolves their WebGPU architecture through a `gpu_info` lookup
652        // rather than the `mDeviceId == 0` path every Apple silicon part
653        // takes, and nothing has probed one. Dropping them costs identities;
654        // guessing an architecture token costs coherence.
655        if vendor != "Apple" {
656            return None;
657        }
658        return Some(CorpusModel {
659            backend: Backend::Metal,
660            vendor: vendor.to_string(),
661            model: model.to_string(),
662            device_id: None,
663            weight: 0.0,
664        });
665    }
666
667    if let Some(body) = rest.strip_suffix(", D3D11") {
668        if D3D11_VENDOR_EXCLUSIONS.iter().any(|(v, _)| *v == vendor) {
669            return None;
670        }
671        // Both shipped D3D11 tiers are feature level 11, and ANGLE writes the
672        // feature level into the string as its shader model: FL11 reports
673        // `vs_5_0 ps_5_0`, FL10 reports `vs_4_1` or `vs_4_0`. The corpus really
674        // does contain the latter -- GeForce 210, GT 220, 8500 GT, Quadro NVS
675        // 295, older Intel HD -- and putting one on an FL11 tier would serve
676        // D3D11_REQ_* constants that card cannot reach, beneath a string
677        // claiming a shader model it does not have. Same contradiction the
678        // tiers exist to remove, so these are dropped.
679        //
680        // It is also what licenses `compose_renderer` to hardcode vs_5_0.
681        if !body.contains("vs_5_0 ps_5_0") {
682            return None;
683        }
684        let named = body.split(" Direct3D11").next()?.trim();
685        // Keep the device id when the string carries one. Chrome has appended
686        // it for some time, so most entries have it, and an observed pairing
687        // beats anything a name lookup can reconstruct: one marketing name
688        // spans many SKUs, and the corpus knows which ones actually occur.
689        let (model, device_id) = match named.rsplit_once(" (0x") {
690            Some((before, hex)) => {
691                let hex = hex.strip_suffix(')').unwrap_or(hex);
692                (before.trim(), u32::from_str_radix(hex, 16).ok())
693            }
694            None => (named, None),
695        };
696        if model.is_empty() {
697            return None;
698        }
699        return Some(CorpusModel {
700            backend: Backend::D3d11,
701            vendor: vendor.to_string(),
702            model: model.to_string(),
703            device_id,
704            weight: 0.0,
705        });
706    }
707
708    // Every other backend: desktop GL, GLES, Vulkan. Real, but device-derived
709    // or unmodelled, so out of scope for a catalogue built on shared tiers.
710    None
711}
