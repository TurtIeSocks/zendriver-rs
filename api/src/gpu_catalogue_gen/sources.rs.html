1//! Where the catalogue's two inputs come from, and at which commit.
2//!
3//! Both are fetched rather than vendored, because both are reachable at an
4//! immutable commit SHA. That keeps regeneration reproducible without copying
5//! 1.5 MB of `pci.ids` into the tree or redistributing a dual BSD-3/GPLv2 file.
6//!
7//! Pinning against `pci-ids.ucw.cz` directly is not an option: it serves a
8//! moving file with no content-addressed URL, so the catalogue could change
9//! with no input in this repo changing. `pciutils/pciids` is a read-only git
10//! mirror of that same database, regenerated automatically, which is what makes
11//! a pin possible at all.
12
13/// Read-only git mirror of the PCI ID Database.
14pub const PCI_IDS_COMMIT: &str = "7a900ad2ad4b91c3a9d4f4da60e247864e6a73aa";
15
16/// The fingerprint corpus whose `videoCard` node carries driver-reported
17/// renderer strings.
18///
19/// The runtime cache in `zendriver-fingerprints` tracks `master`, which is
20/// right for a cache and wrong for a generator: the committed catalogue must
21/// not change unless a constant in this file changes.
22pub const CORPUS_COMMIT: &str = "358002c33e6e0b1cec4eb167be29ba9b3f854216";
23
24/// Raw URL for the pinned `pci.ids`.
25#[must_use]
26pub fn pci_ids_url() -> String {
27    format!("https://raw.githubusercontent.com/pciutils/pciids/{PCI_IDS_COMMIT}/pci.ids")
28}
29
30/// Raw URL for the pinned fingerprint network definition.
31#[must_use]
32pub fn corpus_url() -> String {
33    format!(
34        "https://raw.githubusercontent.com/apify/fingerprint-suite/{CORPUS_COMMIT}\
35         /packages/fingerprint-generator/src/data_files/fingerprint-network-definition.zip"
36    )
37}

39#[cfg(test)]
40mod tests {
41    use super::*;
42
43    #[test]
44    fn both_sources_are_pinned_to_a_full_commit_sha() {
45        // A branch name here would let the committed catalogue change without
46        // any input in this repo changing, which is the failure mode hardest
47        // to notice: nothing in a diff would show it.
48        for (label, pin) in [("pci.ids", PCI_IDS_COMMIT), ("corpus", CORPUS_COMMIT)] {
49            assert_eq!(pin.len(), 40, "{label} pin must be a full SHA, got {pin:?}");
50            assert!(
51                pin.chars().all(|c| c.is_ascii_hexdigit()),
52                "{label} pin must be hex, got {pin:?}"
53            );
54        }
55    }
56
57    #[test]
58    fn both_urls_embed_their_pin() {
59        assert!(pci_ids_url().contains(PCI_IDS_COMMIT));
60        assert!(corpus_url().contains(CORPUS_COMMIT));
61        // Line continuations in the format string must not leave whitespace in
62        // the path.
63        assert!(
64            !corpus_url().contains(' '),
65            "corpus URL has a space in it: {}",
66            corpus_url()
67        );
68    }
