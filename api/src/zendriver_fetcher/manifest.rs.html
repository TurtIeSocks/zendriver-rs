1//! Chrome for Testing manifest fetcher.
2//!
3//! Fetches and parses the `known-good-versions-with-downloads.json` manifest
4//! published by Google's Chrome for Testing project, plus the per-channel
5//! `last-known-good-versions-with-downloads.json` manifest used to resolve
6//! the `Beta`/`Dev`/`Canary` channels.
7
8use std::collections::HashMap;
9
10use serde::Deserialize;
11
12use crate::error::FetcherError;
13
14#[derive(Debug, Deserialize)]
15pub(crate) struct KnownGoodVersionsResponse {
16    pub versions: Vec<VersionEntry>,
17}
18
19/// One version in the flat manifest.
20///
21/// Only the fields resolution reads are modelled — the manifest also carries a
22/// Chromium `revision` per entry, which nothing here keys on, and serde drops
23/// what is not declared.
24#[derive(Debug, Deserialize)]
25pub(crate) struct VersionEntry {
26    pub version: String,
27    pub downloads: Downloads,
28}
29
30#[derive(Debug, Deserialize)]
31pub(crate) struct Downloads {
32    pub chrome: Vec<Download>,
33}
34
35#[derive(Debug, Deserialize)]
36pub(crate) struct Download {
37    pub platform: String,
38    pub url: String,
39}
40
41/// Fetch and parse Chrome for Testing's flat version manifest from `url`.
42///
43/// The URL is a parameter rather than a constant so tests can point it at a
44/// wiremock server; callers pass [`DEFAULT_CFT_URL`](crate::fetcher::DEFAULT_CFT_URL).
45pub(crate) async fn fetch_manifest_from(
46    url: &str,
47) -> Result<KnownGoodVersionsResponse, FetcherError> {
48    Ok(serde_json::from_str(&fetch_text(url).await?)?)
49}
50
51/// Chrome for Testing's per-channel manifest —
52/// `last-known-good-versions-with-downloads.json`. Keyed by channel name
53/// (`"Stable"`, `"Beta"`, `"Dev"`, `"Canary"`) rather than a flat version
54/// list, since each channel tracks its own latest known-good build.
55#[derive(Debug, Deserialize)]
56pub(crate) struct ChannelsResponse {
57    pub channels: HashMap<String, ChannelEntry>,
58}
59
60/// One channel's entry in [`ChannelsResponse`] — same shape as
61/// [`VersionEntry`] (the manifest omits only the redundant `channel` name
62/// field, already carried as this entry's map key).
63#[derive(Debug, Deserialize)]
64pub(crate) struct ChannelEntry {
65    pub version: String,
66    pub downloads: Downloads,
67}
68
69/// Same rationale as [`fetch_manifest_from`] — the URL is a parameter so tests
70/// can point it at a wiremock server.
71pub(crate) async fn fetch_channels_manifest_from(
72    url: &str,
73) -> Result<ChannelsResponse, FetcherError> {
74    Ok(serde_json::from_str(&fetch_text(url).await?)?)
75}
76
77/// GET `url` and return the body, failing on a non-success status.
78///
79/// The `error_for_status` is the point: without it a 404 or a CDN 503 is an
80/// HTML error page handed to `serde_json`, and the operator gets
81/// "expected value at line 1 column 1" instead of the status code.
82async fn fetch_text(url: &str) -> Result<String, FetcherError> {
83    Ok(crate::tls::get(url)
84        .await?
85        .error_for_status()?
86        .text()
87        .await?)
88}

90/// One entry from GitHub's `GET /repos/{owner}/{repo}/releases` response.
91///
92/// ungoogled-chromium publishes no manifest — the release list *is* the
93/// index, one per OS. Only the fields resolution needs are modelled; GitHub's
94/// payload is large and the rest is ignored.
95#[derive(Debug, Deserialize)]
96pub(crate) struct GitHubRelease {
97    pub tag_name: String,
98    #[serde(default)]
99    pub draft: bool,
100    #[serde(default)]
101    pub prerelease: bool,
102    #[serde(default)]
103    pub assets: Vec<GitHubAsset>,
104}
105
106/// A file attached to a [`GitHubRelease`].
107#[derive(Debug, Deserialize)]
108pub(crate) struct GitHubAsset {
109    pub name: String,
110    pub browser_download_url: String,
111}
112
113/// Fetch a repo's release list, newest first (GitHub's own ordering).
114///
115/// One request with `per_page=100` rather than pagination: each page costs
116/// one of the 60 unauthenticated requests per hour, and a single page covers
117/// well over a year of ungoogled releases.
118pub(crate) async fn fetch_github_releases(
119    api_base: &str,
120    repo: &str,
121) -> Result<Vec<GitHubRelease>, FetcherError> {
122    let url = format!(
123        "{}/repos/{repo}/releases?per_page=100",
124        api_base.trim_end_matches('/')
125    );
126    let text = crate::tls::get_github(&url).await?;
127    Ok(serde_json::from_str(&text)?)
128}

130/// Fetch the newest revision published for `platform_dir` in Chromium's
131/// snapshot bucket.
132///
133/// `LAST_CHANGE` is a plain-text integer, not JSON. The snapshot bucket has
134/// no manifest of any kind — which is precisely why snapshots cannot be
135/// resolved by version.
136pub(crate) async fn fetch_last_change(
137    snapshot_base: &str,
138    platform_dir: &str,
139) -> Result<u64, FetcherError> {
140    let url = format!(
141        "{}/{platform_dir}/LAST_CHANGE",
142        snapshot_base.trim_end_matches('/')
143    );
144    let text = fetch_text(&url).await?;
145    text.trim().parse::<u64>().map_err(|_| {
146        FetcherError::VersionNotFound(format!(
147            "{url} returned {:?}, which is not a revision number",
148            text.chars().take(64).collect::<String>()
149        ))
150    })
151}

153#[cfg(test)]
154#[allow(clippy::panic, clippy::unwrap_used)]
155mod tests {
156    use super::*;
157    use wiremock::matchers::{method, path};
158    use wiremock::{Mock, MockServer, ResponseTemplate};
159
160    const FIXTURE_JSON: &str = r#"{"versions":[{"version":"120.0.6099.234","revision":"1234","downloads":{"chrome":[{"platform":"linux64","url":"https://example.com/chrome-linux64.zip"},{"platform":"mac-x64","url":"https://example.com/chrome-mac-x64.zip"}]}}]}"#;
161
162    #[tokio::test]
163    async fn parses_known_good_versions_from_stub_server() {
164        let server = MockServer::start().await;
165
166        Mock::given(method("GET"))
167            .and(path("/known-good-versions.json"))
168            .respond_with(ResponseTemplate::new(200).set_body_string(FIXTURE_JSON))
169            .mount(&server)
170            .await;
171
172        let url = format!("{}/known-good-versions.json", server.uri());
173        let manifest = fetch_manifest_from(&url).await.unwrap();
174
175        assert_eq!(manifest.versions.len(), 1);
176        let v = &manifest.versions[0];
177        assert_eq!(v.version, "120.0.6099.234");
178        // The fixture carries `revision` too; undeclared fields are dropped
179        // rather than failing the parse.
180        assert_eq!(v.downloads.chrome.len(), 2);
181        assert_eq!(v.downloads.chrome[0].platform, "linux64");
182        assert_eq!(
183            v.downloads.chrome[0].url,
184            "https://example.com/chrome-linux64.zip"
185        );
186        assert_eq!(v.downloads.chrome[1].platform, "mac-x64");
187        assert_eq!(
188            v.downloads.chrome[1].url,
189            "https://example.com/chrome-mac-x64.zip"
190        );
191    }
192
193    /// A CDN that answers 503 with an HTML error page must surface as the
194    /// status code. Feeding that page to `serde_json` instead reports
195    /// "expected value at line 1 column 1", which sends the operator looking
196    /// for a manifest-format change that never happened.
197    #[tokio::test]
198    async fn an_http_error_is_reported_as_http_not_as_a_parse_failure() {
199        let server = MockServer::start().await;
200        Mock::given(method("GET"))
201            .and(path("/manifest.json"))
202            .respond_with(
203                ResponseTemplate::new(503).set_body_string("<html>Service Unavailable</html>"),
204            )
205            .mount(&server)
206            .await;
207
208        let err = fetch_manifest_from(&format!("{}/manifest.json", server.uri()))
209            .await
210            .unwrap_err();
211        assert!(matches!(err, FetcherError::Http(_)), "{err:?}");
212        assert!(err.to_string().contains("503"), "{err}");
213    }
