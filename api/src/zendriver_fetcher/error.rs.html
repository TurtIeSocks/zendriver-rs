1//! Fetcher-layer errors.
2
3/// Errors surfaced by [`crate::Fetcher`] during manifest fetch, download,
4/// extract, or verification.
5#[derive(Debug, thiserror::Error)]
6#[non_exhaustive]
7pub enum FetcherError {
8    /// HTTP request failed (network down, 4xx/5xx response, etc.).
9    #[error("http: {0}")]
10    Http(#[from] reqwest::Error),
11
12    /// Local filesystem I/O failed (cache write, file permission, ...).
13    #[error("io: {0}")]
14    Io(#[from] std::io::Error),
15
16    /// Manifest JSON could not be parsed.
17    #[error("manifest: {0}")]
18    Manifest(#[from] serde_json::Error),
19
20    /// The requested [`crate::VersionSpec::Explicit`] string was not present
21    /// in the manifest. Carries the requested version string.
22    #[error("version not found: {0}")]
23    VersionNotFound(String),
24
25    /// The current platform is not covered by Chrome for Testing, or the
26    /// requested non-stable [`crate::Channel`] is not yet wired.
27    #[error("unsupported platform")]
28    UnsupportedPlatform,
29
30    /// SHA256 checksum mismatch on the downloaded archive.
31    #[error("integrity check failed: expected {expected}, got {actual}")]
32    IntegrityFailed {
33        /// SHA256 from the manifest.
34        expected: String,
35        /// SHA256 computed from the downloaded bytes.
36        actual: String,
37    },
38
39    /// The downloaded zip could not be extracted.
40    #[error("extraction: {0}")]
41    Extraction(String),
42
43    /// The requested [`crate::VersionSpec`] cannot be honoured by the
44    /// requested [`crate::Distribution`].
45    ///
46    /// Carries the reason *and* the alternative, because every case is "you
47    /// asked for something this index cannot express": an explicit version
48    /// against the revision-keyed snapshot bucket, or a release channel
49    /// against a distribution that has none. Resolving those to "the newest
50    /// build" instead would hand back a browser other than the one requested,
51    /// which is worse than failing.
52    #[error("{distribution} cannot resolve {selector}: {reason}")]
53    UnsupportedSelector {
54        /// Distribution that refused the selector.
55        distribution: &'static str,
56        /// The selector, as a short human-readable phrase.
57        selector: String,
58        /// Why it cannot be honoured, and what to use instead.
59        reason: String,
60    },
61
62    /// The release exists but publishes no asset for the target platform.
63    ///
64    /// Routine for ungoogled-chromium, whose three per-OS repos release on
65    /// independent cadences — a version present on Windows may simply not be
66    /// built for macOS yet.
67    #[error("{repo} release {tag} has no asset for {platform} (expected one ending in {suffix})")]
68    AssetNotFound {
69        /// `owner/name` of the repo searched.
70        repo: String,
71        /// Release tag searched.
72        tag: String,
73        /// Platform whose asset is missing.
74        platform: &'static str,
75        /// Asset filename suffix that was looked for.
76        suffix: &'static str,
77    },
78
79    /// GitHub's API refused the request for rate-limiting reasons.
80    ///
81    /// Unauthenticated `api.github.com` allows 60 requests per hour per IP;
82    /// setting `GITHUB_TOKEN` raises it to 5000. Surfaced as its own variant
83    /// so the fix is in the message instead of buried in an HTTP 403.
84    #[error(
85        "github api rate limit reached (unauthenticated limit is 60 requests/hour{reset_hint}); \
86         set GITHUB_TOKEN to raise it to 5000/hour"
87    )]
88    GitHubRateLimited {
89        /// `", resets at <unix timestamp>"` when GitHub sent an
90        /// `x-ratelimit-reset` header, otherwise empty.
91        reset_hint: String,
92    },
93
94    /// The downloaded archive is in a format this host cannot unpack.
95    ///
96    /// In practice this is ungoogled-chromium's macOS `.dmg`, which is
97    /// unpacked with `hdiutil` and therefore only on a macOS host.
98    #[error("unsupported archive: {0}")]
99    UnsupportedArchive(String),
100}

102#[cfg(test)]
103#[allow(clippy::panic, clippy::unwrap_used)]
104mod tests {
105    use super::*;
106
107    #[test]
108    fn display_version_not_found() {
109        let e = FetcherError::VersionNotFound("123.4.5.6".into());
110        assert_eq!(e.to_string(), "version not found: 123.4.5.6");
111    }
112
113    #[test]
114    fn display_integrity_failed() {
115        let e = FetcherError::IntegrityFailed {
116            expected: "abc".into(),
117            actual: "def".into(),
118        };
119        assert_eq!(
120            e.to_string(),
121            "integrity check failed: expected abc, got def"
122        );
123    }
124
125    #[test]
126    fn display_unsupported_platform() {
127        let e = FetcherError::UnsupportedPlatform;
128        assert_eq!(e.to_string(), "unsupported platform");
129    }
130
131    /// The whole point of this variant is that the message says what to do
132    /// instead, so the message is what gets pinned.
133    #[test]
134    fn display_unsupported_selector_names_the_alternative() {
135        let e = FetcherError::UnsupportedSelector {
136            distribution: "Chromium snapshot",
137            selector: "version 151.0.7922.76".into(),
138            reason: "snapshots are keyed by revision; use --revision".into(),
139        };
140        assert_eq!(
141            e.to_string(),
142            "Chromium snapshot cannot resolve version 151.0.7922.76: \
143             snapshots are keyed by revision; use --revision"
144        );
145    }
146
147    #[test]
148    fn display_rate_limited_names_the_limit_and_the_fix() {
149        let e = FetcherError::GitHubRateLimited {
150            reset_hint: ", resets at 1786023295".into(),
151        };
152        let msg = e.to_string();
153        assert!(msg.contains("60 requests/hour"), "{msg}");
154        assert!(msg.contains("GITHUB_TOKEN"), "{msg}");
155        assert!(msg.contains("resets at 1786023295"), "{msg}");
156    }
157
158    #[test]
159    fn display_asset_not_found_names_repo_tag_and_suffix() {
160        let e = FetcherError::AssetNotFound {
161            repo: "ungoogled-software/ungoogled-chromium-macos".into(),
162            tag: "151.0.7922.71-1.1".into(),
163            platform: "mac-arm64",
164            suffix: "_arm64-macos.dmg",
165        };
166        assert_eq!(
167            e.to_string(),
168            "ungoogled-software/ungoogled-chromium-macos release 151.0.7922.71-1.1 \
169             has no asset for mac-arm64 (expected one ending in _arm64-macos.dmg)"
170        );
171    }
