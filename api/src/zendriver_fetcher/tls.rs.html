1//! Process-wide rustls crypto provider, and the one HTTP entry point that guarantees it is set.
2//!
3//! reqwest 0.12 chose a provider for you as a side effect of the feature name: `rustls-tls`
4//! expanded to `__rustls-ring`. reqwest 0.13 stopped guessing. Its `rustls` feature bundles
5//! aws-lc-rs, whose `-sys` C/asm crate breaks musl cross-compilation, so this workspace uses
6//! `rustls-no-provider` and names the provider itself.
7//!
8//! The failure mode that shape creates is a **runtime panic** — rustls aborts with "no
9//! process-level CryptoProvider available" the first time it builds a TLS config — not a compile
10//! error. A missed call site therefore ships. That is why every request in this crate goes through
11//! [`get`] rather than calling `reqwest::get` directly: there is one door, and it installs the
12//! provider before opening.
13
14use std::sync::Once;
15
16/// Install the ring crypto provider as the process default, once.
17///
18/// Safe to call from anywhere, any number of times, and from more than one crate: the first caller
19/// wins and later ones are a no-op. `install_default` returns `Err` when a provider is already
20/// installed, which is exactly the "someone else got here first" case and is not an error for us.
21pub fn install_default_crypto_provider() {
22    static ONCE: Once = Once::new();
23    ONCE.call_once(|| {
24        let _ = rustls::crypto::ring::default_provider().install_default();
25    });
26}
27
28/// `reqwest::get`, with the crypto provider guaranteed installed first.
29pub(crate) async fn get(url: &str) -> Result<reqwest::Response, reqwest::Error> {
30    install_default_crypto_provider();
31    reqwest::get(url).await
32}

34#[cfg(test)]
35#[allow(clippy::unwrap_used)]
36mod tests {
37    use super::*;
38
39    /// The guard is idempotent — a second call must not panic or poison the `Once`.
40    #[test]
41    fn install_is_idempotent() {
42        install_default_crypto_provider();
43        install_default_crypto_provider();
44        install_default_crypto_provider();
45    }
46
47    /// A TLS config must be constructible after installing. This is the assertion that would have
48    /// caught a missing provider: without one, `builder()` panics rather than failing to compile.
49    #[test]
50    fn tls_config_builds_after_install() {
51        install_default_crypto_provider();
52        let _ = rustls::ClientConfig::builder()
53            .with_root_certificates(rustls::RootCertStore::empty())
54            .with_no_client_auth();
55    }
56
57    /// End-to-end proof over real HTTPS, through the same door the crate's own requests use.
58    ///
59    /// Ignored by default because it needs the network; the nightly ignored-tests job runs it.
60    /// The offline tests above prove the provider installs, but only a real handshake proves the
61    /// request path actually goes through it — a wiremock server speaks plain HTTP and would pass
62    /// even with no provider installed at all.
63    #[tokio::test]
64    #[ignore = "network"]
65    async fn https_request_succeeds_through_the_guarded_door() {
66        let resp =
67            get("https://googlechromelabs.github.io/chrome-for-testing/known-good-versions.json")
68                .await
69                .expect("HTTPS must succeed with the ring provider installed");
70        assert!(resp.status().is_success(), "status: {}", resp.status());
71    }
72}
