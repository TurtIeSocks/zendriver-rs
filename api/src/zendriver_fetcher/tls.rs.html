1//! Process-wide rustls crypto provider, and the one HTTP entry point that guarantees it is set.
2//!
3//! reqwest 0.12 chose a provider for you as a side effect of the feature name: `rustls-tls`
4//! expanded to `__rustls-ring`. reqwest 0.13 stopped guessing. Its `rustls` feature bundles
5//! aws-lc-rs, whose `-sys` C/asm crate breaks musl cross-compilation, so this workspace uses
6//! `rustls-no-provider` and names the provider itself.
7//!
8//! The failure mode that shape creates is a **runtime panic** — rustls aborts with "no
9//! process-level CryptoProvider available" the first time it builds a TLS config — not a compile
10//! error. A missed call site therefore ships. That is why every request in this crate goes through
11//! [`get`] rather than calling `reqwest::get` directly: there is one door, and it installs the
12//! provider before opening.
13
14use std::sync::Once;
15
16/// Install the ring crypto provider as the process default, once.
17///
18/// Safe to call from anywhere, any number of times, and from more than one crate: the first caller
19/// wins and later ones are a no-op. `install_default` returns `Err` when a provider is already
20/// installed, which is exactly the "someone else got here first" case and is not an error for us.
21pub fn install_default_crypto_provider() {
22    static ONCE: Once = Once::new();
23    ONCE.call_once(|| {
24        let _ = rustls::crypto::ring::default_provider().install_default();
25    });
26}
27
28/// `reqwest::get`, with the crypto provider guaranteed installed first.
29pub(crate) async fn get(url: &str) -> Result<reqwest::Response, reqwest::Error> {
30    install_default_crypto_provider();
31    reqwest::get(url).await
32}

34/// User-Agent sent to `api.github.com`.
35///
36/// Not cosmetic: GitHub's API answers **403** to any request without a
37/// User-Agent, which reads exactly like a rate-limit rejection and sends you
38/// hunting for a token you do not need.
39const GITHUB_USER_AGENT: &str = concat!("zendriver-fetcher/", env!("CARGO_PKG_VERSION"));
40
41/// GET a JSON document from GitHub's REST API.
42///
43/// Sends the required User-Agent, pins the `2022-11-28` API version, and
44/// attaches `GITHUB_TOKEN` as a bearer token when the environment has one.
45/// A rejection that looks like rate limiting becomes
46/// [`FetcherError::GitHubRateLimited`](crate::FetcherError::GitHubRateLimited),
47/// whose message names the 60/hour unauthenticated ceiling and the variable
48/// that lifts it — a bare `403` would leave the caller guessing.
49pub(crate) async fn get_github(url: &str) -> Result<String, crate::error::FetcherError> {
50    install_default_crypto_provider();
51
52    let mut req = reqwest::Client::builder()
53        .user_agent(GITHUB_USER_AGENT)
54        .build()?
55        .get(url)
56        .header("Accept", "application/vnd.github+json")
57        .header("X-GitHub-Api-Version", "2022-11-28");
58
59    if is_github_api(url)
60        && let Ok(token) = std::env::var("GITHUB_TOKEN")
61        && !token.trim().is_empty()
62    {
63        req = req.bearer_auth(token.trim());
64    }
65
66    let resp = req.send().await?;
67
68    if is_rate_limited(resp.status(), resp.headers()) {
69        return Err(crate::error::FetcherError::GitHubRateLimited {
70            reset_hint: reset_hint(resp.headers()),
71        });
72    }
73
74    Ok(resp.error_for_status()?.text().await?)
75}

77/// Is `url` GitHub's own API host, over TLS?
78///
79/// The release-lookup base is overridable — that is how the tests point it at
80/// a wiremock server — and `GITHUB_TOKEN` is a credential. It goes to
81/// `https://api.github.com` or it does not go at all; an unauthenticated
82/// request to somewhere else is a smaller failure than a leaked token.
83///
84/// The scheme is half the check: `http://api.github.com` is the right host and
85/// would still put a bearer token on the wire in cleartext.
86fn is_github_api(url: &str) -> bool {
87    reqwest::Url::parse(url)
88        .is_ok_and(|u| u.scheme() == "https" && u.host_str() == Some("api.github.com"))
89}

91/// True when a GitHub response is a rate-limit rejection.
92///
93/// GitHub signals the primary limit with `403` + `x-ratelimit-remaining: 0`,
94/// and secondary/abuse limits with `429`. A `403` with budget left over is
95/// something else (a bad token, say) and is left to `error_for_status`.
96fn is_rate_limited(status: reqwest::StatusCode, headers: &reqwest::header::HeaderMap) -> bool {
97    if status == reqwest::StatusCode::TOO_MANY_REQUESTS {
98        return true;
99    }
100    if status != reqwest::StatusCode::FORBIDDEN {
101        return false;
102    }
103    headers
104        .get("x-ratelimit-remaining")
105        .and_then(|v| v.to_str().ok())
106        .and_then(|v| v.trim().parse::<u64>().ok())
107        .is_some_and(|remaining| remaining == 0)
108}
109
110/// `", resets at <unix timestamp>"` when GitHub told us when the window rolls
111/// over, otherwise an empty string.
112fn reset_hint(headers: &reqwest::header::HeaderMap) -> String {
113    headers
114        .get("x-ratelimit-reset")
115        .and_then(|v| v.to_str().ok())
116        .map(|ts| format!(", resets at {}", ts.trim()))
117        .unwrap_or_default()
118}

120#[cfg(test)]

125    /// The guard is idempotent — a second call must not panic or poison the `Once`.
126    #[test]
127    fn install_is_idempotent() {
128        install_default_crypto_provider();
129        install_default_crypto_provider();
130        install_default_crypto_provider();
131    }
132
133    /// A TLS config must be constructible after installing. This is the assertion that would have
134    /// caught a missing provider: without one, `builder()` panics rather than failing to compile.
135    #[test]
136    fn tls_config_builds_after_install() {
137        install_default_crypto_provider();
138        let _ = rustls::ClientConfig::builder()
139            .with_root_certificates(rustls::RootCertStore::empty())
140            .with_no_client_auth();
141    }
142
143    fn headers(pairs: &[(&str, &str)]) -> reqwest::header::HeaderMap {
144        let mut h = reqwest::header::HeaderMap::new();
145        for (k, v) in pairs {
146            h.insert(
147                reqwest::header::HeaderName::from_bytes(k.as_bytes()).unwrap(),
148                v.parse().unwrap(),
149            );
150        }
151        h
152    }
153
154    #[test]
155    fn forbidden_with_no_budget_left_is_rate_limiting() {
156        assert!(is_rate_limited(
157            reqwest::StatusCode::FORBIDDEN,
158            &headers(&[("x-ratelimit-remaining", "0")]),
159        ));
160    }
161
162    /// A 403 that still has budget is a different problem (bad token, blocked
163    /// resource) and must not be reported as rate limiting.
164    #[test]
165    fn forbidden_with_budget_left_is_not_rate_limiting() {
166        assert!(!is_rate_limited(
167            reqwest::StatusCode::FORBIDDEN,
168            &headers(&[("x-ratelimit-remaining", "41")]),
169        ));
170        assert!(!is_rate_limited(
171            reqwest::StatusCode::FORBIDDEN,
172            &headers(&[]),
173        ));
174    }
175
176    #[test]
177    fn too_many_requests_is_rate_limiting_regardless_of_headers() {
178        assert!(is_rate_limited(
179            reqwest::StatusCode::TOO_MANY_REQUESTS,
180            &headers(&[]),
181        ));
182    }
183
184    #[test]
185    fn success_is_never_rate_limiting() {
186        assert!(!is_rate_limited(
187            reqwest::StatusCode::OK,
188            &headers(&[("x-ratelimit-remaining", "0")]),
189        ));
190    }

192    /// The token is scoped to GitHub's host, not to "whatever base was
193    /// configured" — the base is overridable, and a bearer token sent to an
194    /// arbitrary host is a leaked credential.
195    #[test]
196    fn only_githubs_own_api_host_is_credentialed() {
197        assert!(is_github_api(
198            "https://api.github.com/repos/ungoogled-software/ungoogled-chromium-macos/releases"
199        ));
200
201        assert!(!is_github_api("http://127.0.0.1:8080/repos/x/y/releases"));
202        // Right host, no TLS: the token would go out in cleartext.
203        assert!(!is_github_api("http://api.github.com/repos/x/y/releases"));
204        // Suffix and userinfo tricks both resolve to a host that is not ours.
205        assert!(!is_github_api("https://api.github.com.evil.test/repos"));
206        assert!(!is_github_api("https://api.github.com@evil.test/repos"));
207        assert!(!is_github_api("https://github.com/repos/x/y/releases"));
208        assert!(!is_github_api("not a url"));
209    }
210
211    #[test]
212    fn reset_hint_is_empty_without_the_header() {
213        assert_eq!(reset_hint(&headers(&[])), "");
214        assert_eq!(
215            reset_hint(&headers(&[("x-ratelimit-reset", "1786023295")])),
216            ", resets at 1786023295"
217        );
218    }

220    /// End-to-end proof over real HTTPS, through the same door the crate's own requests use.
221    ///
222    /// Ignored by default because it needs the network; the nightly ignored-tests job runs it.
223    /// The offline tests above prove the provider installs, but only a real handshake proves the
224    /// request path actually goes through it — a wiremock server speaks plain HTTP and would pass
225    /// even with no provider installed at all.
226    #[tokio::test]
227    #[ignore = "network"]
228    async fn https_request_succeeds_through_the_guarded_door() {
229        let resp =
230            get("https://googlechromelabs.github.io/chrome-for-testing/known-good-versions.json")
231                .await
232                .expect("HTTPS must succeed with the ring provider installed");
233        assert!(resp.status().is_success(), "status: {}", resp.status());
234    }
