1//! Cache directory layout + path resolution.
2//!
3//! Chrome for Testing keeps the layout it has always had, straight under the
4//! cache root:
5//!
6//! ```text
7//! <cache_dir>/
8//!   <version>/
9//!     chrome-<platform_cft>/
10//!       chrome                                            (Linux)
11//!       chrome.exe                                        (Windows)
12//!       Google Chrome for Testing.app/Contents/MacOS/...  (macOS)
13//! ```
14//!
15//! The distributions added later namespace themselves under their
16//! [`Distribution::slug`]:
17//!
18//! ```text
19//! <cache_dir>/
20//!   ungoogled/151.0.7922.71/...
21//!   snapshot/r1674890/...
22//! ```
23//!
24//! The asymmetry is deliberate and load-bearing in both directions. Chrome
25//! for Testing stays un-prefixed so caches populated by earlier versions of
26//! this crate keep hitting. The others are prefixed because a Chrome version
27//! number is *not* unique across distributions — CfT 151.0.7922.71 and
28//! ungoogled 151.0.7922.71 are different binaries, and sharing a directory
29//! would serve whichever landed first.
30//!
31//! Per-build dirs are written atomically: download + unpack land under a
32//! `<build>.tmp/` sibling, then a single rename promotes it.
33
34use std::env;
35use std::path::{Path, PathBuf};

37use crate::distribution::Distribution;
38use crate::platform::Platform;

40/// Default cache root: OS cache dir if available, otherwise the temp dir.
41///
42/// Always suffixed with `zendriver/chrome` so multiple consumers don't
43/// collide at the top level.
44pub(crate) fn default_cache_dir() -> PathBuf {
45    dirs::cache_dir()
46        .unwrap_or_else(env::temp_dir)
47        .join("zendriver/chrome")
48}
49
50/// `path` with `suffix` appended to its last component: `<build>` + `.tmp`
51/// becomes the `<build>.tmp` staging sibling.
52///
53/// Appends to the `OsString` rather than going through
54/// `format!("{}", path.display())`, because `display()` substitutes U+FFFD for
55/// bytes that are not valid UTF-8. On a cache directory containing any such
56/// byte that spelling would stage into — and try to promote from — a directory
57/// other than the one intended.
58pub(crate) fn with_suffix(path: &Path, suffix: &str) -> PathBuf {
59    let mut joined = path.to_path_buf().into_os_string();
60    joined.push(suffix);
61    PathBuf::from(joined)
62}
63
64/// Directory holding one unpacked build.
65///
66/// See the module docs for why Chrome for Testing is the un-prefixed case.
67pub(crate) fn build_dir(cache_dir: &Path, distribution: Distribution, build_id: &str) -> PathBuf {
68    match distribution {
69        Distribution::ChromeForTesting => cache_dir.join(build_id),
70        other => cache_dir.join(other.slug()).join(build_id),
71    }
72}

74/// Path to the Chrome for Testing binary *relative* to its build directory.
75///
76/// Used against the published `<cache>/<version>/` layout and against the
77/// in-progress `<cache>/<version>.tmp/` staging dir, so the binary can be
78/// made executable *before* it is atomically renamed into place.
79///
80/// The leading directory comes from [`Platform::cft_top_dir`] — the same value
81/// the extractor pins the archive to, so the two cannot drift into disagreeing
82/// about where the binary landed.
83pub(crate) fn cft_binary_subpath(platform: Platform) -> PathBuf {
84    let top = PathBuf::from(platform.cft_top_dir());
85    match platform {
86        Platform::LinuxX64 => top.join("chrome"),
87        Platform::MacX64 | Platform::MacArm64 => top
88            .join("Google Chrome for Testing.app")
89            .join("Contents")
90            .join("MacOS")
91            .join("Google Chrome for Testing"),
92        Platform::Win32 | Platform::Win64 => top.join("chrome.exe"),
93    }
94}

96/// Path to the binary inside an unpacked Chromium snapshot.
97///
98/// Snapshot archives are plain Chromium: the macOS bundle is `Chromium.app`
99/// and the Linux/Windows binary keeps the `chrome` name, all under the
100/// archive's own per-OS top directory (`chrome-mac`, not `chrome-mac-arm64`).
101pub(crate) fn snapshot_binary_subpath(platform: Platform) -> PathBuf {
102    let top = PathBuf::from(platform.snapshot_archive_stem());
103    match platform {
104        Platform::LinuxX64 => top.join("chrome"),
105        Platform::MacX64 | Platform::MacArm64 => top
106            .join("Chromium.app")
107            .join("Contents")
108            .join("MacOS")
109            .join("Chromium"),
110        Platform::Win32 | Platform::Win64 => top.join("chrome.exe"),
111    }
112}

114/// Filename an ungoogled AppImage is stored under inside its build dir.
115///
116/// An AppImage is not unpacked, so this is both the stored name and the
117/// executable path.
118pub(crate) const UNGOOGLED_APPIMAGE_NAME: &str = "chrome.AppImage";
119
120/// Bundle directory inside ungoogled-chromium's macOS disk image.
121pub(crate) const UNGOOGLED_APP_BUNDLE: &str = "Chromium.app";
122
123/// Path to the binary inside an unpacked ungoogled-chromium build.
124///
125/// `archive_top` is the zip's top-level directory on Windows — ungoogled
126/// names it after the asset (`ungoogled-chromium_151.0.7922.71-1.1_windows_x64/`),
127/// so unlike every other case here it is not a constant and has to be
128/// threaded through from the resolved asset name.
129pub(crate) fn ungoogled_binary_subpath(platform: Platform, archive_top: &str) -> PathBuf {
130    match platform {
131        Platform::LinuxX64 => PathBuf::from(UNGOOGLED_APPIMAGE_NAME),
132        Platform::MacX64 | Platform::MacArm64 => PathBuf::from(UNGOOGLED_APP_BUNDLE)
133            .join("Contents")
134            .join("MacOS")
135            .join("Chromium"),
136        Platform::Win32 | Platform::Win64 => PathBuf::from(archive_top).join("chrome.exe"),
137    }
138}

145    #[test]
146    fn default_cache_dir_ends_with_zendriver_chrome() {
147        let p = default_cache_dir();
148        assert!(
149            p.ends_with("zendriver/chrome"),
150            "expected suffix zendriver/chrome, got {}",
151            p.display()
152        );
153    }
154
155    /// The staging sibling has to be exactly the build directory plus a
156    /// suffix — a lossy round-trip through `display()` would promote a
157    /// different directory than the one that was written.
158    #[test]
159    fn with_suffix_appends_to_the_last_component() {
160        assert_eq!(
161            with_suffix(Path::new("/tmp/cache/120.0.6099.234"), ".tmp"),
162            Path::new("/tmp/cache/120.0.6099.234.tmp")
163        );
164        assert_eq!(
165            with_suffix(Path::new("/tmp/cache/ungoogled/151.0.7922.71"), ".tmp.zip"),
166            Path::new("/tmp/cache/ungoogled/151.0.7922.71.tmp.zip")
167        );
168    }
169
170    /// Non-UTF-8 bytes in the cache path must survive verbatim.
171    #[cfg(unix)]
172    #[test]
173    fn with_suffix_preserves_non_utf8_path_bytes() {
174        use std::ffi::OsStr;
175        use std::os::unix::ffi::OsStrExt as _;
176
177        let base = Path::new(OsStr::from_bytes(b"/tmp/ca\xffche/120.0.6099.234"));
178        assert_eq!(
179            with_suffix(base, ".tmp").as_os_str().as_bytes(),
180            b"/tmp/ca\xffche/120.0.6099.234.tmp"
181        );
182    }

184    #[test]
185    fn cft_binary_path_linux_layout() {
186        let root = Path::new("/tmp/cache");
187        let p = build_dir(root, Distribution::ChromeForTesting, "120.0.6099.234")
188            .join(cft_binary_subpath(Platform::LinuxX64));
189        assert_eq!(
190            p,
191            Path::new("/tmp/cache/120.0.6099.234/chrome-linux64/chrome")
192        );

219    /// Chrome for Testing must keep the un-prefixed layout, or every cache
220    /// populated before the distribution axis existed silently misses.
221    #[test]
222    fn chrome_for_testing_build_dir_is_unprefixed() {
223        let root = Path::new("/tmp/cache");
224        assert_eq!(
225            build_dir(root, Distribution::ChromeForTesting, "120.0.6099.234"),
226            Path::new("/tmp/cache/120.0.6099.234")

230    /// ...and the others must not be, or two distributions sharing a Chrome
231    /// version would share a directory.
232    #[test]
233    fn other_distributions_namespace_by_slug() {
234        let root = Path::new("/tmp/cache");
235        assert_eq!(
236            build_dir(root, Distribution::UngoogledChromium, "151.0.7922.71"),
237            Path::new("/tmp/cache/ungoogled/151.0.7922.71")
238        );
239        assert_eq!(
240            build_dir(root, Distribution::ChromiumSnapshot, "r1674890"),
241            Path::new("/tmp/cache/snapshot/r1674890")
242        );
243        assert_ne!(
244            build_dir(root, Distribution::ChromeForTesting, "151.0.7922.71"),
245            build_dir(root, Distribution::UngoogledChromium, "151.0.7922.71"),
246        );
247    }
248
249    #[test]
250    fn snapshot_binary_subpaths_use_the_snapshot_top_dir_not_the_cft_one() {
251        assert_eq!(
252            snapshot_binary_subpath(Platform::LinuxX64),
253            Path::new("chrome-linux/chrome")
254        );
255        assert_eq!(
256            snapshot_binary_subpath(Platform::Win64),
257            Path::new("chrome-win/chrome.exe")
258        );
259        assert_eq!(
260            snapshot_binary_subpath(Platform::MacArm64),
261            Path::new("chrome-mac/Chromium.app/Contents/MacOS/Chromium")
262        );
263        // Snapshots are Chromium, not Chrome for Testing — different tree.
264        assert_ne!(
265            snapshot_binary_subpath(Platform::LinuxX64),
266            cft_binary_subpath(Platform::LinuxX64)
267        );
268    }

270    #[test]
271    fn ungoogled_binary_subpaths_cover_the_three_packagings() {
272        let win_top = "ungoogled-chromium_151.0.7922.71-1.1_windows_x64";
273        assert_eq!(
274            ungoogled_binary_subpath(Platform::Win64, win_top),
275            Path::new(win_top).join("chrome.exe")
276        );
277        assert_eq!(
278            ungoogled_binary_subpath(Platform::LinuxX64, ""),
279            Path::new("chrome.AppImage")
280        );
281        assert_eq!(
282            ungoogled_binary_subpath(Platform::MacArm64, ""),
283            Path::new("Chromium.app/Contents/MacOS/Chromium")
284        );
285    }
