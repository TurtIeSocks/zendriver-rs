1//! Which *build* of Chromium to fetch.
2//!
3//! The crate started life as a Chrome for Testing downloader, and Chrome for
4//! Testing remains the default — [`Distribution::default()`] is
5//! [`Distribution::ChromeForTesting`], so a caller that never mentions a
6//! distribution gets exactly the behaviour it always had.
7//!
8//! The three distributions differ **only** in how a
9//! [`VersionSpec`](crate::VersionSpec) + [`Platform`] pair turns into a
10//! download URL. Everything downstream — streaming, SHA256 verification,
11//! extraction, the atomic per-build cache — is shared. Resolution is the
12//! whole distribution axis:
13//!
14//! | Distribution | Index | Keyed by |
15//! |---|---|---|
16//! | [`ChromeForTesting`](Distribution::ChromeForTesting) | one JSON manifest for every platform | version |
17//! | [`UngoogledChromium`](Distribution::UngoogledChromium) | three GitHub repos, one per OS | version (tag *prefix*) |
18//! | [`ChromiumSnapshot`](Distribution::ChromiumSnapshot) | a GCS bucket per platform | **revision** |
19//!
20//! That last row is the awkward one and the reason
21//! [`VersionSpec::Revision`](crate::VersionSpec::Revision) exists — see
22//! [`Distribution::ChromiumSnapshot`].
23
24use crate::platform::Platform;

26/// Which Chromium build to download.
27///
28/// Defaults to [`Distribution::ChromeForTesting`].
29///
30/// ```
31/// use zendriver_fetcher::Distribution;
32/// assert_eq!(Distribution::default(), Distribution::ChromeForTesting);
33/// ```
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
35#[non_exhaustive]
36pub enum Distribution {
37    /// Google's [Chrome for Testing][cft] builds: one JSON manifest listing
38    /// every version on every platform, with a stable URL scheme and no
39    /// auto-update. The default, and the only distribution that is really
40    /// *Chrome* rather than Chromium.
41    ///
42    /// [cft]: https://googlechromelabs.github.io/chrome-for-testing/
43    #[default]
44    ChromeForTesting,
45
46    /// [ungoogled-chromium][ug]: Chromium with Google integration stripped out.
47    ///
48    /// There is no single manifest. Binaries are built by **three separate
49    /// per-OS repos** under the `ungoogled-software` org, each cutting its own
50    /// releases on its own cadence, so version availability genuinely differs
51    /// per platform — as of 2026-08-06 Windows and Linux were on
52    /// `151.0.7922.71` while macOS was still on `150.0.7871.46`. Resolution
53    /// reports what the *requested* platform actually has; it never assumes
54    /// parity.
55    ///
56    /// The org's main `ungoogled-chromium` repo also tags releases, but they
57    /// carry **zero assets** — it is the source/patches repo. It is
58    /// deliberately not used as a binary source.
59    ///
60    /// [ug]: https://github.com/ungoogled-software/ungoogled-chromium
61    UngoogledChromium,
62
63    /// Per-commit [Chromium continuous-build snapshots][snap] from Google's
64    /// GCS bucket.
65    ///
66    /// **Snapshots are keyed by revision, not by version.** The bucket is laid
67    /// out `<platform>/<revision>/chrome-<os>.zip` and publishes no
68    /// version→revision index, so
69    /// [`VersionSpec::Explicit`](crate::VersionSpec::Explicit) cannot be
70    /// resolved here and is refused rather than silently downgraded to "the
71    /// newest snapshot" — handing back a different browser than the one asked
72    /// for is worse than an error. Pin a build with
73    /// [`VersionSpec::Revision`](crate::VersionSpec::Revision) instead, or take
74    /// the tip with [`VersionSpec::Latest`](crate::VersionSpec::Latest).
75    ///
76    /// [snap]: https://commondatastorage.googleapis.com/chromium-browser-snapshots/index.html
77    ChromiumSnapshot,
78}
79
80impl Distribution {
81    /// Every distribution, in menu order. Used by the CLI's interactive
82    /// picker and by tests that assert exhaustive coverage.
83    pub const ALL: &'static [Distribution] = &[
84        Distribution::ChromeForTesting,
85        Distribution::UngoogledChromium,
86        Distribution::ChromiumSnapshot,
87    ];
88
89    /// Short lowercase slug — the value accepted by `--distribution` and used
90    /// as the cache sub-directory name.
91    ///
92    /// ```
93    /// use zendriver_fetcher::Distribution;
94    /// assert_eq!(Distribution::UngoogledChromium.slug(), "ungoogled");
95    /// ```
96    pub fn slug(self) -> &'static str {
97        match self {
98            Distribution::ChromeForTesting => "cft",
99            Distribution::UngoogledChromium => "ungoogled",
100            Distribution::ChromiumSnapshot => "snapshot",
101        }
102    }
103
104    /// Human-readable name, for CLI menus and error messages.
105    pub fn title(self) -> &'static str {
106        match self {
107            Distribution::ChromeForTesting => "Chrome for Testing",
108            Distribution::UngoogledChromium => "ungoogled-chromium",
109            Distribution::ChromiumSnapshot => "Chromium snapshot",
110        }
111    }
112
113    /// Parse a slug. Accepts a few obvious aliases so the CLI is forgiving.
114    ///
115    /// ```
116    /// use zendriver_fetcher::Distribution;
117    /// assert_eq!(
118    ///     Distribution::parse("chrome-for-testing"),
119    ///     Some(Distribution::ChromeForTesting)
120    /// );
121    /// assert_eq!(Distribution::parse("nope"), None);
122    /// ```
123    pub fn parse(s: &str) -> Option<Self> {
124        match s.trim().to_ascii_lowercase().as_str() {
125            "cft" | "chrome-for-testing" | "chrome_for_testing" | "chrome" => {
126                Some(Distribution::ChromeForTesting)
127            }
128            "ungoogled" | "ungoogled-chromium" | "ungoogled_chromium" => {
129                Some(Distribution::UngoogledChromium)
130            }
131            "snapshot" | "snapshots" | "chromium-snapshot" | "chromium_snapshot" => {
132                Some(Distribution::ChromiumSnapshot)
133            }
134            _ => None,
135        }
136    }
137
138    /// The `owner/name` GitHub repo publishing ungoogled-chromium binaries for
139    /// `platform`.
140    ///
141    /// The three OSes are built by three different repos — that split is the
142    /// entire reason ungoogled needs per-platform resolution rather than one
143    /// manifest lookup.
144    ///
145    /// ```
146    /// use zendriver_fetcher::{Distribution, Platform};
147    /// assert_eq!(
148    ///     Distribution::ungoogled_repo(Platform::MacArm64),
149    ///     "ungoogled-software/ungoogled-chromium-macos"
150    /// );
151    /// ```
152    pub fn ungoogled_repo(platform: Platform) -> &'static str {
153        match platform {
154            Platform::MacX64 | Platform::MacArm64 => "ungoogled-software/ungoogled-chromium-macos",
155            Platform::Win32 | Platform::Win64 => "ungoogled-software/ungoogled-chromium-windows",
156            // "portablelinux" ships relocatable AppImage/tarball builds; the
157            // sibling `ungoogled-chromium-archlinux` / `-debian` repos publish
158            // distro packages, which are not usable as a drop-in binary.
159            Platform::LinuxX64 => "ungoogled-software/ungoogled-chromium-portablelinux",
160        }
161    }
162}

164#[cfg(test)]
165#[allow(clippy::panic, clippy::unwrap_used)]
166mod tests {
167    use super::*;
168
169    #[test]
170    fn default_is_chrome_for_testing_so_existing_callers_are_untouched() {
171        assert_eq!(Distribution::default(), Distribution::ChromeForTesting);
172    }
173
174    #[test]
175    fn slug_round_trips_through_parse_for_every_distribution() {
176        for &d in Distribution::ALL {
177            assert_eq!(Distribution::parse(d.slug()), Some(d), "slug {}", d.slug());
178        }
179    }
180
181    #[test]
182    fn parse_is_case_insensitive_and_rejects_unknown() {
183        assert_eq!(
184            Distribution::parse("  UnGoogled  "),
185            Some(Distribution::UngoogledChromium)
186        );
187        assert_eq!(Distribution::parse("firefox"), None);
188    }
189
190    #[test]
191    fn ungoogled_maps_the_three_oses_to_three_different_repos() {
192        let mac = Distribution::ungoogled_repo(Platform::MacArm64);
193        let win = Distribution::ungoogled_repo(Platform::Win64);
194        let linux = Distribution::ungoogled_repo(Platform::LinuxX64);
195
196        assert_eq!(mac, "ungoogled-software/ungoogled-chromium-macos");
197        assert_eq!(win, "ungoogled-software/ungoogled-chromium-windows");
198        assert_eq!(linux, "ungoogled-software/ungoogled-chromium-portablelinux");
199
200        // The point of the mapping: three OSes, three distinct repos.
201        assert_ne!(mac, win);
202        assert_ne!(win, linux);
203        assert_ne!(mac, linux);
204
205        // Both arches of an OS share that OS's repo.
206        assert_eq!(mac, Distribution::ungoogled_repo(Platform::MacX64));
207        assert_eq!(win, Distribution::ungoogled_repo(Platform::Win32));
208    }
209
210    #[test]
211    fn ungoogled_never_points_at_the_assetless_source_repo() {
212        // `ungoogled-software/ungoogled-chromium` tags releases with zero
213        // assets (it is the patch set, not a build). Resolving against it
214        // would always fail to find a binary.
215        for &p in &[
216            Platform::MacX64,
217            Platform::MacArm64,
218            Platform::Win32,
219            Platform::Win64,
220            Platform::LinuxX64,
221        ] {
222            assert_ne!(
223                Distribution::ungoogled_repo(p),
224                "ungoogled-software/ungoogled-chromium",
225                "{p:?} points at the source-only repo"
226            );
227        }
228    }
229}
