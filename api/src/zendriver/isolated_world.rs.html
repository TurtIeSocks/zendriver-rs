1//! Per-frame isolated-world execution-context cache.
2//!
3//! Shared between [`crate::tab::Tab`] (main-frame eval) and
4//! [`crate::frame::Frame`] (per-frame eval). Holds the
5//! `executionContextId` returned by `Page.createIsolatedWorld` (cached
6//! after first use; invalidated when Chrome reports
7//! "Cannot find context with specified id", typically after a navigation
8//! destroys the previous context).
9//!
10//! Pre-P4 this lived as a private type inside `tab.rs`. P4 promotes it to
11//! a shared module so [`crate::frame::Frame`] can carry its own per-frame
12//! cache without duplicating the struct definition. The visibility stays
13//! `pub(crate)` — neither the cache nor its contents are part of the public
14//! API.
15
16/// Cache of the `executionContextId` for one frame's `zendriver-eval`
17/// isolated world.
18///
19/// Starts `None`. [`crate::tab::Tab::ensure_isolated_world`] populates it
20/// with a `Page.getFrameTree` + `Page.createIsolatedWorld` pair;
21/// [`crate::frame::Frame::ensure_isolated_world`] already knows its own
22/// `frameId` and skips the first of those. Later calls short-circuit on the
23/// cached value.
24///
25/// When [`crate::tab::Tab::evaluate`] or [`crate::frame::Frame::evaluate`]
26/// catches Chrome's `-32000 Cannot find context with specified id` error
27/// (see [`is_stale_context`]), it sets `context_id = None` and retries, so
28/// the next `ensure_isolated_world` call runs that discovery again from
29/// scratch. The frame id is deliberately not cached alongside it: reusing
30/// one across the navigation that invalidated the context is how a stale
31/// `frameId` reaches `Page.createIsolatedWorld`, and the round-trip it
32/// would save is one per navigation.
33#[derive(Default, Debug)]
34pub(crate) struct IsolatedWorldCache {
35    pub(crate) context_id: Option<i64>,
36}
37
38/// Whether `e` is Chrome telling us the `executionContextId` we cached no
39/// longer exists — the signal to drop `context_id` and retry once.
40///
41/// Chrome reports this as `-32000 Cannot find context with specified id`,
42/// but [`crate::error::ZendriverError`]'s `From<CallError>` maps `-32000`
43/// onto [`crate::error::ZendriverError::Navigation`], so the match is on
44/// that variant and not on `Cdp`.
45///
46/// Single source of truth on purpose: [`crate::tab::Tab::evaluate`] and
47/// [`crate::frame::Frame::evaluate`] both need it, and a magic string
48/// duplicated across two files is exactly the thing that drifts. (It did:
49/// `Frame::evaluate` had no retry at all until this was hoisted, so a
50/// navigated iframe's handle failed permanently while the tab-level call
51/// recovered transparently.)
52pub(crate) fn is_stale_context(e: &crate::error::ZendriverError) -> bool {
53    matches!(e, crate::error::ZendriverError::Navigation(m) if m.contains("Cannot find context"))
54}
