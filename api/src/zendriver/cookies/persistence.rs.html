1//! JSON persistence for [`CookieJar`].
2//!
3//! Round-trips the entire browser cookie store through a file on disk so
4//! callers can hydrate a fresh browser session with cookies captured from a
5//! prior run — e.g. resume an authenticated scrape without re-running the
6//! login flow.
7//!
8//! The on-disk shape is the pretty-printed [`Vec<Cookie>`] (snake_case JSON
9//! per the module-level docs in [`crate::cookies`]) — straightforward to
10//! diff, edit by hand, or feed to other tools.
11//!
12//! ## Lossless round-trip
13//!
14//! The `url` field on [`crate::cookies::Cookie`] is input-only: CDP uses
15//! it on `.set()` to infer `domain` / `path` / `secure`, but never emits
16//! it on reads. `.save_to_file()` therefore serializes whatever
17//! `.all()` returned — `url` always `None`, omitted by serde — and
18//! `.load_from_file()` reads back the same shape. `domain` / `path` /
19//! `secure` are populated explicitly on every cookie from `.all()`, so
20//! `.set_many()` after a load reconstructs the store without needing
21//! `url` re-inference. If you hand-author a JSON file with a non-null
22//! `url`, it round-trips faithfully too (serde preserves `Some` values).
23

32impl CookieJar {
33    /// Snapshot the cookie store to a JSON file at `path`.
34    ///
35    /// Issues a single `Storage.getCookies` round-trip, then writes the
36    /// pretty-printed array. The write is atomic — a sibling temp file is
37    /// filled and renamed over the destination — so an interrupted save
38    /// leaves the previous file intact instead of a truncated one that no
39    /// longer parses. The file is overwritten if it already exists. Parent
40    /// directories must already exist — `save_to_file` does not create them.
41    ///
42    /// On Unix an existing file keeps its permissions across saves, and a file
43    /// created by this call is `0600`: it holds live session cookies, so the
44    /// default is owner-only rather than whatever the process umask allows.
45    /// Because the write finishes with a `rename`, a `path` that is a *symlink*
46    /// is replaced by a regular file — the link target stops receiving saves.
47    /// Pass the real path if you were relying on that indirection.
48    ///
49    /// # Errors
50    ///
51    /// Returns [`crate::ZendriverError::Io`] if the path is unwritable;
52    /// [`crate::ZendriverError::Transport`] / `Cdp` on CDP failures.
53    ///
54    /// Because the save fills a sibling temp file and renames it over the
55    /// destination, it needs more than a writable destination: the *parent
56    /// directory* must be writable and executable, and the rename must be able
57    /// to replace what is at `path`. A destination that cannot be replaced by a
58    /// rename therefore fails even though it is perfectly writable — a
59    /// bind-mounted single file (`docker run -v $PWD/cookies.json:/app/cookies.json`)
60    /// is the common case, and a file held open by another process is the
61    /// Windows one.
62    ///
63    /// # Examples
64    ///
65    /// ```no_run
66    /// # async fn ex() -> zendriver::Result<()> {
67    /// # let browser = zendriver::Browser::builder().launch().await?;
68    /// browser.cookies().save_to_file("cookies.json").await?;
69    /// # Ok(()) }
70    /// ```
71    pub async fn save_to_file(&self, path: impl AsRef<Path>) -> Result<()> {
72        let cookies = self.all().await?;
73        let bytes = serde_json::to_vec_pretty(&cookies)?;
74        write_atomic(path.as_ref(), &bytes).await?;
75        Ok(())
76    }
77
78    /// Hydrate the browser cookie store from a JSON file at `path`.
79    ///
80    /// Reads the file, deserializes a `Vec<Cookie>`, and dispatches a
81    /// single `Storage.setCookies` bulk-set. Existing cookies in the
82    /// browser are NOT cleared first — call [`CookieJar::clear`] before
83    /// this method for a fresh slate.
84    ///
85    /// # Errors
86    ///
87    /// Returns [`crate::ZendriverError::Io`] if the file is unreadable;
88    /// [`crate::ZendriverError::Serde`] if the JSON is malformed.
89    ///
90    /// # Examples
91    ///
92    /// ```no_run
93    /// # async fn ex() -> zendriver::Result<()> {
94    /// # let browser = zendriver::Browser::builder().launch().await?;
95    /// browser.cookies().load_from_file("cookies.json").await?;
96    /// # Ok(()) }
97    /// ```
98    pub async fn load_from_file(&self, path: impl AsRef<Path>) -> Result<()> {
99        let bytes = fs::read(path).await?;
100        let cookies: Vec<crate::cookies::Cookie> = serde_json::from_slice(&bytes)?;
101        self.set_many(cookies).await
102    }
103
104    /// Snapshot only the cookies matching `filter` to a JSON file at `path`.
105    ///
106    /// Like [`Self::save_to_file`] — including the atomic temp-file-then-
107    /// rename write and its permission/symlink behavior — but applies the
108    /// `filter` predicate to the result of
109    /// [`CookieJar::all`] before writing, handy for persisting just one
110    /// site's cookies out of a shared store. The predicate receives each
111    /// [`crate::cookies::Cookie`] by reference and returns `true` to keep it.
112    ///
113    /// # Errors
114    ///
115    /// Returns [`crate::ZendriverError::Io`] if the path is unwritable;
116    /// [`crate::ZendriverError::Transport`] / `Cdp` on CDP failures. The
117    /// temp-file-and-rename requirements are [`Self::save_to_file`]'s, in
118    /// full: a writable, executable parent directory, and a destination a
119    /// rename can replace.
120    ///
121    /// # Examples
122    ///
123    /// ```no_run
124    /// # async fn ex() -> zendriver::Result<()> {
125    /// # let browser = zendriver::Browser::builder().launch().await?;
126    /// browser
127    ///     .cookies()
128    ///     .save_to_file_matching("example.json", |c| c.domain.contains("example.com"))
129    ///     .await?;
130    /// # Ok(()) }
131    /// ```
132    pub async fn save_to_file_matching(
133        &self,
134        path: impl AsRef<Path>,
135        filter: impl Fn(&crate::cookies::Cookie) -> bool,
136    ) -> Result<()> {
137        let cookies: Vec<crate::cookies::Cookie> = self
138            .all()
139            .await?
140            .into_iter()
141            .filter(|c| filter(c))
142            .collect();
143        let bytes = serde_json::to_vec_pretty(&cookies)?;
144        write_atomic(path.as_ref(), &bytes).await?;
145        Ok(())
146    }
147
148    /// Hydrate only the cookies matching `filter` from a JSON file at `path`.
149    ///
150    /// Like [`Self::load_from_file`], but applies the `filter` predicate to
151    /// the parsed `Vec<Cookie>` before the `Storage.setCookies` bulk-set —
152    /// so a file holding many sites' cookies can be loaded selectively.
153    /// Existing cookies are NOT cleared first; call [`CookieJar::clear`]
154    /// beforehand for a fresh slate.
155    ///
156    /// # Errors
157    ///
158    /// Returns [`crate::ZendriverError::Io`] if the file is unreadable;
159    /// [`crate::ZendriverError::Serde`] if the JSON is malformed.
160    ///
161    /// # Examples
162    ///
163    /// ```no_run
164    /// # async fn ex() -> zendriver::Result<()> {
165    /// # let browser = zendriver::Browser::builder().launch().await?;
166    /// browser
167    ///     .cookies()
168    ///     .load_from_file_matching("cookies.json", |c| c.domain.contains("example.com"))
169    ///     .await?;
170    /// # Ok(()) }
171    /// ```
172    pub async fn load_from_file_matching(
173        &self,
174        path: impl AsRef<Path>,
175        filter: impl Fn(&crate::cookies::Cookie) -> bool,
176    ) -> Result<()> {
177        let bytes = fs::read(path).await?;
178        let cookies: Vec<crate::cookies::Cookie> = serde_json::from_slice(&bytes)?;
179        let cookies: Vec<crate::cookies::Cookie> =
180            cookies.into_iter().filter(|c| filter(c)).collect();
181        self.set_many(cookies).await
182    }
183}

200    /// Mode of `path`, permission bits only. Unix-only helper for the
201    /// permission tests below. (The helper's exhaustive coverage — the
202    /// setuid/setgid carry-over, the mid-write window, the exclusive create —
203    /// lives with the helper itself in [`crate::io`]; what is left here is the
204    /// jar's own end-to-end behavior.)
205    #[cfg(unix)]
206    fn mode_of(path: &std::path::Path) -> u32 {
207        use std::os::unix::fs::PermissionsExt;
208        std::fs::metadata(path).unwrap().permissions().mode() & 0o777
209    }

211    /// Drive one save through the chosen public entry point, answering the
212    /// `Storage.getCookies` round-trip with a single cookie, and return once
213    /// the file is on disk.
214    ///
215    /// `matching` picks `save_to_file_matching` (with a keep-everything
216    /// predicate) over `save_to_file`, so a caller that loops over both values
217    /// asserts its property against *both* save paths — which is the whole
218    /// point of the tests below, since the two call the write helper
219    /// independently and only one of them might be changed.
220    async fn save_one_cookie(path: &std::path::Path, matching: bool) {
221        let (mut mock, conn) = MockConnection::pair();
222        let jar = CookieJar::new(conn.clone());
223
224        let save = tokio::spawn({
225            let j = jar.clone();
226            let p = path.to_path_buf();
227            async move {
228                if matching {
229                    j.save_to_file_matching(p, |_| true).await
230                } else {
231                    j.save_to_file(p).await
232                }
233            }
234        });
235
236        let id = mock.expect_cmd("Storage.getCookies").await;
237        mock.reply(
238            id,
239            json!({
240                "cookies": [
241                    { "name": "a", "value": "1", "domain": ".x.test", "path": "/",
242                      "httpOnly": false, "secure": false },
243                ]
244            }),
245        )
246        .await;
247        save.await.unwrap().unwrap();
248
249        conn.shutdown();
250    }

257    /// A mode the user chose on the jar survives a save.
258    ///
259    /// `0640` rather than `0600` on purpose: it is both narrower than the
260    /// `0644` a default umask produces *and* different from the `0600` the temp
261    /// file is created with, so this is the fixture that notices if the
262    /// destination-mode carry-over inside [`crate::io::write_atomic`] is
263    /// dropped. A `0600` jar would come back `0600` either way and assert
264    /// nothing.
265    ///
266    /// It says nothing about *which* write the jar used — `fs::write` on an
267    /// existing inode preserves its mode too. The two tests that can tell the
268    /// writes apart are [`both_save_paths_write_atomically`] and
269    /// [`both_save_paths_replace_a_symlinked_destination`].
270    #[cfg(unix)]
271    #[tokio::test]
272    async fn save_to_file_preserves_the_destination_mode() {
273        use std::os::unix::fs::PermissionsExt;
274
275        let dir = tempfile::tempdir().unwrap();
276        let path = dir.path().join("cookies.json");
277        std::fs::write(&path, b"stale").unwrap();
278        std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o640)).unwrap();
279
280        save_one_cookie(&path, false).await;
281
282        let mode = mode_of(&path);
283        assert_eq!(
284            mode, 0o640,
285            "a save must carry the destination's own mode over, got {mode:o}"
286        );
287        assert_eq!(dir_entry_count(dir.path()), 1, "no temp file may be left");
288    }

290    /// End-to-end round-trip: dump the cookie store to disk, then load it back
291    /// into a fresh jar. The mock receives `Storage.getCookies` on save,
292    /// then `Storage.setCookies` on load — assert the payload preserves both
293    /// entries with their CDP camelCase fields intact.
294    #[tokio::test]
295    async fn save_and_load_roundtrip_preserves_cookies() {
296        let (mut mock, conn) = MockConnection::pair();
297        let jar = CookieJar::new(conn.clone());
298        let tmp = tempfile::NamedTempFile::new().unwrap();
299        let path = tmp.path().to_path_buf();
300
301        // --- Save half: Storage.getCookies → write to tempfile.
302        let save = tokio::spawn({
303            let j = jar.clone();
304            let p = path.clone();
305            async move { j.save_to_file(p).await }
306        });
307
308        let id = mock.expect_cmd("Storage.getCookies").await;
309        mock.reply(
310            id,
311            json!({
312                "cookies": [
313                    {
314                        "name": "a",
315                        "value": "1",
316                        "domain": ".x.test",
317                        "path": "/",
318                        "expires": 1_700_000_000.0,
319                        "httpOnly": true,
320                        "secure": true,
321                        "sameSite": "Lax",
322                    },
323                    {
324                        "name": "b",
325                        "value": "2",
326                        "domain": "x.test",
327                        "path": "/api",
328                        "httpOnly": false,
329                        "secure": false,
330                    },
331                ]
332            }),
333        )
334        .await;
335        save.await.unwrap().unwrap();
336
337        // Sanity-check the on-disk shape — snake_case, two entries.
338        let on_disk = std::fs::read_to_string(&path).unwrap();
339        let parsed: serde_json::Value = serde_json::from_str(&on_disk).unwrap();
340        let arr = parsed.as_array().unwrap();
341        assert_eq!(arr.len(), 2);
342        assert_eq!(arr[0]["name"], "a");
343        assert_eq!(arr[0]["http_only"], true);
344        assert_eq!(arr[0]["same_site"], "Lax");
345        assert_eq!(arr[1]["name"], "b");
346
347        // --- Load half: read tempfile → Storage.setCookies bulk-set.
348        let load = tokio::spawn({
349            let j = jar.clone();
350            let p = path.clone();
351            async move { j.load_from_file(p).await }
352        });
353
354        let id = mock.expect_cmd("Storage.setCookies").await;
355        let params = &mock.last_sent()["params"];
356        let cookies = params["cookies"].as_array().unwrap();
357        assert_eq!(cookies.len(), 2);
358        assert_eq!(cookies[0]["name"], "a");
359        assert_eq!(cookies[0]["value"], "1");
360        assert_eq!(cookies[0]["domain"], ".x.test");
361        assert_eq!(cookies[0]["httpOnly"], true);
362        assert_eq!(cookies[0]["sameSite"], "Lax");
363        // No snake_case leakage on the wire.
364        assert!(cookies[0].get("http_only").is_none());
365        assert!(cookies[0].get("same_site").is_none());
366        assert_eq!(cookies[1]["name"], "b");
367        assert_eq!(cookies[1]["path"], "/api");
368
369        mock.reply(id, json!({})).await;
370        load.await.unwrap().unwrap();
371
372        // SameSite preserved through the full round-trip.
373        let reparsed: Vec<crate::cookies::Cookie> = serde_json::from_str(&on_disk).unwrap();
374        assert_eq!(reparsed[0].same_site, Some(SameSite::Lax));
375        assert_eq!(reparsed[1].same_site, None);
376
377        conn.shutdown();
378    }

497    /// Both save paths must reach the destination through
498    /// [`crate::io::write_atomic`] rather than a plain `fs::write`, and the
499    /// assertion that separates the two is the mode of a **jar that did not
500    /// exist yet**: the helper creates its temp file `0600` whatever the
501    /// ambient umask is, while `fs::write` creates at `0666 & ~umask`.
502    ///
503    /// Asserting on an *existing* destination cannot make that distinction —
504    /// `O_WRONLY|O_CREAT|O_TRUNC` on a live inode ignores its mode argument, so
505    /// both writes leave the previous permissions in place. That is why this
506    /// test starts from an empty directory.
507    ///
508    /// One environment caveat on the discriminating power, the same one
509    /// [`crate::io`]'s own mid-write test carries: under a `0077` umask or
510    /// tighter `fs::write` also lands on `0600` and this proves nothing.
511    /// [`both_save_paths_replace_a_symlinked_destination`] is the half that
512    /// holds regardless of umask.
513    ///
514    /// The second save, over a longer stale file, then shows the rename
515    /// replacing the content rather than overwriting a prefix of it, and the
516    /// directory count shows the temp file was renamed rather than abandoned.
517    #[tokio::test]
518    async fn both_save_paths_write_atomically() {
519        for matching in [false, true] {
520            let dir = tempfile::tempdir().unwrap();
521            let path = dir.path().join("cookies.json");
522            assert!(
523                !path.exists(),
524                "matching={matching}: the mode assertion below only means something \
525                 for a destination this save creates"
526            );
527
528            save_one_cookie(&path, matching).await;
529
530            #[cfg(unix)]
531            {
532                let mode = mode_of(&path);
533                assert_eq!(
534                    mode, 0o600,
535                    "matching={matching}: a jar created by a save holds live session \
536                     cookies and must be owner-only, got {mode:o}"
537                );
538            }
539            assert_eq!(saved_cookies(&path).len(), 1, "matching={matching}");
540            assert_eq!(
541                dir_entry_count(dir.path()),
542                1,
543                "matching={matching}: save must leave no temp file behind"
544            );
545
546            std::fs::write(&path, b"stale-and-longer-than-the-new-content").unwrap();
547            save_one_cookie(&path, matching).await;
548
549            let parsed = saved_cookies(&path);
550            assert_eq!(parsed.len(), 1, "matching={matching}");
551            assert_eq!(parsed[0].name, "a", "matching={matching}");
552            assert_eq!(
553                dir_entry_count(dir.path()),
554                1,
555                "matching={matching}: the second save must leave no temp file either"
556            );
557        }
558    }

560    /// The umask-independent half: `fs::write` follows a symlinked destination
561    /// and writes through to its target, while the atomic write finishes with a
562    /// `rename` that unlinks the symlink and leaves a regular file in its
563    /// place. No ambient setting changes that, so this separates the two writes
564    /// on every box.
565    ///
566    /// It doubles as the pin for the behavior change `save_to_file`'s rustdoc
567    /// warns about: a caller who symlinked their jar at some canonical store
568    /// stops feeding that store.
569    #[cfg(unix)]
570    #[tokio::test]
571    async fn both_save_paths_replace_a_symlinked_destination() {
572        for matching in [false, true] {
573            let dir = tempfile::tempdir().unwrap();
574            let real = dir.path().join("real.json");
575            let link = dir.path().join("cookies.json");
576            std::fs::write(&real, b"original").unwrap();
577            std::os::unix::fs::symlink(&real, &link).unwrap();
578
579            save_one_cookie(&link, matching).await;
580
581            assert!(
582                !std::fs::symlink_metadata(&link).unwrap().is_symlink(),
583                "matching={matching}: the rename must leave a regular file where the link was"
584            );
