1//! [`Element::screenshot`] — element-scoped PNG capture.
2//!
3//! Dispatch sequence (with internal refresh-on-stale recovery), matching the
4//! `scroll → gate → measure → dispatch` shape every action in
5//! [`mod@crate::element::actions`] uses:
6//!   1. [`Element::scroll_into_view`] — the visibility gate in step 2
7//!      requires the element to overlap the viewport, so a below-the-fold
8//!      element has to be scrolled to before it can be gated. The clip in
9//!      step 4 does *not* need the scroll: it is document-relative and the
10//!      conversion there is correct wherever the element sits.
11//!   2. Visibility gate — we need pixels to capture; overlay occlusion +
12//!      disabled state are irrelevant here, so the gate is the lightest
13//!      preset.
14//!   3. [`Element::bounding_box`] — viewport-relative quad of the element.
15//!   4. `Page.getLayoutMetrics` — the visual viewport's page offset, added
16//!      to that quad. `Page.captureScreenshot` reads `clip` in document
17//!      coordinates, so a viewport-relative rect would crop the wrong band
18//!      of a scrolled page.
19//!   5. `Page.captureScreenshot { format: "png", clip: { x, y, width,
20//!      height, scale: 1 } }` — crop to the element's rect at native scale.
21//!   6. base64-decode the `data` field into raw PNG bytes.
22
23use std::time::Duration;
24
25use base64::Engine as _;
26use base64::engine::general_purpose::STANDARD as BASE64;
27use serde_json::{Value, json};
28
29use crate::element::Element;
30use crate::error::{Result, ZendriverError};
31use crate::query::actionability::{self, ActionabilityCheck};
32
33/// Default deadline for the visibility gate before the capture call.
34/// Matches the value used by other Element actions; per-call override
35/// lands in P4 when the per-action options structs grow.
36const DEFAULT_ACTIONABILITY_TIMEOUT: Duration = Duration::from_secs(5);
37
38impl Element {
39    /// Capture a PNG screenshot cropped to this element's bounding box.
40    ///
41    /// Scrolls the element into view, waits up to 5 s for it to become
42    /// visible, reads its bbox via `DOM.getBoxModel`, converts that to
43    /// document coordinates with `Page.getLayoutMetrics`, then sends
44    /// `Page.captureScreenshot` with a matching `clip` rect (at `scale: 1`).
45    /// Returns the raw PNG bytes.
46    ///
47    /// The scroll leads because the gate requires the element to overlap the
48    /// viewport; the coordinate conversion follows because Chrome reads the
49    /// clip in document space while the box model is viewport-relative. The
50    /// two are independent — the conversion is correct at any scroll offset.
51    ///
52    /// For full-viewport captures, see [`crate::Tab::screenshot`].
53    ///
54    /// # Errors
55    ///
56    /// Returns [`ZendriverError::NotActionable`] if the element doesn't
57    /// become visible within the 5s gate timeout;
58    /// [`ZendriverError::Navigation`] if Chrome returns no bbox or no
59    /// screenshot data.
60    ///
61    /// # Examples
62    ///
63    /// ```no_run
64    /// # async fn ex() -> zendriver::Result<()> {
65    /// # let browser = zendriver::Browser::builder().launch().await?;
66    /// # let tab = browser.main_tab();
67    /// let card = tab.find().css(".card").one().await?;
68    /// let png_bytes = card.screenshot().await?;
69    /// tokio::fs::write("card.png", png_bytes).await?;
70    /// # Ok(()) }
71    /// ```
72    pub async fn screenshot(&self) -> Result<Vec<u8>> {
73        self.with_refresh(|| async move {
74            // Scroll first, exactly as every action in `element::actions`
75            // does, and for one reason: the visibility predicate requires the
76            // element to overlap the viewport, so gating before the scroll
77            // would reject every below-the-fold element. It is NOT for the
78            // clip — see the conversion below, which is correct at any scroll
79            // offset.
80            self.scroll_into_view().await?;
81            actionability::wait_actionable(
82                self,
83                ActionabilityCheck::VISIBLE_ONLY,
84                DEFAULT_ACTIONABILITY_TIMEOUT,
85            )
86            .await?;
87            let bbox = self
88                .bounding_box()
89                .await?
90                .ok_or_else(|| ZendriverError::Navigation("element has no bounding box".into()))?;
91
92            // Chrome reads `clip` in DOCUMENT coordinates while
93            // `bounding_box` is viewport-relative, so the scroll offset has
94            // to be added back or a scrolled page gets cropped from the wrong
95            // region entirely. The offset comes from CDP rather than
96            // `window.scrollX` — one less page-controlled input on a path a
97            // page would happily lie to.
98            let metrics = self
99                .inner
100                .tab
101                .call("Page.getLayoutMetrics", json!({}))
102                .await?;
103            let viewport = metrics.get("cssVisualViewport");
104            let offset = |axis: &str| viewport.and_then(|v| v.get(axis)).and_then(Value::as_f64);
105            let (page_x, page_y) = match (offset("pageX"), offset("pageY")) {
106                (Some(x), Some(y)) => (x, y),
107                _ => {
108                    // Falling back to (0, 0) reverts the clip to viewport
109                    // coordinates — precisely the bug this conversion exists
110                    // to fix — and the symptom is a screenshot cropped from
111                    // the wrong band of the page, which reads as a product
112                    // bug with nothing in the logs pointing here. Not
113                    // reachable on any browser this crate supports
114                    // (`cssVisualViewport` long predates the Chrome 121 floor
115                    // the visibility gate imposes), so this is a tripwire
116                    // rather than a handled case.
117                    tracing::warn!(
118                        "Page.getLayoutMetrics returned no cssVisualViewport page offset; \
119                         the screenshot clip falls back to viewport coordinates and will be \
120                         cropped from the wrong region on a scrolled page"
121                    );
122                    (0.0, 0.0)
123                }
124            };
125
126            let res = self
127                .inner
128                .tab
129                .call(
130                    "Page.captureScreenshot",
131                    json!({
132                        "format": "png",
133                        "clip": {
134                            "x": bbox.x + page_x,
135                            "y": bbox.y + page_y,
136                            "width": bbox.width,
137                            "height": bbox.height,
138                            "scale": 1,
139                        },
140                    }),
141                )
142                .await?;
143            let data = res.get("data").and_then(|v| v.as_str()).ok_or_else(|| {
144                ZendriverError::Navigation("Page.captureScreenshot returned no data".into())
145            })?;
146            BASE64.decode(data).map_err(|e| {
147                ZendriverError::Navigation(format!("invalid base64 in screenshot: {e}"))
148            })
149        })
150        .await
151    }
152}

154#[cfg(test)]
155#[allow(clippy::panic, clippy::unwrap_used)]
156mod tests {
157    use super::*;
158    use crate::tab::Tab;
159    use crate::test_support::{expect, serve_call_js};
160    use zendriver_transport::SessionHandle;
161    use zendriver_transport::testing::MockConnection;
162
163    #[tokio::test]
164    async fn screenshot_sends_page_capturescreenshot_with_clip_matching_bbox() {
165        let (mut mock, conn) = MockConnection::pair();
166        let sess = SessionHandle::new(conn.clone(), "S1");
167        let tab = Tab::new_for_test(sess);
168        let el = Element::from_jsret(tab, 7, "R1".to_string());
169
170        let fut = tokio::spawn({
171            let e = el.clone();
172            async move { e.screenshot().await }
173        });
174
175        // Step 1: scroll_into_view — MUST land before both the gate and the
176        // measurement, or an off-screen element is rejected as "not visible"
177        // and, past that, clipped from stale coordinates.
178        let id = expect(&mut mock, "Runtime.callFunctionOn").await;
179        assert!(
180            mock.last_sent()["params"]["functionDeclaration"]
181                .as_str()
182                .unwrap()
183                .contains("scrollIntoView"),
184            "the scroll must be dispatched before the gate and DOM.getBoxModel",
185        );
186        mock.reply(id, json!({ "result": { "type": "undefined" } }))
187            .await;
188
189        // Step 2: actionability gate (VISIBLE_ONLY = only check_visible).
190        //
191        // The predicate is asserted by what it is NOT: its JS is owned by
192        // `query::actionability` and gets rewritten there, but "the gate is
193        // not the scroll" is the property this test needs.
194        let gate_js = serve_call_js(&mut mock, json!({ "value": true, "type": "boolean" })).await;
195        assert!(
196            gate_js.contains("getBoundingClientRect") && !gate_js.contains("scrollIntoView"),
197            "expected the visibility predicate after the scroll, got: {gate_js}",
198        );
199
200        // Step 3: bounding_box → DOM.getBoxModel.
201        let id = expect(&mut mock, "DOM.getBoxModel").await;
202        mock.reply(
203            id,
204            json!({
205                "model": {
206                    "content": [10.0, 20.0, 110.0, 20.0, 110.0, 70.0, 10.0, 70.0],
207                    "padding": [10.0, 20.0, 110.0, 20.0, 110.0, 70.0, 10.0, 70.0],
208                    "border":  [10.0, 20.0, 110.0, 20.0, 110.0, 70.0, 10.0, 70.0],
209                    "margin":  [10.0, 20.0, 110.0, 20.0, 110.0, 70.0, 10.0, 70.0],
210                    "width":  100,
211                    "height": 50
212                }
213            }),
214        )
215        .await;
216
217        // Step 4: the scroll offset that turns the viewport-relative box into
218        // the document-relative rect Chrome's clip actually wants.
219        let id = expect(&mut mock, "Page.getLayoutMetrics").await;
220        mock.reply(
221            id,
222            json!({
223                "cssVisualViewport": { "pageX": 5.0, "pageY": 3000.0, "clientWidth": 800,
224                                        "clientHeight": 600, "offsetX": 0, "offsetY": 0,
225                                        "scale": 1, "zoom": 1 },
226            }),
227        )
228        .await;
229
230        // Step 5: Page.captureScreenshot, clip offset into document space:
231        // x = 10 + 5, y = 20 + 3000.
232        let id = expect(&mut mock, "Page.captureScreenshot").await;
233        let sent = mock.last_sent();
234        assert_eq!(sent["params"]["format"], "png");
235        let clip = &sent["params"]["clip"];
236        assert_eq!(clip["x"], 15.0);
237        assert_eq!(
238            clip["y"], 3020.0,
239            "the clip must be in document coordinates, not viewport ones",
240        );
241        assert_eq!(clip["width"], 100.0);
242        assert_eq!(clip["height"], 50.0);
243        assert_eq!(clip["scale"], 1);
244        mock.reply(id, json!({ "data": "UE5HIQ==" })).await;
245
246        let bytes = fut.await.unwrap().unwrap();
247        assert_eq!(bytes, b"PNG!");
248        conn.shutdown();
249    }
