1//! bs4-like combinable predicate matchers. A `PredicateSet` compiles to a
2//! CSS selector (structural parts) + a JS boolean post-filter (regex/text).
3//! Pure — no CDP, fully unit-testable.
4
5use serde_json::json;
6
7#[derive(Debug, Clone, Default)]
8pub(crate) struct PredicateSet {
9    pub(crate) tag: Option<String>,
10    pub(crate) attrs: Vec<AttrPred>,
11    pub(crate) texts: Vec<TextPred>,
12}
13
14/// Trailing `bool` on every value-bearing variant below is `case_insensitive`
15/// — `true` requests the CSS `i` flag (`to_css_selector`) / a lower-cased
16/// compare (`to_js_filter`). `Has` (no value) and `Regex` (already
17/// case-insensitive via an inline `(?i)` pattern flag) don't carry one.
18#[derive(Debug, Clone)]
19pub(crate) enum AttrPred {
20    Exact(String, String, bool),
21    Contains(String, String, bool),
22    StartsWith(String, String, bool),
23    EndsWith(String, String, bool),
24    Has(String),
25    Regex(String, String), // (name, pattern) — JS post-filter, not CSS
26}
27
28#[derive(Debug, Clone)]
29pub(crate) enum TextPred {
30    Contains(String, bool),
31    /// Exact text match under XPath `normalize-space()` semantics — see
32    /// [`normalize_space`]. Both sides are folded by that rule.
33    ///
34    /// This is *not* fully equivalent to the `text_exact` selector kind,
35    /// which compiles to XPath `//*[normalize-space(.)=<needle>]`. The two
36    /// agree on whitespace folding and disagree on what they fold: XPath's
37    /// `normalize-space(.)` reads the node's string-value, the
38    /// concatenation of every descendant text node, while this filter reads
39    /// `el.innerText || el.textContent`, which is rendered text. For
40    /// `<div>a<span style="display:none">b</span></div>` XPath sees `"ab"`
41    /// and this sees `"a"`, so the two kinds match different elements.
42    /// Block boundaries differ too, and because `innerText` falls back to
43    /// `textContent` whenever it is empty, the divergence is
44    /// state-dependent rather than purely markup-dependent.
45    ///
46    /// Switching between the two selector kinds can therefore change the
47    /// match set. The source string is pinned by
48    /// `equals_filter_reads_inner_text_with_a_text_content_fallback`.
49    Equals(String, bool),
50    Matches(String), // regex pattern
51}
52
53/// JS tail that applies [`normalize_space`]'s rule to the page-side text —
54/// appended to `TXT` in [`PredicateSet::to_js_filter`].
55///
56/// The edge strip is `replace(/^ | $/g,"")`, not `trim()`. `trim()` removes
57/// the whole ECMAScript `WhiteSpace` + `LineTerminator` class, which
58/// includes U+00A0, U+FEFF, VT, FF and every Unicode space separator —
59/// none of which XPath `normalize-space()` touches. Since the collapse
60/// above has already reduced any leading or trailing run to exactly one
61/// space, stripping that single space is byte-for-byte what
62/// [`normalize_space`] does to the needle.
63const JS_NORMALIZE_SPACE: &str = r#".replace(/[ \t\r\n]+/g," ").replace(/^ | $/g,"")"#;
64
65/// XPath `normalize-space()` semantics: collapse every run of XML
66/// whitespace (space, tab, CR, LF) to a single space, then strip a
67/// leading/trailing one.
68///
69/// Applied to the *needle* at compile time and mirrored on the page-side
70/// text by [`JS_NORMALIZE_SPACE`], so both operands of a `text_equals`
71/// compare are folded by the same rule. Before this, the JS post-filter
72/// only `trim()`ed while the `text_exact` selector compiled to XPath
73/// `normalize-space(.)`, so `<b>Hello   world</b>` matched one and not the
74/// other.
75///
76/// Deliberately limited to the four XML whitespace characters rather than
77/// the broader JS `\s` / Rust `char::is_whitespace` class: `normalize-space()`
78/// leaves `&nbsp;` (U+00A0) and other Unicode spaces alone, and widening the
79/// class here would re-introduce on `&nbsp;` exactly the divergence this
80/// normalization exists to remove. The page side has to match that
81/// restraint at the edges as well as in the interior, which is why
82/// [`JS_NORMALIZE_SPACE`] cannot use `trim()`.
83pub(crate) fn normalize_space(s: &str) -> String {
84    s.split([' ', '\t', '\r', '\n'])
85        .filter(|part| !part.is_empty())
86        .collect::<Vec<_>>()
87        .join(" ")
88}
89
90/// Quote a value as a JSON string (`"v"`) — safe for both CSS attribute
91/// values and JS string literals (the selector is later JSON-embedded into
92/// the JS source, double-escaping correctly).
93fn q(v: &str) -> String {
94    json!(v).to_string()
95}
96
97/// CSS case-insensitivity flag suffix — `" i"` inside the bracket right
98/// before `]` when `ci` is set, empty string (byte-identical to the
99/// pre-case-insensitive output) otherwise. Valid for `=`/`*=`/`^=`/`$=`
100/// attribute selectors (not `[name]` presence, which has no value to flag).
101fn ci_flag(ci: bool) -> &'static str {
102    if ci { " i" } else { "" }
103}
104
105impl PredicateSet {
106    pub(crate) fn is_empty(&self) -> bool {
107        self.tag.is_none() && self.attrs.is_empty() && self.texts.is_empty()
108    }
109
110    /// Structural predicates → a CSS selector. `attr_regex` + text predicates
111    /// are post-filters and are NOT emitted here. Empty set → `"*"`.
112    ///
113    /// Attribute *values* are JSON-quoted via [`q`] so a quote/backslash
114    /// can't break the `[name="value"]` literal. Attribute *names* are
115    /// caller-supplied identifiers and are emitted verbatim — a malformed
116    /// name yields a malformed selector, which surfaces as a JS
117    /// `SyntaxError` from `querySelectorAll` (a `JsException`), never an
118    /// escape out of the JS string (the whole selector is JSON-embedded
119    /// before evaluation by the resolver).
120    pub(crate) fn to_css_selector(&self) -> String {
121        let mut s = self.tag.clone().unwrap_or_default();
122        for a in &self.attrs {
123            match a {
124                AttrPred::Exact(n, v, ci) => {
125                    s.push_str(&format!("[{n}={}{}]", q(v), ci_flag(*ci)));
126                }
127                AttrPred::Contains(n, v, ci) => {
128                    s.push_str(&format!("[{n}*={}{}]", q(v), ci_flag(*ci)));
129                }
130                AttrPred::StartsWith(n, v, ci) => {
131                    s.push_str(&format!("[{n}^={}{}]", q(v), ci_flag(*ci)));
132                }
133                AttrPred::EndsWith(n, v, ci) => {
134                    s.push_str(&format!("[{n}$={}{}]", q(v), ci_flag(*ci)));
135                }
136                AttrPred::Has(n) => s.push_str(&format!("[{n}]")),
137                AttrPred::Regex(..) => {}
138            }
139        }
140        if s.is_empty() { "*".to_string() } else { s }
141    }
142
143    /// Post-filter predicates (`attr_regex` + all text predicates) → a JS
144    /// boolean expression over a bound `el`. Returns `"true"` when there are
145    /// no post-filters (so the caller can always `.filter(el => <expr>)`).
146    ///
147    /// [`TextPred::Equals`] normalizes both operands per [`normalize_space`];
148    /// [`TextPred::Contains`] and [`TextPred::Matches`] run against the raw
149    /// text, since a substring or regex needle may legitimately carry its own
150    /// whitespace.
151    pub(crate) fn to_js_filter(&self) -> String {
152        const TXT: &str = r#"(el.innerText||el.textContent||"")"#;
153        let mut checks: Vec<String> = Vec::new();
154        for a in &self.attrs {
155            if let AttrPred::Regex(n, p) = a {
156                checks.push(format!(
157                    "new RegExp({}).test(el.getAttribute({})||\"\")",
158                    q(p),
159                    q(n)
160                ));
161            }
162        }
163        for t in &self.texts {
164            match t {
165                TextPred::Contains(s, false) => checks.push(format!("{TXT}.includes({})", q(s))),
166                TextPred::Contains(s, true) => checks.push(format!(
167                    "{TXT}.toLowerCase().includes({})",
168                    q(&s.to_lowercase())
169                )),
170                TextPred::Equals(s, false) => checks.push(format!(
171                    "{TXT}{JS_NORMALIZE_SPACE}==={}",
172                    q(&normalize_space(s))
173                )),
174                TextPred::Equals(s, true) => checks.push(format!(
175                    "{TXT}{JS_NORMALIZE_SPACE}.toLowerCase()==={}",
176                    q(&normalize_space(&s.to_lowercase()))
177                )),
178                TextPred::Matches(p) => checks.push(format!("new RegExp({}).test({TXT})", q(p))),
179            }
180        }
181        if checks.is_empty() {
182            "true".to_string()
183        } else {
184            checks.join("&&")
185        }
186    }
187}

373    #[test]
374    fn normalize_space_leaves_nbsp_alone_like_xpath() {
375        // XPath `normalize-space()` only folds space/tab/CR/LF. Folding
376        // U+00A0 here would make `text_equals` disagree with `text_exact`
377        // on `&nbsp;` markup — the exact bug this normalization fixes.
378        assert_eq!(normalize_space("a\u{a0}\u{a0}b"), "a\u{a0}\u{a0}b");
379    }

381    #[test]
382    fn normalize_space_keeps_nbsp_and_bom_at_the_edges() {
383        // The needle side has to preserve these, because XPath
384        // `normalize-space()` does. The page side must therefore preserve
385        // them too — see `js_normalize_space_edge_strip_is_not_trim`.
386        assert_eq!(normalize_space("\u{a0}OK"), "\u{a0}OK");
387        assert_eq!(normalize_space("OK\u{a0}"), "OK\u{a0}");
388        assert_eq!(normalize_space("\u{feff}a"), "\u{feff}a");
389        // A real XML-whitespace edge still goes, including when it sits
390        // outside the preserved character.
391        assert_eq!(normalize_space("  \u{a0}OK  "), "\u{a0}OK");
392    }

394    #[test]
395    fn js_normalize_space_edge_strip_is_not_trim() {
396        // `trim()` removes the whole ECMAScript WhiteSpace class — U+00A0,
397        // U+FEFF, VT, FF, every Unicode space separator — none of which
398        // XPath `normalize-space()` touches. Verified against V8: input
399        // codepoints `a0,4f,4b` come back `4f,4b` under `trim()` and
400        // unchanged under the edge strip. With `trim()` on the page side,
401        // `<b>&nbsp;OK</b>` matched `text_exact("\u{a0}OK")` but not
402        // `text_equals("\u{a0}OK")` — the same divergence this
403        // normalization exists to remove, displaced from interior runs to
404        // the edges.
405        let p = PredicateSet {
406            tag: None,
407            attrs: vec![],
408            texts: vec![TextPred::Equals("OK".into(), false)],
409        };
410        let f = p.to_js_filter();
411        assert!(
412            !f.contains(".trim()"),
413            "the page-side fold must not use trim(), got: {f}"
414        );
415        assert!(
416            f.contains(r#".replace(/^ | $/g,"")"#),
417            "the page-side fold must strip exactly one collapsed edge space, got: {f}"
418        );
419    }
