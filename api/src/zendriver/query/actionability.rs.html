1//! Playwright-style actionability predicates: visible / stable / enabled /
2//! receives_pointer. Each runs a small JS function on the element's remote
3//! handle via `Element::call_on_main` and returns a `bool`.
4//!
5//! `call_on_main` binds the element as the first positional argument, so
6//! every body opens `function(el, ...)` and any caller-supplied arguments
7//! follow it.
8//!
9//! The aggregate gate ([`wait_actionable`]) polls the four predicates in
10//! order and raises `NotActionable` on deadline; `Element::click_with`,
11//! `hover`, `focus`, `type_text`, and `screenshot` (in `element::actions`
12//! / `element::screenshot`) each pick the [`ActionabilityCheck`] preset
13//! matching what they need before dispatching.
14//!
15//! # A note on this module's unit tests
16//!
17//! The predicates are JavaScript, and `MockConnection` replays CDP frames
18//! without a JS engine, so the `probe_source_*` tests below assert on the
19//! *source text* each predicate puts on the wire. They pin that a clause is
20//! present and spelled the way a past defect proved it has to be; they do
21//! not execute it, and a mutation that rewrites a clause into something
22//! equally well-formed but wrong passes them.
23//!
24//! The live-Chrome tier reaches part of that, and does not gate a merge.
25//! `tests/find_visible_only.rs` (gated on the `integration-tests` feature)
26//! drives `visible_only(true)` against a real browser, so `check_visible`'s
27//! `display: none` path does execute there — but every test in that file is
28//! `#[ignore]`d, and the per-PR integration job runs plain
29//! `cargo nextest run -E 'kind(test)'` with no `--run-ignored`. So it runs
30//! only in the scheduled `nightly-ignored-tests` lane, which carries
31//! `continue-on-error: true`.
32//!
33//! Nothing else is covered behaviorally: the opacity, viewport and
34//! quirks-mode clauses are pinned only by the source-text assertions below,
35//! and `check_stable` / `check_enabled` / `check_receives_pointer` have no
36//! live fixture at all. Widening any of those clauses means writing the
37//! fixture that would catch the mistake; do not assume one is waiting.

47/// Set of actionability checks an action wants the element to satisfy
48/// before its CDP dispatch. Per-field booleans gate the corresponding
49/// `check_*` predicate in `wait_actionable`, and `hit_point` carries the
50/// coordinate the hit-test should probe. Four named presets cover the
51/// common combinations (`FULL`, `HOVER`, `VISIBLE_ONLY`, `TEXT_INPUT`);
52/// callers may also construct ad-hoc sets directly, and a caller with an
53/// explicit click position writes
54/// `ActionabilityCheck { hit_point: opts.position, ..FULL }`.
55#[derive(Clone, Copy, Debug, PartialEq)]
56pub(crate) struct ActionabilityCheck {
57    pub visible: bool,
58    pub stable: bool,
59    pub enabled: bool,
60    pub receives_pointer: bool,
61    /// Offset from the bbox top-left that [`check_receives_pointer`] should
62    /// hit-test, matching [`crate::ClickOptions::position`]. `None` probes
63    /// the bbox centre. Ignored unless `receives_pointer` is set.
64    pub hit_point: Option<(f64, f64)>,
65}
66
67impl ActionabilityCheck {
68    /// All four predicates — used by `click` and similar pointer-driven
69    /// actions where layout stability + an unobstructed hit point matter.
70    pub(crate) const FULL: Self = Self {
71        visible: true,
72        stable: true,
73        enabled: true,
74        receives_pointer: true,
75        hit_point: None,
76    };
77
78    /// `FULL` minus the enabled check — used by `hover` / `hover_fast`.
79    /// Hovering does not activate the element, so a disabled control still
80    /// accepts `mouseover` and gating on `enabled` would reject a hover the
81    /// browser performs happily.
82    pub(crate) const HOVER: Self = Self {
83        visible: true,
84        stable: true,
85        enabled: false,
86        receives_pointer: true,
87        hit_point: None,
88    };
89
90    /// Visibility only — used by `screenshot` (we just need pixels to
91    /// capture; we don't care if a sibling overlay covers part of the
92    /// element).
93    pub(crate) const VISIBLE_ONLY: Self = Self {
94        visible: true,
95        stable: false,
96        enabled: false,
97        receives_pointer: false,
98        hit_point: None,
99    };
100
101    /// Text-input combo — used by `type_text` / `focus` where the element
102    /// must be visible + enabled but doesn't need a hit-tested pointer
103    /// path (keystrokes route through the focused element, not the
104    /// cursor's position).
105    pub(crate) const TEXT_INPUT: Self = Self {
106        visible: true,
107        stable: false,
108        enabled: true,
109        receives_pointer: false,
110        hit_point: None,
111    };
112
113    /// How many `Runtime.callFunctionOn` probes one pass of
114    /// [`wait_actionable`] sends for this set — one per enabled predicate.
115    ///
116    /// Exists so a test fixture can serve exactly the gate's calls without
117    /// hard-coding a count per preset: a count that drifts from the set does
118    /// not fail cleanly, it eats the *next* unrelated call and the failure
119    /// surfaces at a later, unrelated `expect`.
120    #[cfg(test)]
121    pub(crate) const fn probe_count(self) -> usize {
122        self.visible as usize
123            + self.enabled as usize
124            + self.stable as usize
125            + self.receives_pointer as usize
126    }

129/// `true` iff the element is rendered **and on-screen**. An element is
130/// visible when all of the following hold:
131///
132/// - it is attached to the document (`isConnected`);
133/// - it passes the platform's own `Element.checkVisibility` test with
134///   `opacityProperty` + `visibilityProperty` + `contentVisibilityAuto`,
135///   which covers `display: none`, `visibility: hidden` / `collapse`,
136///   `content-visibility`, and `opacity: 0` on the element **or any
137///   ancestor**;
138/// - it has a positive bounding box;
139/// - its bounding box **intersects the viewport** — any overlap counts, so
140///   a partially-scrolled-in element still passes;
141/// - its *effective* opacity — the product of its own computed opacity and
142///   every ancestor's — is at least 1%.
143///
144/// The last two are the reason this isn't just a `checkVisibility` call.
145/// `checkVisibility` only rejects an *exact* `opacity: 0`, so the standard
146/// `opacity: 0.001` scraping honeypot passes it while being invisible to a
147/// human; and `checkVisibility` says nothing about where the element sits
148/// relative to the viewport.
149///
150/// Requires Chrome 121+, not 105+. `Element.checkVisibility` itself shipped
151/// in 105, but the three option names passed below (`opacityProperty`,
152/// `visibilityProperty`, `contentVisibilityAuto`) shipped in 121 — the first
153/// two as aliases for 105's `checkOpacity` / `checkVisibilityCSS`, the third
154/// as a new check (crbug feature 5070043440480256). Below 105 the call
155/// throws, surfacing as a `JsException`: a loud failure rather than a
156/// silently wrong answer. On 105–120 it does NOT throw — WebIDL drops
157/// dictionary members it doesn't recognize — so the call silently degrades
158/// to the default `display: none` test and stops covering
159/// `visibility: hidden`/`collapse` and `content-visibility: auto`. The
160/// effective-opacity loop below is independent of that and still holds.
161///
162/// Live callers: `Element::is_visible` and the `visible_only` filter in
163/// `FindBuilder`/`FindAllBuilder` (`query::mod`).
164pub(crate) async fn check_visible(el: &Element) -> Result<bool> {
165    let js = r#"
166        function(el) {
167            if (!el || !el.isConnected) return false;
168
169            // Platform primitive: handles `display: none`, `content-visibility`,
170            // `visibility: hidden`/`collapse` and `opacity: 0` on the element OR any
171            // ancestor — none of which the hand-rolled predecessor caught, since it
172            // only ever read the element's own computed style, and its layout probe
173            // reported a false negative for `position: fixed`.
174            if (!el.checkVisibility({
175                opacityProperty: true,
176                visibilityProperty: true,
177                contentVisibilityAuto: true
178            })) return false;
179
180            const rect = el.getBoundingClientRect();
181            if (rect.width <= 0 || rect.height <= 0) return false;
182
183            // On-screen test: the bbox must overlap the viewport by at least a
184            // sliver. `getBoundingClientRect` is already viewport-relative, so
185            // the box it must be compared against is the LAYOUT viewport. Two
186            // things about reading that box are load-bearing:
187            //
188            // 1. It must never come from `window.inner*`. Every profile except
189            //    `StealthProfile::off()` — including `native()`, which
190            //    `Browser::builder()` installs by default — runs
191            //    `zendriver-stealth`'s `patches/screen.js`, which rewrites
192            //    `window.inner*` to the persona's screen size minus a chrome
193            //    inset, while `scroll_into_view` moves the element with
194            //    Chrome's real viewport. The inset is whatever the caller's
195            //    `ScreenSpec` measured, falling back to a derived 86px when no
196            //    capture supplied one, so its size is not knowable from here —
197            //    which is the point: no constant can compensate for it.
198            //    Measured under the default profile (the 86px fallback):
199            //    `innerHeight` 994 against a real 1080. Comparing a
200            //    real-geometry scroll against that fabricated viewport leaves
201            //    every element landing in the inset band permanently "not
202            //    visible", which broke `focus` — and with it `type_text` /
203            //    `press` / `type_keys` — on any below-the-fold field.
204            //
205            // 2. WHICH element reports the layout viewport depends on the
206            //    rendering mode, and reading the wrong one fails silently.
207            //    Standards mode: `documentElement`. Quirks (`BackCompat`):
208            //    `documentElement.client*` reports the document's CONTENT box
209            //    and `body.client*` reports the viewport. Measured on a 6000px
210            //    doctype-less page: `documentElement.clientHeight` 6024,
211            //    `body.clientHeight` 1080. Reading `documentElement`
212            //    unconditionally therefore compares the bbox against the whole
213            //    document on any page without a doctype — a no-op that fails
214            //    open, on exactly the sloppy legacy pages most likely to be
215            //    carrying a honeypot.
216            const box = document.compatMode === 'BackCompat'
217                ? document.body
218                : document.documentElement;
219            // The `BackCompat` arm can be null: a document whose `<body>` was
220            // removed still renders elements appended to `documentElement`, and
221            // still answers `isConnected` for them (reproduced against Chrome).
222            // The fallback is `visualViewport` rather than `window.inner*`
223            // because it is the other unspoofed source, and it reads the real
224            // 1080 in both modes. It is the VISUAL viewport, which diverges
225            // from the layout viewport under pinch-zoom or an on-screen
226            // keyboard — neither reachable through this crate's emulation
227            // surface, and neither triggerable by page script.
228            const vw = box ? box.clientWidth : window.visualViewport.width;
229            const vh = box ? box.clientHeight : window.visualViewport.height;
230            if (rect.right <= 0 || rect.bottom <= 0 || rect.left >= vw || rect.top >= vh) {
231                return false;
232            }
233
234            // Effective opacity: `checkVisibility` rejects only an exact `opacity: 0`,
235            // so a near-zero honeypot (`opacity: 0.001`) sails through it. Multiply the
236            // ancestor chain the way the compositor does and reject anything below 1%.
237            // The threshold is deliberately not `> 0`: exact-zero is the case honeypots
238            // avoid precisely because it is the one everyone already checks. 1% sits well
239            // below anything a real UI renders at rest and well above the honeypot range.
240            // Done last — `getComputedStyle` per ancestor is the priciest step here.
241            // Opacity is in [0, 1], so the running product only ever decreases and
242            // cannot climb back over the threshold: once it is under, stop walking.
243            let effective = 1;
244            for (let node = el; node; node = node.parentElement) {
245                const own = parseFloat(getComputedStyle(node).opacity);
246                if (Number.isFinite(own)) effective *= own;
247                if (effective < 0.01) return false;
248            }
249
250            return true;
251        }
252    "#;
253    let res = el.call_on_main(js, json!([])).await?;
254    Ok(res.get("value").and_then(|v| v.as_bool()).unwrap_or(false))
255}

257/// `true` iff the element's bounding box is unchanged across two
258/// `requestAnimationFrame` ticks (within 0.5px on each of x/y/w/h). This
259/// catches mid-transition layout shifts that would race a synthesized
260/// click.
261pub(crate) async fn check_stable(el: &Element) -> Result<bool> {
262    let js = r#"
263        function(el) {
264            return new Promise(resolve => {
265                if (!el || !el.isConnected) { resolve(false); return; }
266                const first = el.getBoundingClientRect();
267                requestAnimationFrame(() => {
268                    requestAnimationFrame(() => {
269                        const second = el.getBoundingClientRect();
270                        const stable =
271                            Math.abs(first.x - second.x) < 0.5 &&
272                            Math.abs(first.y - second.y) < 0.5 &&
273                            Math.abs(first.width - second.width) < 0.5 &&
274                            Math.abs(first.height - second.height) < 0.5;
275                        resolve(stable);
276                    });
277                });
278            });
279        }
280    "#;
281    let res = el.call_on_main(js, json!([])).await?;
282    Ok(res.get("value").and_then(|v| v.as_bool()).unwrap_or(false))
283}

285/// `true` iff the element is not disabled: native `el.disabled` is
286/// false-ish AND `aria-disabled` is not `'true'`. Non-form elements
287/// (which have no `disabled` property) are considered enabled.
288pub(crate) async fn check_enabled(el: &Element) -> Result<bool> {
289    let js = r#"
290        function(el) {
291            if (!el) return false;
292            // `disabled === false` for form controls; `undefined` for non-form elements
293            // (which we treat as enabled).
294            if (el.disabled === true) return false;
295            const ariaDisabled = el.getAttribute && el.getAttribute('aria-disabled');
296            if (ariaDisabled === 'true') return false;
297            return true;
298        }
299    "#;
300    let res = el.call_on_main(js, json!([])).await?;
301    Ok(res.get("value").and_then(|v| v.as_bool()).unwrap_or(false))
302}

304/// `true` iff a synthesized click at the point that will actually be clicked
305/// would land on the element (or one of its descendants). Walks the ancestor
306/// chain of `document.elementFromPoint(cx, cy)`; if our element appears in that
307/// chain, pointer events reach it. Returns `false` when a sibling overlay
308/// covers the hit point.
309///
310/// `position` is the caller's offset from the bbox top-left, matching
311/// [`crate::ClickOptions::position`]; `None` hit-tests the centre. Probing a
312/// point the dispatch will not use lets the gate pass while the real click
313/// lands on an overlay.
314pub(crate) async fn check_receives_pointer(
315    el: &Element,
316    position: Option<(f64, f64)>,
317) -> Result<bool> {
318    let js = r#"
319        function(el, dx, dy) {
320            if (!el || !el.isConnected) return false;
321            const rect = el.getBoundingClientRect();
322            if (rect.width <= 0 || rect.height <= 0) return false;
323            // A caller with no explicit position passes no arguments at all,
324            // so dx/dy arrive as `undefined` — `Number.isFinite` covers that
325            // and any non-numeric junk in one test, falling back to the centre.
326            const cx = Number.isFinite(dx) ? rect.left + dx : rect.left + rect.width / 2;
327            const cy = Number.isFinite(dy) ? rect.top + dy : rect.top + rect.height / 2;
328            let hit = document.elementFromPoint(cx, cy);
329            while (hit) {
330                if (hit === el) return true;
331                hit = hit.parentElement;
332            }
333            return false;
334        }
335    "#;
336    // `Runtime.callFunctionOn` takes an array of CallArgument *objects*, so
337    // each coordinate has to be wrapped in `{ "value": … }`; passing the bare
338    // numbers made Chrome reject the whole call with "Invalid parameters"
339    // ("Failed to deserialize params.arguments"), which failed every gated
340    // `click` / `hover` / `tap` against a real browser. The mock tests could
341    // not catch it — `MockConnection` doesn't validate CDP parameter shapes.
342    //
343    // `call_on_main` prepends the element handle, so these land at
344    // arguments[1] / arguments[2] — which is where the leading `el`
345    // parameter in the declaration above leaves `dx` / `dy`.
346    let args = match position {
347        Some((dx, dy)) => json!([{ "value": dx }, { "value": dy }]),
348        None => json!([]),
349    };
350    let res = el.call_on_main(js, args).await?;
351    Ok(res.get("value").and_then(|v| v.as_bool()).unwrap_or(false))
352}

354/// Poll each predicate in `require` at 50 ms intervals until all enabled
355/// checks pass, or `timeout` elapses. On deadline, returns
356/// [`ZendriverError::NotActionable`] with the first-failing check's
357/// human-readable reason ("not visible", "not enabled", "not stable (still
358/// animating)", or "occluded by overlay").
359///
360/// Predicates are evaluated in fixed order: visible → enabled → stable →
361/// receives_pointer — the two cheap reads first, so the expensive
362/// two-frame stability wait and the hit-test never run while the element is
363/// still hidden or disabled. The *set* of checks is Playwright's; this
364/// ordering is not, and does not claim to be — Playwright's actionability
365/// docs list visible → stable → receives events → enabled.
366///
367/// `require.hit_point` is the offset the caller will click at, forwarded to
368/// the hit-test so the gate probes the point the dispatch actually uses.
369/// `None` probes the bbox centre. It is re-read from the live bbox on every
370/// poll, so an element that moves mid-wait is hit-tested where it currently
371/// is.
372pub(crate) async fn wait_actionable(
373    el: &Element,
374    require: ActionabilityCheck,
375    timeout: Duration,
376) -> Result<()> {
377    let deadline = Instant::now() + timeout;
378    let poll_interval = Duration::from_millis(50);
379    loop {
380        let mut failed_reason: Option<&'static str> = None;
381        if require.visible && !check_visible(el).await? {
382            failed_reason = Some("not visible");
383        } else if require.enabled && !check_enabled(el).await? {
384            failed_reason = Some("not enabled");
385        } else if require.stable && !check_stable(el).await? {
386            failed_reason = Some("not stable (still animating)");
387        } else if require.receives_pointer && !check_receives_pointer(el, require.hit_point).await?
388        {
389            failed_reason = Some("occluded by overlay");
390        }
391        match failed_reason {
392            None => return Ok(()),
393            Some(reason) => {
394                if Instant::now() >= deadline {
395                    return Err(ZendriverError::NotActionable(timeout, reason.to_owned()));
396                }
397                tokio::time::sleep(poll_interval).await;
398            }
399        }
400    }
401}

537    async fn check_visible_resolves_the_probe_result_it_was_handed() {
538        let (_, hidden) = probe_check_visible(false).await;
539        assert!(!hidden, "a `false` probe result must resolve to hidden");
540        let (_, shown) = probe_check_visible(true).await;
541        assert!(shown, "a `true` probe result must resolve to visible");
542    }

544    /// `check_receives_pointer` must wrap its coordinates as CallArgument
545    /// *objects* (`{ "value": … }`).
546    ///
547    /// Regression pin for a live-only bug: the bare `[dx, dy]` this used to
548    /// send made Chrome reject the whole call with `-32602 Invalid parameters`
549    /// ("Failed to deserialize params.arguments"), so every gated `click` /
550    /// `hover` / `tap` failed against a real browser while the mock suite
551    /// stayed green — `MockConnection` replays frames without validating CDP
552    /// parameter shapes. Hence an explicit assertion on the shape.
553    #[tokio::test]
554    async fn check_receives_pointer_wraps_its_coordinates_as_call_arguments() {
555        // `call_on_main` prepends the element handle, so the coordinates are
556        // the second and third arguments — each one an object, never a bare
557        // number.
558        assert_eq!(
559            probe_pointer_args(Some((3.0, 4.0))).await,
560            json!([{ "objectId": "R1" }, { "value": 3.0 }, { "value": 4.0 }]),
561            "an explicit click position must travel as two CallArgument objects",
562        );
563
564        // No position → no coordinates at all, leaving dx/dy `undefined` in
565        // the JS, which is what its `Number.isFinite` centre-fallback reads.
566        assert_eq!(
567            probe_pointer_args(None).await,
568            json!([{ "objectId": "R1" }]),
569            "a centre hit-test must send no coordinates rather than bare nulls",
570        );
571    }

573    /// Drive one `check_receives_pointer` probe against a fresh mock and hand
574    /// back the `arguments` array it put on the wire.
575    async fn probe_pointer_args(at: Option<(f64, f64)>) -> Value {
576        let (mut mock, conn) = MockConnection::pair();
577        let sess = SessionHandle::new(conn.clone(), "S1");
578        let tab = Tab::new_for_test(sess);
579        let el = Element::from_jsret(tab, 99, "R1".to_string());
580
581        // Spawn before serving: nothing goes out until the probe is running.
582        let fut = tokio::spawn(async move { check_receives_pointer(&el, at).await });
583
584        let params = serve_call(&mut mock, json!({ "value": true, "type": "boolean" })).await;
585
586        assert!(fut.await.unwrap().unwrap());
587        conn.shutdown();
588        params["arguments"].clone()
589    }

591    /// Every predicate binds the element as its first positional argument,
592    /// and `check_receives_pointer` takes its coordinates in `(dx, dy)` order.
593    ///
594    /// `call_on_main` prepends the element handle to `arguments`, so a body
595    /// that forgot its leading `el` parameter would silently read the handle
596    /// as its first *caller* argument — and, for the three no-argument
597    /// predicates, see `el` as `undefined` and answer `false` for everything.
598    /// This drives all four past that in one go.
599    ///
600    /// The whole parameter list is asserted, not just the leading name: a
601    /// `starts_with("function(el")` check accepts `function(elephant, dy, dx)`
602    /// — right prefix, transposed coordinates, every hit-test off by the
603    /// difference between the two offsets.
604    #[tokio::test]
605    async fn every_predicate_binds_the_element_as_its_first_argument() {
606        let (mut mock, conn) = MockConnection::pair();
607        let sess = SessionHandle::new(conn.clone(), "S1");
608        let tab = Tab::new_for_test(sess);
609        let el = Element::from_jsret(tab.clone(), 99, "R1".to_string());
610
611        let fut = tokio::spawn({
612            let e = el.clone();
613            async move {
614                (
615                    check_visible(&e).await.unwrap(),
616                    check_enabled(&e).await.unwrap(),
617                    check_stable(&e).await.unwrap(),
618                    check_receives_pointer(&e, None).await.unwrap(),
619                )
620            }
621        });
622
623        let mut calls = Vec::new();
624        for _ in 0..4 {
625            calls.push(serve_call(&mut mock, json!({ "value": true, "type": "boolean" })).await);
626        }
627
628        assert_eq!(fut.await.unwrap(), (true, true, true, true));
629
630        let sources: Vec<&str> = calls
631            .iter()
632            .map(|c| c["functionDeclaration"].as_str().unwrap())
633            .collect();
634
635        // Pin which source belongs to which predicate, so a future reorder of
636        // the gate can't quietly re-point one of these assertions.
637        assert!(sources[0].contains("checkVisibility"), "{}", sources[0]);
638        assert!(sources[1].contains("aria-disabled"), "{}", sources[1]);
639        assert!(
640            sources[2].contains("requestAnimationFrame"),
641            "{}",
642            sources[2]
643        );
644        assert!(sources[3].contains("elementFromPoint"), "{}", sources[3]);
645
646        // The three no-argument predicates bind `el` and nothing else; the
647        // hit-test binds the two offsets after it, dx before dy.
648        let expected = [vec!["el"], vec!["el"], vec!["el"], vec!["el", "dx", "dy"]];
649        for ((call, js), want) in calls.iter().zip(&sources).zip(expected) {
650            assert_eq!(js_params(js), want, "{js}");
651            assert_eq!(
652                call["arguments"][0]["objectId"], "R1",
653                "the element must be the first argument the body reads",
654            );
655        }
656
657        conn.shutdown();
658    }
