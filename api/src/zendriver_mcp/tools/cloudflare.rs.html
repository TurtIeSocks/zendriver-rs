1//! Cloudflare Turnstile bypass tool — `browser_solve_turnstile`. Gated
2//! behind the `cloudflare` feature.
3//!
4//! ## Outcome mapping
5//!
6//! The lib's [`CloudflareBypass::wait_for_clearance`] models its terminal
7//! state as a `Result<ClearanceOutcome, CloudflareError>` where:
8//!
9//! - `Ok(ClearanceOutcome::TokenAcquired(t))` — a configured token input
10//!   (`cf-turnstile-response` by default) picked up a token.
11//! - `Ok(ClearanceOutcome::ChallengeGone)` — the challenge stopped being
12//!   actionable without a token (e.g. clearance cookie shortcut).
13//! - `Ok(ClearanceOutcome::TimedOut { saw_challenge })` — the per-call
14//!   deadline elapsed (whether or not a challenge was ever seen).
15//!
16//! Agents typically want all three lumped into a single discriminated
17//! union of *expected* outcomes — a timeout in turnstile flow is a normal
18//! "didn't finish, try again or give up" signal, not a server error. So
19//! the MCP layer collapses `TimedOut` into a third [`Outcome::Timeout`]
20//! variant, and keeps every `CloudflareError` (network failure, JS error)
21//! as a real MCP error.
22//!
23//! [`CloudflareBypass::wait_for_clearance`]: zendriver_cloudflare::CloudflareBypass::wait_for_clearance
24

42/// Input for `browser_solve_turnstile`.
43#[derive(Debug, Deserialize, JsonSchema)]
44#[serde(deny_unknown_fields)]
45pub struct SolveInput {
46    /// Maximum total wait for a terminal outcome, in milliseconds. Default
47    /// 30_000 (30s) — the lib's documented sane default for a real Turnstile
48    /// flow. Lower values can be used in tests or when the agent wants to
49    /// fail fast and fall back to another strategy.
50    #[serde(default = "default_timeout")]
51    pub timeout_ms: u64,
52    /// Override the bypass driver's internal poll cadence, in milliseconds.
53    /// Defaults to the lib's own default (500ms). Lowering speeds up
54    /// detection at the cost of more CDP `Runtime.evaluate` round-trips;
55    /// raising is a safe knob for slow / sandboxed environments.
56    #[serde(default, skip_serializing_if = "Option::is_none")]
57    pub poll_interval_ms: Option<u64>,
58    /// Override the page markers that identify the Turnstile widget. Omit to
59    /// use the lib's defaults, which track Cloudflare's current markup.
60    /// Cloudflare owns this markup and changes it without notice, so an
61    /// agent that finds the defaults no longer matching a page can point the
62    /// solver at the new markers here instead of waiting on a release.
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub selectors: Option<SelectorsInput>,
65    /// Override whether, how often and where the widget is clicked. Omit for
66    /// the lib's defaults.
67    #[serde(default, skip_serializing_if = "Option::is_none")]
68    pub click_policy: Option<ClickPolicyInput>,
69}
70
71/// Page markers identifying the Turnstile widget — the request-side mirror
72/// of the lib's `TurnstileSelectors`.
73///
74/// Every field is optional and falls back to the lib default *individually*,
75/// so overriding one marker does not silently blank the others.
76#[derive(Debug, Default, Deserialize, JsonSchema)]
77#[serde(deny_unknown_fields)]
78pub struct SelectorsInput {
79    /// Substring an `<iframe>`'s `src` must contain for it to be the
80    /// challenge widget. Defaults to `"challenges.cloudflare.com"`. An empty
81    /// string disables the iframe signal rather than matching every iframe.
82    #[serde(default, skip_serializing_if = "Option::is_none")]
83    pub iframe_src_contains: Option<String>,
84    /// CSS selector for the challenge container, used as a presence signal.
85    /// Defaults to `".cf-turnstile, .turnstile, [data-sitekey]"`.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub container: Option<String>,
88    /// CSS selectors for the hidden input carrying the clearance token, in
89    /// preference order. Defaults to
90    /// `["[name=\"cf-turnstile-response\"]", "[name=\"cf_challenge_response\"]"]`.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub token_inputs: Option<Vec<String>>,
93}
94
95impl SelectorsInput {
96    /// Fold into the lib type, field by field, over the lib's defaults.
97    fn into_lib(self) -> TurnstileSelectors {
98        let d = TurnstileSelectors::default();
99        TurnstileSelectors {
100            iframe_src_contains: self.iframe_src_contains.unwrap_or(d.iframe_src_contains),
101            container: self.container.unwrap_or(d.container),
102            token_inputs: self.token_inputs.unwrap_or(d.token_inputs),
103        }
104    }
105}

107/// Click behaviour — the request-side mirror of the lib's `ClickPolicy`.
108/// Same per-field fallback rule as [`SelectorsInput`].
109#[derive(Debug, Default, Deserialize, JsonSchema)]
110#[serde(deny_unknown_fields)]
111pub struct ClickPolicyInput {
112    /// How many clicks one solve may spend on the widget. `0` watches
113    /// without ever clicking, for pages whose own script drives the widget.
114    /// Defaults to `3`.
115    #[serde(default, skip_serializing_if = "Option::is_none")]
116    pub max_attempts: Option<u32>,
117    /// Poll ticks to wait after a click before spending another attempt.
118    /// Defaults to `4`.
119    #[serde(default, skip_serializing_if = "Option::is_none")]
120    pub retry_ticks: Option<u32>,
121    /// Horizontal click point inside the widget's box, as a fraction of its
122    /// width. Defaults to `0.15`.
123    #[serde(default, skip_serializing_if = "Option::is_none")]
124    pub x_fraction: Option<f64>,
125    /// Vertical click point inside the widget's box, as a fraction of its
126    /// height. Defaults to `0.5`.
127    #[serde(default, skip_serializing_if = "Option::is_none")]
128    pub y_fraction: Option<f64>,
129}
130
131impl ClickPolicyInput {
132    /// Fold into the lib type, field by field, over the lib's defaults.
133    fn into_lib(self) -> ClickPolicy {
134        let d = ClickPolicy::default();
135        ClickPolicy {
136            max_attempts: self.max_attempts.unwrap_or(d.max_attempts),
137            retry_ticks: self.retry_ticks.unwrap_or(d.retry_ticks),
138            x_fraction: self.x_fraction.unwrap_or(d.x_fraction),
139            y_fraction: self.y_fraction.unwrap_or(d.y_fraction),
140        }
141    }
142}

148/// Apply a request's overrides to a bypass driver, leaving anything the
149/// request omitted at the lib's own default.
150///
151/// Split out of [`solve_turnstile`] purely so it can be tested: the tool
152/// needs a live tab to get a driver at all, this needs only a session, and a
153/// `MockConnection` supplies one. Without the split, the request fields
154/// converting correctly and the request fields ever being *read* were two
155/// different claims and only the first had a test.
156fn configure<'a>(mut bypass: CloudflareBypass<'a>, input: SolveInput) -> CloudflareBypass<'a> {
157    if let Some(p) = input.poll_interval_ms {
158        bypass = bypass.poll_interval(Duration::from_millis(p));
159    }
160    if let Some(selectors) = input.selectors {
161        bypass = bypass.selectors(selectors.into_lib());
162    }
163    if let Some(policy) = input.click_policy {
164        bypass = bypass.click_policy(policy.into_lib());
165    }
166    bypass
167}

169/// Terminal outcome of a turnstile bypass attempt.
170///
171/// `Solved` and `ChallengeGone` mirror the lib's `ClearanceOutcome`
172/// variants. `Timeout` mirrors the lib's `ClearanceOutcome::TimedOut` — a
173/// deadline in a turnstile flow is a normal outcome, not an error —
174/// surfacing it on the success channel so agents can branch on `outcome`
175/// without try/catch around a timeout.
176#[derive(Debug, Serialize, JsonSchema, PartialEq, Eq)]
177#[serde(rename_all = "snake_case")]
178pub enum Outcome {
179    /// Turnstile produced a token — the value held by the first matching
180    /// `selectors.token_inputs` entry, `cf-turnstile-response` by default.
181    /// The token is available in [`SolveOutput::token`].
182    Solved,
183    /// The challenge stopped being actionable without yielding a token
184    /// (e.g. a clearance cookie shortcut). `token` will be `None`.
185    ///
186    /// Not proof the gate was passed: this also fires when the widget is
187    /// still mounted but hidden or zero-size, so verify the page before
188    /// treating what you read next as post-challenge content.
189    ChallengeGone,
190    /// `timeout_ms` elapsed without reaching either success state. `token`
191    /// will be `None`. Not a hard error — agents can retry or give up.
192    Timeout,
193}
194
195/// Output of `browser_solve_turnstile`.
196#[derive(Debug, Serialize, JsonSchema)]
197pub struct SolveOutput {
198    /// Which terminal state the bypass reached.
199    pub outcome: Outcome,
200    /// Turnstile response token. Populated only when `outcome == Solved`;
201    /// `None` for `ChallengeGone` and `Timeout`.
202    #[serde(skip_serializing_if = "Option::is_none")]
203    pub token: Option<String>,
204}
205
206/// Drive the Turnstile clearance flow on the current tab, returning the
207/// terminal outcome (`Solved`, `ChallengeGone`, or `Timeout`) within
208/// `timeout_ms`.
209///
210/// See module-level docs for outcome semantics.
211pub async fn solve_turnstile(
212    state: Arc<Mutex<SessionState>>,
213    input: SolveInput,
214) -> Result<SolveOutput, ErrorData> {
215    let s = state.lock().await;
216    let tab = current_tab(&s).await?;
217    // Build the bypass driver. `tab.cloudflare()` borrows `tab` for the
218    // bypass's `'_` lifetime, but `wait_for_clearance` consumes `self`, so
219    // the bypass lives only for the single await we're about to do.
220    let timeout = Duration::from_millis(input.timeout_ms);
221    match configure(tab.cloudflare(), input)
222        .wait_for_clearance(timeout)
223        .await
224    {
225        Ok(ClearanceOutcome::TokenAcquired(t)) => Ok(SolveOutput {
226            outcome: Outcome::Solved,
227            token: Some(t),
228        }),
229        Ok(ClearanceOutcome::ChallengeGone) => Ok(SolveOutput {
230            outcome: Outcome::ChallengeGone,
231            token: None,
232        }),
233        // TimedOut is a lib-side success terminal; collapse it into the
234        // success-channel `Outcome::Timeout` — see module docs. The old
235        // `NoChallenge` error used to fall into the `Err` arm below; it is
236        // now folded into this outcome (with `saw_challenge: false`).
237        Ok(ClearanceOutcome::TimedOut { .. }) => Ok(SolveOutput {
238            outcome: Outcome::Timeout,
239            token: None,
240        }),
241        // Everything else (CDP call failed, JS error) is a real error the
242        // agent should surface. Route through the lib's
243        // `From<CloudflareError> for ZendriverError` so the existing
244        // `map_error` knows how to format it.
245        Err(other) => Err(map_error(McpServerError::from(ZendriverError::from(other)))),
246    }
247}

279    /// The tool's promise is that an agent can follow Cloudflare's markup
280    /// when it moves. That needs two things to be true, and only one of them
281    /// had a test: that the request types fold onto the lib types correctly
282    /// (below), and that the fold is ever actually *called*. Deleting both
283    /// wiring branches from the tool left this module green.
284    ///
285    /// So drive a real bypass over a mock CDP connection and read the wire.
286    /// The markers must reach the evaluator the driver dispatches, and the
287    /// click must land where the request's fractions put it — 15% / 50% of
288    /// the same box would be (112, 220), so neither coordinate can be
289    /// produced by a driver that quietly kept its defaults.
290    #[tokio::test]
291    async fn request_overrides_reach_the_driver() {
292        use serde_json::{Value, json};
293        use zendriver::{CloudflareBypass, SessionHandle};
294        use zendriver_transport::testing::MockConnection;
295
296        /// Widget box: 80×40 at (100, 200).
297        fn widget() -> Value {
298            json!({ "x": 100.0, "y": 200.0, "width": 80.0, "height": 40.0 })
299        }
300        fn eval_reply(value: Value) -> Value {
301            json!({ "result": { "type": "object", "value": value } })
302        }
303        async fn next_eval(mock: &mut MockConnection) -> (u64, String) {
304            let id =
305                tokio::time::timeout(Duration::from_secs(5), mock.expect_cmd("Runtime.evaluate"))
306                    .await
307                    .expect("driver never dispatched an evaluator");
308            let expr = mock.last_sent()["params"]["expression"]
309                .as_str()
310                .unwrap()
311                .to_string();
312            (id, expr)
313        }
314
315        let (mut mock, conn) = MockConnection::pair();
316        let sess = SessionHandle::new(conn.clone(), "S1");
317
318        let input = SolveInput {
319            timeout_ms: 5_000,
320            poll_interval_ms: Some(1),
321            selectors: Some(SelectorsInput {
322                iframe_src_contains: Some("gate.alien.invalid".into()),
323                ..SelectorsInput::default()
324            }),
325            click_policy: Some(ClickPolicyInput {
326                x_fraction: Some(0.5),
327                y_fraction: Some(0.25),
328                ..ClickPolicyInput::default()
329            }),
330        };
331
332        // Borrowed, not spawned: the bypass borrows the session, so its
333        // future is not `'static`. Driver and mock server share a task.
334        let driver = async {
335            configure(CloudflareBypass::new(&sess), input)
336                .wait_for_clearance(Duration::from_millis(5_000))
337                .await
338        };
339
340        let serving = async {
341            let (id, expr) = next_eval(&mut mock).await;
342            assert!(
343                expr.contains("gate.alien.invalid"),
344                "the request's marker never reached the page"
345            );
346            assert!(
347                !expr.contains("challenges.cloudflare.com"),
348                "the driver kept its default marker despite the request"
349            );
350            mock.reply(
351                id,
352                eval_reply(json!({ "token": null, "bbox": widget(), "hasMarkers": true })),
353            )
354            .await;
355
356            // Scroll-and-re-measure, then the three events of one click.
357            let (id, expr) = next_eval(&mut mock).await;
358            assert!(expr.contains("scrollIntoView"));
359            mock.reply(id, eval_reply(widget())).await;
360
361            let mut point = (f64::NAN, f64::NAN);
362            for _ in 0..3 {
363                let id = tokio::time::timeout(
364                    Duration::from_secs(5),
365                    mock.expect_cmd("Input.dispatchMouseEvent"),
366                )
367                .await
368                .expect("driver never clicked");
369                let sent = mock.last_sent().clone();
370                point = (
371                    sent["params"]["x"].as_f64().unwrap(),
372                    sent["params"]["y"].as_f64().unwrap(),
373                );
374                mock.reply(id, json!({})).await;
375            }
376
377            let (id, _) = next_eval(&mut mock).await;
378            mock.reply(
379                id,
380                eval_reply(json!({ "token": "T", "bbox": null, "hasMarkers": true })),
381            )
382            .await;
383            point
384        };
385
386        let (outcome, point) = tokio::join!(driver, serving);
387        assert_eq!(
388            point,
389            (140.0, 210.0),
390            "the click ignored the request's fractions"
391        );
392        assert!(matches!(
393            outcome.unwrap(),
394            zendriver::ClearanceOutcome::TokenAcquired(_)
395        ));
396        conn.shutdown();
397    }

399    /// An omitted field means "keep the lib default", not "zero". Folding
400    /// with `unwrap_or_default()` instead would turn an agent overriding one
401    /// marker into an agent that silently blanked the other two, and an
402    /// agent tuning `retry_ticks` into one that disabled clicking outright.
403    #[test]
404    fn omitted_fields_fall_back_to_the_lib_defaults_one_by_one() {
405        let selectors = SelectorsInput {
406            iframe_src_contains: Some("gate.example.invalid".into()),
407            ..SelectorsInput::default()
408        }
409        .into_lib();
410        let d = TurnstileSelectors::default();
411        assert_eq!(selectors.iframe_src_contains, "gate.example.invalid");
412        assert_eq!(selectors.container, d.container);
413        assert_eq!(selectors.token_inputs, d.token_inputs);
414
415        let policy = ClickPolicyInput {
416            retry_ticks: Some(1),
417            ..ClickPolicyInput::default()
418        }
419        .into_lib();
420        let d = ClickPolicy::default();
421        assert_eq!(policy.retry_ticks, 1);
422        assert_eq!(policy.max_attempts, d.max_attempts);
423        assert_eq!(policy.x_fraction, d.x_fraction);
424        assert_eq!(policy.y_fraction, d.y_fraction);
425    }
