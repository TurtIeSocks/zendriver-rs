1//! StealthObserver: applies a [`StealthProfile`] to each new attached target.
2//!
3//! Installed in the [`zendriver_transport`] actor's observer chain. On
4//! `Target.attachedToTarget`, the actor pauses the new target, walks every
5//! observer serially, then releases the debugger via
6//! `Runtime.runIfWaitingForDebugger`. The observer's job is to push every
7//! UA/screen/timezone/locale override and (for spoofed) the bootstrap script
8//! _before_ the debugger releases, so the first script the page runs sees the
9//! patched globals.
10
11use serde_json::json;
12use zendriver_transport::{CallError, ObserverError, PausedSession, TargetObserver};
13
14use crate::patches::{bootstrap_script, bootstrap_script_native_webgl, geometry_bootstrap_with};
15use crate::persona::GeoPos;
16use crate::persona::specs::{ScreenSpec, UaMetadata};
17use crate::{Fingerprint, Persona, ProfileKind, StealthProfile};
18
19/// Observer that applies a [`StealthProfile`] + [`Fingerprint`] to every page
20/// target. Workers and iframes are skipped — workers have no DOM and iframes
21/// inherit patches from the parent target in flat session mode.
22#[derive(Debug)]
23pub struct StealthObserver {
24    profile: StealthProfile,
25    fingerprint: Fingerprint,
26    /// Pre-rendered bootstrap source. Empty for `Off`/`Native` — we never send
27    /// `Page.addScriptToEvaluateOnNewDocument` in those modes, so there is no
28    /// need to pay the patches-bundle cost for them.
29    bootstrap: String,
30    /// Mock geolocation coordinates from the resolved [`Persona`], sent via
31    /// `Emulation.setGeolocationOverride`. Unlike `timezone`/`locale` (carried
32    /// on [`Fingerprint`]), geolocation has no `Fingerprint` counterpart, so
33    /// it's captured here straight off the persona at construction time.
34    geolocation: Option<GeoPos>,
35    /// Custom UA-CH from the resolved [`Persona`]'s `ua.ua_metadata`. When
36    /// set, [`UaMetadata::resolve`] fills any unset sub-field from
37    /// `fingerprint.ua_metadata` and the result drives
38    /// `Emulation.setUserAgentOverride.userAgentMetadata` in place of the
39    /// fingerprint-derived value outright.
40    ua_metadata: Option<UaMetadata>,
41    /// Custom screen / device-metrics from the resolved [`Persona`]. When
42    /// set, drives `Emulation.setDeviceMetricsOverride` in place of the
43    /// fixed 1920x1080 default.
44    screen: Option<ScreenSpec>,
45}
46
47impl StealthObserver {
48    /// Build a new observer. Bootstrap source is composed eagerly so the
49    /// per-target hot path only pays a `clone`/borrow.
50    ///
51    /// The bootstrap is driven by a [`Persona`] (surface-spoofing config) plus
52    /// the [`Fingerprint`] (coherent UA / Chrome identity). This constructor
53    /// uses [`Persona::default`] — no surface overrides, identity-only patches
54    /// keep their current behavior. The launch path (a later task) will thread
55    /// a caller-supplied persona via [`StealthObserver::with_persona`].
56    #[must_use]
57    pub fn new(profile: StealthProfile, fingerprint: Fingerprint) -> Self {
58        Self::with_persona(profile, fingerprint, Persona::default())
59    }
60
61    /// Build a new observer with an explicit [`Persona`] driving the surface
62    /// patches. `identity` still supplies the coherent UA / Chrome version.
63    #[must_use]
64    pub fn with_persona(
65        profile: StealthProfile,
66        fingerprint: Fingerprint,
67        persona: Persona,
68    ) -> Self {
69        let bootstrap = match profile.kind() {
70            ProfileKind::Spoofed => {
71                if profile.native_webgl_enabled() {
72                    bootstrap_script_native_webgl(&persona, &fingerprint)
73                } else {
74                    bootstrap_script(&persona, &fingerprint)
75                }
76            }
77            // `Native` gets the geometry repair and nothing else. It still receives
78            // `Emulation.setDeviceMetricsOverride` below, which sets `inner*` and
79            // `screen.width`/`height` but cannot reach `outer*`/`avail*` — leaving
80            // `outerWidth` at the OS default under a 1920 `innerWidth` (content wider than
81            // its own window) and `availHeight === height` (no taskbar inset). Those are
82            // artifacts this library introduces, not properties of the host, so repairing
83            // them keeps `Native` native rather than spoofing anything.
84            // The persona's screen still reaches this arm even in `Native`: it
85            // carries no identity, only the geometry CDP cannot set. A profile
86            // captured on real hardware brings its own insets, and presenting
87            // the derived defaults instead would describe a machine that does
88            // not exist. `None` is byte-identical to the old call.
89            ProfileKind::Native => geometry_bootstrap_with(persona.screen.as_ref()),
90            ProfileKind::Off => String::new(),
91        };
92        let geolocation = persona.geolocation;
93        let ua_metadata = persona.ua.as_ref().and_then(|u| u.ua_metadata.clone());
94        let screen = persona.screen;
95        Self {
96            profile,
97            fingerprint,
98            bootstrap,
99            geolocation,
100            ua_metadata,
101            screen,
102        }
103    }
104}

106#[async_trait::async_trait]
107impl TargetObserver for StealthObserver {
108    fn name(&self) -> &'static str {
109        "stealth"
110    }
111
112    async fn on_target_attached(&self, session: PausedSession<'_>) -> Result<(), ObserverError> {
113        // Workers + iframes are skipped — workers have no DOM; iframes inherit
114        // patches via the parent in flat mode.
115        if session.target_info.kind != "page" {
116            return Ok(());
117        }
118        if self.profile.kind() == ProfileKind::Off {
119            return Ok(());
120        }
121
122        session.call("Page.enable", json!({})).await?;
123
124        // UA override — Emulation.setUserAgentOverride carries the Client-Hints
125        // metadata too, so we don't have to send Network.setUserAgentOverride
126        // separately.
127        let accept_language = {
128            let langs = crate::lang::resolve_languages(&Persona::default(), &self.fingerprint);
129            // `Emulation.setUserAgentOverride.acceptLanguage` wants a PLAIN
130            // comma-separated locale list (e.g. `en-US,en`) — Chrome appends the
131            // `;q=` weights itself. Passing an already-weighted string (the
132            // `accept_language()` header form) makes Chrome double them, yielding
133            // a malformed `Accept-Language: en-US,en;q=0.9;q=0.9`. Send the bare
134            // list so the emitted header is a clean `en-US,en;q=0.9`.
135            langs.join(",")
136        };
137        // Persona UA-CH wins when supplied — field-wise, falling back to the
138        // fingerprint-derived value for any sub-field the persona left
139        // unset. Absent persona UA-CH → today's behavior (fingerprint's
140        // UAM verbatim).
141        let user_agent_metadata = match &self.ua_metadata {
142            Some(custom) => custom.resolve(&self.fingerprint.ua_metadata),
143            None => self.fingerprint.ua_metadata.clone(),
144        };
145        session
146            .call(
147                "Emulation.setUserAgentOverride",
148                json!({
149                    "userAgent": &self.fingerprint.ua_string,
150                    "acceptLanguage": accept_language,
151                    // `js_string()`, NOT `ch_platform()`. This CDP parameter is defined as "the
152                    // platform navigator.platform should return", so it takes the legacy JS value
153                    // (`Win32` / `MacIntel` / `Linux x86_64`). The Client-Hints spelling
154                    // (`Windows` / `macOS` / `Linux`) belongs to `userAgentMetadata.platform`,
155                    // which is sent separately just below and derives it in `Fingerprint::new`.
156                    //
157                    // Sending the CH spelling here made every session report a
158                    // `navigator.platform` no real browser emits. It was invisible under
159                    // `spoofed`, whose bootstrap re-patches `navigator.platform` from
160                    // `platformJs` and so happened to paper over it — but the bootstrap is
161                    // deliberately EMPTY for `Off`/`Native` (see `bootstrap` above), leaving
162                    // `native` profiles presenting `"macOS"` where Chrome reports `"MacIntel"`.
163                    // Measured against a real Chrome build, not inferred.
164                    "platform": self.fingerprint.platform.js_string(),
165                    "userAgentMetadata": &user_agent_metadata,
166                }),
167            )
168            .await?;
169
170        // Screen-size override + focus emulation: keeps headless from leaking
171        // an oddly-shaped viewport and from reporting `document.hasFocus()`
172        // false for the (always-backgrounded) headless tab. Persona screen
173        // wins when supplied; absent → today's fixed 1920x1080 default.
174        let (screen_width, screen_height, device_scale_factor) = match self.screen {
175            Some(s) => (s.width, s.height, s.device_pixel_ratio),
176            None => (1920, 1080, 1.0),
177        };
178        session
179            .call(
180                "Emulation.setDeviceMetricsOverride",
181                json!({
182                    "width": screen_width,
183                    "height": screen_height,
184                    "deviceScaleFactor": device_scale_factor,
185                    "mobile": false,
186                    "screenWidth": screen_width,
187                    "screenHeight": screen_height,
188                }),
189            )
190            .await?;
191
192        session
193            .call(
194                "Emulation.setFocusEmulationEnabled",
195                json!({ "enabled": true }),
196            )
197            .await?;
198
199        if let Some(ref tz) = self.fingerprint.timezone {
200            session
201                .call("Emulation.setTimezoneOverride", json!({ "timezoneId": tz }))
202                .await?;
203        }
204        // Keep the JS-visible locale (navigator.language, Intl) coherent with
205        // the always-sent Accept-Language. Prefer an explicit fingerprint
206        // locale; otherwise, if the fingerprint pins a `languages` list, derive
207        // the locale from its primary entry — a `languages`-without-`locale`
208        // fingerprint must not leave the JS locale at Chrome's default while the
209        // Accept-Language header says otherwise. A pure-default fingerprint (no
210        // locale, no languages) keeps Chrome's native locale; we don't force one.
211        let effective_locale = self.fingerprint.locale.clone().or_else(|| {
212            self.fingerprint
213                .languages
214                .as_ref()
215                .and_then(|langs| langs.first())
216                .cloned()
217        });
218        if let Some(ref locale) = effective_locale {
219            // `Emulation.setLocaleOverride` is browser-global: only one override
220            // can be in effect at a time. A page with a cross-origin OOPIF (its
221            // own target -> its own session) attaches this observer per session,
222            // so a later session finds the override already set by the first and
223            // Chrome replies `[-32000] Another locale override is already in
224            // effect`. That is benign — the same coherent locale is already
225            // applied — and must NOT propagate: propagating detaches the observer
226            // and strips the frame's remaining patches (CSP bypass, the bootstrap
227            // fingerprint script). Tolerate that one reply; surface every other error.
228            match session
229                .call("Emulation.setLocaleOverride", json!({ "locale": locale }))
230                .await
231            {
232                Ok(_) => {}
233                Err(CallError::Rpc(_, ref message, _))
234                    if message.contains("locale override is already in effect") =>
235                {
236                    tracing::debug!(
237                        %locale,
238                        "locale override already in effect (set by an earlier session); tolerating"
239                    );
240                }
241                Err(e) => return Err(e.into()),
242            }
243        }
244        if let Some(ref geo) = self.geolocation {
245            // Sets the value the Geolocation API *would* return — it does
246            // NOT grant the `geolocation` permission. Chrome still gates the
247            // API behind a permission prompt/grant; auto-granting it here
248            // would be a separate (and itself suspicious) signal, so we
249            // deliberately leave permissioning to the caller.
250            let mut params = json!({
251                "latitude": geo.latitude,
252                "longitude": geo.longitude,
253            });
254            if let Some(accuracy) = geo.accuracy {
255                params["accuracy"] = json!(accuracy);
256            }
257            session
258                .call("Emulation.setGeolocationOverride", params)
259                .await?;
260        }
261
262        // CSP bypass stays SPOOFED-only: it weakens the page's own security policy, which is
263        // only justified by the full identity bootstrap needing to run.
264        if self.profile.kind() == ProfileKind::Spoofed && self.profile.bypass_csp_enabled() {
265            session
266                .call("Page.setBypassCSP", json!({ "enabled": true }))
267                .await?;
268        }
269
270        // Injection is driven by whether there IS a bootstrap, not by the profile kind.
271        // `Native` now carries the geometry repair (and nothing else), and gating the send on
272        // `Spoofed` was a second, independent gate that would have left that repair assembled
273        // but never sent. `Off` still produces an empty bootstrap and so still sends nothing.
274        if !self.bootstrap.is_empty() {
275            // Inject into the MAIN world (no `worldName`). The bootstrap's
276            // patches mutate `Navigator.prototype`, `window.chrome`,
277            // `WebGLRenderingContext.prototype`, etc. — every isolated
278            // world gets its own copy of these prototypes, so a patch
279            // applied in a named/isolated world is invisible to the
280            // page's own scripts (and to `evaluate_main`, the surface
281            // detection sites probe). Running the bootstrap in the main
282            // world is the only way these prototype mutations actually
283            // affect the document under test.
284            session
285                .call(
286                    "Page.addScriptToEvaluateOnNewDocument",
287                    json!({
288                        "source": &self.bootstrap,
289                        "includeCommandLineAPI": false,
290                        "runImmediately": true,
291                    }),
292                )
293                .await?;
294        }
295
296        Ok(())
297    }
298}

308    /// `Emulation.setUserAgentOverride` carries the platform on TWO distinct axes, and they
309    /// must not be swapped:
310    ///
311    /// * top-level `platform` — CDP defines this as "the platform navigator.platform should
312    ///   return", so it takes the legacy JS spelling `Win32` / `MacIntel` / `Linux x86_64`
313    ///   ([`Platform::js_string`]).
314    /// * `userAgentMetadata.platform` — the Client-Hints spelling `Windows` / `macOS` /
315    ///   `Linux` ([`Platform::ch_platform`]).
316    ///
317    /// The two sets are disjoint, so this asserts membership rather than a fixture literal and
318    /// therefore catches the swap on whatever platform the test host happens to be.
319    ///
320    /// Regression guarded: the top-level param was sent as `ch_platform()`, so every session
321    /// reported a `navigator.platform` no real browser emits. It was invisible under `spoofed`,
322    /// whose bootstrap re-patches the property from `platformJs`, but that bootstrap is empty
323    /// for `Off`/`Native` — so `native` profiles shipped the bad value straight through.
324    fn assert_platform_axes_not_swapped(sent: &serde_json::Value) {
325        const JS_SPELLINGS: [&str; 3] = ["Win32", "MacIntel", "Linux x86_64"];
326        const CH_SPELLINGS: [&str; 3] = ["Windows", "macOS", "Linux"];
327
328        let top = sent["params"]["platform"]
329            .as_str()
330            .expect("setUserAgentOverride.platform must be sent");
331        assert!(
332            JS_SPELLINGS.contains(&top),
333            "setUserAgentOverride.platform sets navigator.platform, so it must be one of \
334             {JS_SPELLINGS:?} (Platform::js_string). Got {top:?} — if that is one of \
335             {CH_SPELLINGS:?} then the two platform axes are swapped, and no real browser \
336             reports those for navigator.platform."
337        );
338
339        // The CH axis must still carry the CH spelling: a fix that swapped both would just
340        // trade one impossible value for another.
341        let ch = sent["params"]["userAgentMetadata"]["platform"]
342            .as_str()
343            .expect("userAgentMetadata.platform must be sent");
344        assert!(
345            CH_SPELLINGS.contains(&ch),
346            "userAgentMetadata.platform is the Client-Hints axis and must be one of \
347             {CH_SPELLINGS:?} (Platform::ch_platform). Got {ch:?}."
348        );
349    }
