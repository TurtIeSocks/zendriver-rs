1//! StealthObserver: applies a [`StealthProfile`] to each new attached target.
2//!
3//! Installed in the [`zendriver_transport`] actor's observer chain. On
4//! `Target.attachedToTarget`, the actor pauses the new target, walks every
5//! observer serially, then releases the debugger via
6//! `Runtime.runIfWaitingForDebugger`. The observer's job is to push every
7//! UA/screen/timezone/locale override and (for spoofed) the bootstrap script
8//! _before_ the debugger releases, so the first script the page runs sees the
9//! patched globals.

19/// Observer that applies a [`StealthProfile`] + [`Fingerprint`] to every page
20/// and out-of-process iframe target. Workers are skipped — they have no DOM;
21/// same-process iframes are not targets and get the page's patches.
22#[derive(Debug)]
23pub struct StealthObserver {
24    profile: StealthProfile,
25    fingerprint: Fingerprint,
26    /// Pre-rendered bootstrap source. Empty for `Off`/`Native` — we never send
27    /// `Page.addScriptToEvaluateOnNewDocument` in those modes, so there is no
28    /// need to pay the patches-bundle cost for them.
29    bootstrap: String,
30    /// Mock geolocation coordinates from the resolved [`Persona`], sent via
31    /// `Emulation.setGeolocationOverride`. Unlike
32    /// `timezone`/`locale`/`languages`/`screen` (folded into `fingerprint` by
33    /// [`Fingerprint::overlay_persona`]), geolocation has no `Fingerprint`
34    /// counterpart, so it's captured here straight off the persona at
35    /// construction time.
36    geolocation: Option<GeoPos>,
37    /// Custom UA-CH from the resolved [`Persona`]'s `ua.ua_metadata`. When
38    /// set, [`UaMetadata::resolve`] fills any unset sub-field from
39    /// `fingerprint.ua_metadata` and the result drives
40    /// `Emulation.setUserAgentOverride.userAgentMetadata` in place of the
41    /// fingerprint-derived value outright. Field-wise rather than
42    /// whole-value, which is why it is not folded into `fingerprint`.
43    ua_metadata: Option<UaMetadata>,
44}
45
46impl StealthObserver {
47    /// Build a new observer. Bootstrap source is composed eagerly so the
48    /// per-target hot path only pays a `clone`/borrow.
49    ///
50    /// The bootstrap is driven by a [`Persona`] (surface-spoofing config) plus
51    /// the [`Fingerprint`] (coherent UA / Chrome identity). This constructor
52    /// uses [`Persona::default`] — no surface overrides, identity-only patches
53    /// keep their current behavior. The launch path (a later task) will thread
54    /// a caller-supplied persona via [`StealthObserver::with_persona`].
55    #[must_use]
56    pub fn new(profile: StealthProfile, fingerprint: Fingerprint) -> Self {
57        Self::with_persona(profile, fingerprint, Persona::default())
58    }
59
60    /// Build a new observer with an explicit [`Persona`] driving the surface
61    /// patches. `fingerprint` still supplies the coherent UA / Chrome version.
62    ///
63    /// This is where the persona and the fingerprint are combined, so it is
64    /// where they are merged: the persona's `timezone` / `locale` /
65    /// `languages` / `screen` are folded into `fingerprint` up front, and
66    /// every consumer below — the
67    /// `Emulation.set*Override` calls, the `Accept-Language` header, the
68    /// bootstrap script's geometry and `navigator.languages` patches — then
69    /// reads that one merged value. An explicitly-set persona field wins over
70    /// the fingerprint's; `None` inherits the fingerprint's.
71    #[must_use]
72    pub fn with_persona(
73        profile: StealthProfile,
74        mut fingerprint: Fingerprint,
75        persona: Persona,
76    ) -> Self {
77        fingerprint.overlay_persona(&persona);
78
79        let bootstrap = match profile.kind() {
80            ProfileKind::Spoofed => {
81                if profile.native_webgl_enabled() {
82                    bootstrap_script_native_webgl(&persona, &fingerprint)
83                } else {
84                    bootstrap_script(&persona, &fingerprint)
85                }
86            }
87            // `Native` gets the geometry repair and nothing else. It still receives
88            // `Emulation.setDeviceMetricsOverride` below, which sets `inner*` and
89            // `screen.width`/`height` but cannot reach `outer*`/`avail*` — leaving
90            // `outerWidth` at the OS default under a 1920 `innerWidth` (content wider than
91            // its own window) and `availHeight === height` (no taskbar inset). Those are
92            // artifacts this library introduces, not properties of the host, so repairing
93            // them keeps `Native` native rather than spoofing anything.
94            // The configured screen still reaches this arm even in `Native`: it
95            // carries no identity, only the geometry CDP cannot set. A profile
96            // captured on real hardware brings its own insets, and presenting
97            // the derived defaults instead would describe a machine that does
98            // not exist. `None` is byte-identical to the old call. It reads the
99            // merged `fingerprint.screen` so the JS geometry and the CDP metrics
100            // override below describe the same display — a screen that reached
101            // one but not the other is worse than one that reached neither.
102            ProfileKind::Native => geometry_bootstrap_with(fingerprint.screen.as_ref()),
103            ProfileKind::Off => String::new(),
104        };
105        let geolocation = persona.geolocation;
106        let ua_metadata = persona.ua.as_ref().and_then(|u| u.ua_metadata.clone());
107        Self {
108            profile,
109            fingerprint,
110            bootstrap,
111            geolocation,
112            ua_metadata,
113        }
114    }
115}

117#[async_trait::async_trait]
118impl TargetObserver for StealthObserver {
119    fn name(&self) -> &'static str {
120        "stealth"
121    }
122
123    async fn on_target_attached(&self, session: PausedSession<'_>) -> Result<(), ObserverError> {
124        // An out-of-process iframe is a target of its own and inherits nothing
125        // from its page: it gets the same identity, locale and bootstrap.
126        // Viewport and focus belong to the tab and stay page-only. Workers are
127        // skipped — they have no DOM.
128        let frame = match session.target_info.kind.as_str() {
129            "page" => false,
130            "iframe" => true,
131            _ => return Ok(()),
132        };
133        if self.profile.kind() == ProfileKind::Off {
134            return Ok(());
135        }
136
137        session.call("Page.enable", json!({})).await?;
138
139        // UA override — Emulation.setUserAgentOverride carries the Client-Hints
140        // metadata too, so we don't have to send Network.setUserAgentOverride
141        // separately.
142        let accept_language = {
143            // The fingerprint is already merged with the persona (see
144            // `with_persona`), so its language list is the effective one — a
145            // caller's `Persona::languages` / `Persona::locale` reaches the
146            // header from here.
147            let langs = crate::lang::fingerprint_languages(&self.fingerprint);
148            // `Emulation.setUserAgentOverride.acceptLanguage` wants a PLAIN
149            // comma-separated locale list (e.g. `en-US,en`) — Chrome appends the
150            // `;q=` weights itself. Passing an already-weighted string (the
151            // `accept_language()` header form) makes Chrome double them, yielding
152            // a malformed `Accept-Language: en-US,en;q=0.9;q=0.9`. Send the bare
153            // list so the emitted header is a clean `en-US,en;q=0.9`.
154            langs.join(",")
155        };
156        // Persona UA-CH wins when supplied — field-wise, falling back to the
157        // fingerprint-derived value for any sub-field the persona left
158        // unset. Absent persona UA-CH → today's behavior (fingerprint's
159        // UAM verbatim).
160        let user_agent_metadata = match &self.ua_metadata {
161            Some(custom) => custom.resolve(&self.fingerprint.ua_metadata),
162            None => self.fingerprint.ua_metadata.clone(),
163        };
164        session
165            .call(
166                "Emulation.setUserAgentOverride",
167                json!({
168                    "userAgent": &self.fingerprint.ua_string,
169                    "acceptLanguage": accept_language,
170                    // `js_string()`, NOT `ch_platform()`. This CDP parameter is defined as "the
171                    // platform navigator.platform should return", so it takes the legacy JS value
172                    // (`Win32` / `MacIntel` / `Linux x86_64`). The Client-Hints spelling
173                    // (`Windows` / `macOS` / `Linux`) belongs to `userAgentMetadata.platform`,
174                    // which is sent separately just below and derives it in `Fingerprint::new`.
175                    //
176                    // Sending the CH spelling here made every session report a
177                    // `navigator.platform` no real browser emits. It was invisible under
178                    // `spoofed`, whose bootstrap re-patches `navigator.platform` from
179                    // `platformJs` and so happened to paper over it — but the bootstrap is
180                    // deliberately EMPTY for `Off`/`Native` (see `bootstrap` above), leaving
181                    // `native` profiles presenting `"macOS"` where Chrome reports `"MacIntel"`.
182                    // Measured against a real Chrome build, not inferred.
183                    "platform": self.fingerprint.platform.js_string(),
184                    "userAgentMetadata": &user_agent_metadata,
185                }),
186            )
187            .await?;
188
189        // Screen-size override + focus emulation: keeps headless from leaking
190        // an oddly-shaped viewport and from reporting `document.hasFocus()`
191        // false for the (always-backgrounded) headless tab. The merged
192        // fingerprint's screen (persona's when it set one, else the
193        // `StealthProfile::screen` pin) wins; neither set → the fixed
194        // 1920x1080 default.
195        if !frame {
196            let (screen_width, screen_height, device_scale_factor) = match self.fingerprint.screen {
197                Some(s) => (s.width, s.height, s.device_pixel_ratio),
198                None => (1920, 1080, 1.0),
199            };
200            session
201                .call(
202                    "Emulation.setDeviceMetricsOverride",
203                    json!({
204                        "width": screen_width,
205                        "height": screen_height,
206                        "deviceScaleFactor": device_scale_factor,
207                        "mobile": false,
208                        "screenWidth": screen_width,
209                        "screenHeight": screen_height,
210                    }),
211                )
212                .await?;
213
214            session
215                .call(
216                    "Emulation.setFocusEmulationEnabled",
217                    json!({ "enabled": true }),
218                )
219                .await?;
220        }
221
222        if let Some(ref tz) = self.fingerprint.timezone {
223            session
224                .call("Emulation.setTimezoneOverride", json!({ "timezoneId": tz }))
225                .await?;
226        }
227        // Keep the JS-visible locale (navigator.language, Intl) coherent with
228        // the Accept-Language sent above by resolving both from the same
229        // helper: this is the head of that same list, so the two surfaces
230        // cannot disagree. Resolving them separately is what let them —
231        // `Intl` reported a locale the header never advertised. A fingerprint
232        // configuring neither locale nor languages yields `None` and keeps
233        // Chrome's native locale; we don't force one.
234        let effective_locale = crate::lang::effective_locale(&self.fingerprint);
235        if let Some(ref locale) = effective_locale {
236            // `Emulation.setLocaleOverride` is browser-global: only one override
237            // can be in effect at a time. A page with a cross-origin OOPIF (its
238            // own target -> its own session) attaches this observer per session,
239            // so a later session finds the override already set by the first and
240            // Chrome replies `[-32000] Another locale override is already in
241            // effect`. That is benign — the same coherent locale is already
242            // applied — and must NOT propagate: propagating detaches the observer
243            // and strips the frame's remaining patches (CSP bypass, the bootstrap
244            // fingerprint script). Tolerate that one reply; surface every other error.
245            match session
246                .call("Emulation.setLocaleOverride", json!({ "locale": locale }))
247                .await
248            {
249                Ok(_) => {}
250                Err(CallError::Rpc(_, ref message, _))
251                    if message.contains("locale override is already in effect") =>
252                {
253                    tracing::debug!(
254                        %locale,
255                        "locale override already in effect (set by an earlier session); tolerating"
256                    );
257                }
258                Err(e) => return Err(e.into()),
259            }
260        }
261        if let Some(ref geo) = self.geolocation {
262            // Sets the value the Geolocation API *would* return — it does
263            // NOT grant the `geolocation` permission. Chrome still gates the
264            // API behind a permission prompt/grant; auto-granting it here
265            // would be a separate (and itself suspicious) signal, so we
266            // deliberately leave permissioning to the caller.
267            let mut params = json!({
268                "latitude": geo.latitude,
269                "longitude": geo.longitude,
270            });
271            if let Some(accuracy) = geo.accuracy {
272                params["accuracy"] = json!(accuracy);
273            }
274            session
275                .call("Emulation.setGeolocationOverride", params)
276                .await?;
277        }
278
279        // CSP bypass stays SPOOFED-only: it weakens the page's own security policy, which is
280        // only justified by the full identity bootstrap needing to run.
281        if self.profile.kind() == ProfileKind::Spoofed && self.profile.bypass_csp_enabled() {
282            session
283                .call("Page.setBypassCSP", json!({ "enabled": true }))
284                .await?;
285        }
286
287        // Injection is driven by whether there IS a bootstrap, not by the profile kind.
288        // `Native` now carries the geometry repair (and nothing else), and gating the send on
289        // `Spoofed` was a second, independent gate that would have left that repair assembled
290        // but never sent. `Off` still produces an empty bootstrap and so still sends nothing.
291        if !self.bootstrap.is_empty() {
292            // Inject into the MAIN world (no `worldName`). The bootstrap's
293            // patches mutate `Navigator.prototype`, `window.chrome`,
294            // `WebGLRenderingContext.prototype`, etc. — every isolated
295            // world gets its own copy of these prototypes, so a patch
296            // applied in a named/isolated world is invisible to the
297            // page's own scripts (and to `evaluate_main`, the surface
298            // detection sites probe). Running the bootstrap in the main
299            // world is the only way these prototype mutations actually
300            // affect the document under test.
301            session
302                .call(
303                    "Page.addScriptToEvaluateOnNewDocument",
304                    json!({
305                        "source": &self.bootstrap,
306                        "includeCommandLineAPI": false,
307                        "runImmediately": true,
308                    }),
309                )
310                .await?;
311        }
312
313        Ok(())
314    }
315}

325    /// `Emulation.setUserAgentOverride` carries the platform on TWO distinct axes, and they
326    /// must not be swapped:
327    ///
328    /// * top-level `platform` — CDP defines this as "the platform navigator.platform should
329    ///   return", so it takes the legacy JS spelling `Win32` / `MacIntel` / `Linux x86_64`
330    ///   ([`Platform::js_string`]).
331    /// * `userAgentMetadata.platform` — the Client-Hints spelling `Windows` / `macOS` /
332    ///   `Linux` ([`Platform::ch_platform`]).
333    ///
334    /// The two sets are disjoint, so this asserts membership rather than a fixture literal and
335    /// therefore catches the swap on whatever platform the test host happens to be.
336    ///
337    /// Regression guarded: the top-level param was sent as `ch_platform()`, so every session
338    /// reported a `navigator.platform` no real browser emits. It was invisible under `spoofed`,
339    /// whose bootstrap re-patches the property from `platformJs`, but that bootstrap is empty
340    /// for `Off`/`Native` — so `native` profiles shipped the bad value straight through.
341    fn assert_platform_axes_not_swapped(sent: &serde_json::Value) {
342        const JS_SPELLINGS: [&str; 3] = ["Win32", "MacIntel", "Linux x86_64"];
343        const CH_SPELLINGS: [&str; 3] = ["Windows", "macOS", "Linux"];
344
345        let top = sent["params"]["platform"]
346            .as_str()
347            .expect("setUserAgentOverride.platform must be sent");
348        assert!(
349            JS_SPELLINGS.contains(&top),
350            "setUserAgentOverride.platform sets navigator.platform, so it must be one of \
351             {JS_SPELLINGS:?} (Platform::js_string). Got {top:?} — if that is one of \
352             {CH_SPELLINGS:?} then the two platform axes are swapped, and no real browser \
353             reports those for navigator.platform."
354        );
355
356        // The CH axis must still carry the CH spelling: a fix that swapped both would just
357        // trade one impossible value for another.
358        let ch = sent["params"]["userAgentMetadata"]["platform"]
359            .as_str()
360            .expect("userAgentMetadata.platform must be sent");
361        assert!(
362            CH_SPELLINGS.contains(&ch),
363            "userAgentMetadata.platform is the Client-Hints axis and must be one of \
364             {CH_SPELLINGS:?} (Platform::ch_platform). Got {ch:?}."
365        );
366    }

1252    /// The single trimmed line of `source` starting with `prefix`. Keeps a
1253    /// failure over the bootstrap readable — the script is thousands of lines,
1254    /// and only the substituted token is in question.
1255    fn line_with<'a>(source: &'a str, prefix: &str) -> &'a str {
1256        source
1257            .lines()
1258            .map(str::trim)
1259            .find(|l| l.starts_with(prefix))
1260            .unwrap_or("<no such line>")
1261    }

1270    /// The `params` of the single frame for `method`, panicking with the whole
1271    /// observed sequence when it was never sent.
1272    fn params<'a>(frames: &'a [Value], method: &str) -> &'a Value {
1273        frames
1274            .iter()
1275            .find(|f| f["method"] == method)
1276            .map(|f| &f["params"])
1277            .unwrap_or_else(|| {
1278                panic!(
1279                    "{method} was never sent; the observer sent {:?}",
1280                    method_names(frames)
1281                )
1282            })
1283    }
