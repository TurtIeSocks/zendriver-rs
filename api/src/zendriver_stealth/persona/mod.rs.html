21/// Mock geolocation coordinates, sent verbatim as

31    pub latitude: f64,
32    pub longitude: f64,
33    pub accuracy: Option<f64>,

36/// Unified fingerprint configuration. Every field optional → overlay semantics.
37#[derive(Debug, Clone, Default, Serialize, Deserialize)]
38pub struct Persona {
39    pub platform: Option<Platform>,
40    pub ua: Option<UaSpec>,
41    pub hardware_concurrency: Option<u32>,
42    pub device_memory_gb: Option<u32>,
43    pub timezone: Option<String>,
44    pub locale: Option<String>,
45    /// Ordered language list (e.g. `["de-DE", "de"]`). Drives
46    /// `navigator.languages` and the q-weighted `Accept-Language`. When unset,
47    /// derived from [`locale`](Self::locale).
48    pub languages: Option<Vec<String>>,
49    /// Mock coordinates for the Geolocation API (`Emulation.setGeolocationOverride`).
50    /// Coherence axis alongside [`timezone`](Self::timezone) /
51    /// [`locale`](Self::locale) — keeps `navigator.geolocation` in step with
52    /// the exit IP's real location instead of leaking the host's.
53    pub geolocation: Option<GeoPos>,
54    /// Screen / device-metrics override (`Emulation.setDeviceMetricsOverride`).
55    /// `None` → the observer's fixed 1920x1080 default (today's behavior).
56    pub screen: Option<ScreenSpec>,
57    pub webgl: Option<WebglSpec>,
58    /// WebGPU adapter override — decorate a real adapter's `.info` (deriving
59    /// vendor/architecture from [`webgl`](Self::webgl) when unset, the
60    /// default), and optionally fabricate a synthetic adapter on a GPU-less
61    /// host. **Opt-in only** — see [`WebgpuSpec`] for the full accuracy
62    /// warning and v1 limitations before setting `limits`/`features`.
63    pub webgpu: Option<WebgpuSpec>,
64    /// One coherent GPU identity: every readable WebGL value plus the WebGPU
65    /// adapter, resolved from the tier tables.
66    ///
67    /// `None` resolves from the persona's WebGL renderer via the device table.
68    /// The finer-grained [`WebglSpec`](specs::WebglSpec) and
69    /// [`WebgpuSpec`](specs::WebgpuSpec) still overlay on top of whatever this
70    /// produces, so a caller can pin one value without restating a whole device.
71    pub gpu: Option<GpuProfile>,
72    pub canvas: Option<SurfaceCfg>,
73    pub audio: Option<SurfaceCfg>,
74    pub fonts: Option<FontSpec>,
75    pub client_rects: Option<SurfaceCfg>,
76    pub webrtc: Option<WebrtcSpec>,
77    pub hardware: Option<HardwareSpec>,
78    pub seed: Option<Seed>,
79}

124    /// Claim a catalogued GPU.
125    ///
126    /// Deliberately **not** a new [`Persona`] field. A device's whole
127    /// contribution is its renderer string, and pinning that already selects
128    /// the capability tier, the WebGPU adapter, and the vendor through
129    /// machinery that exists — so this sets [`WebglSpec::unmasked_renderer`]
130    /// and nothing else. A dedicated field would add a fourth precedence layer
131    /// and a serde problem (a [`GpuDevice`] is a pointer into a generated
132    /// table) to buy nothing.
133    ///
134    /// Overwrites only the renderer, so a caller can still pin the vendor or a
135    /// strategy alongside it:
136    ///
137    /// ```no_run
138    /// use zendriver_stealth::{GpuDevice, Persona, Platform};
139    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
140    /// let persona = Persona {
141    ///     platform: Some(Platform::Win32),
142    ///     ..Persona::builder()
143    ///         .gpu_device(GpuDevice::by_name("NVIDIA GeForce RTX 4090")?)
144    ///         .build()
145    /// };
146    /// # Ok(())
147    /// # }
148    /// ```
149    #[must_use]
150    pub fn gpu_device(mut self, device: crate::GpuDevice) -> Self {
151        let mut spec = self.0.webgl.unwrap_or_default();
152        spec.unmasked_renderer = Some(device.renderer());
153        self.0.webgl = Some(spec);
154        self
155    }
156    pub fn build(self) -> Persona {
157        self.0
158    }

169    /// Field-wise merge: `Some` in `over` wins, `None` inherits from `self`.
170    pub fn overlay(self, over: Persona) -> Persona {
171        Persona {
172            platform: over.platform.or(self.platform),
173            // UA-CH composes field-wise (via `UaSpec::overlay`) rather than
174            // one side wholesale-replacing the other, so layering two
175            // personas can patch e.g. just `platform_version` without
176            // clobbering a base persona's `brands`.
177            ua: match (self.ua, over.ua) {
178                (Some(base), Some(add)) => Some(base.overlay(add)),
179                (base, add) => add.or(base),
180            },
181            hardware_concurrency: over.hardware_concurrency.or(self.hardware_concurrency),
182            device_memory_gb: over.device_memory_gb.or(self.device_memory_gb),
183            timezone: over.timezone.or(self.timezone),
184            locale: over.locale.or(self.locale),
185            languages: over.languages.or(self.languages),
186            geolocation: over.geolocation.or(self.geolocation),
187            screen: over.screen.or(self.screen),
188            webgl: over.webgl.or(self.webgl),
189            webgpu: over.webgpu.or(self.webgpu),
190            // Whole-value, not field-wise: a GPU is one coherent artifact
191            // (same rule as `screen`). Merging two personas' GPUs field-wise
192            // could compose a device that exists nowhere — Metal's texture
193            // limits beside D3D11's viewport bound is exactly the
194            // incoherence this branch exists to eliminate.
195            gpu: over.gpu.or(self.gpu),
196            canvas: over.canvas.or(self.canvas),
197            audio: over.audio.or(self.audio),
198            fonts: over.fonts.or(self.fonts),
199            client_rects: over.client_rects.or(self.client_rects),
200            webrtc: over.webrtc.or(self.webrtc),
201            hardware: over.hardware.or(self.hardware),
202            seed: over.seed.or(self.seed),
203        }
204    }

206    /// Force a single surface's render [`Strategy`], creating the surface's
207    /// spec with default values if it was unset.
208    ///
209    /// Used by the public browser builder's `.surface(surface, strategy)` to
210    /// layer per-surface overrides on top of the resolved persona. Each surface
211    /// writes the `strategy` field of its corresponding spec; the [`UaSpec`]
212    /// surface family (identity, not a render strategy) has no strategy field,
213    /// so this is a no-op there.
214    pub fn apply_surface_override(
215        &mut self,
216        surface: surface::Surface,
217        strategy: surface::Strategy,
218    ) {
219        use surface::Surface;
220        match surface {
221            Surface::Canvas => {
222                self.canvas.get_or_insert_with(Default::default).strategy = Some(strategy)
223            }
224            Surface::Audio => {
225                self.audio.get_or_insert_with(Default::default).strategy = Some(strategy)
226            }
227            Surface::ClientRects => {
228                self.client_rects
229                    .get_or_insert_with(Default::default)
230                    .strategy = Some(strategy)
231            }
232            Surface::Webgl => {
233                self.webgl.get_or_insert_with(Default::default).strategy = Some(strategy)
234            }
235            Surface::Fonts => {
236                self.fonts.get_or_insert_with(Default::default).strategy = Some(strategy)
237            }
238            Surface::Webrtc => {
239                self.webrtc.get_or_insert_with(Default::default).strategy = Some(strategy)
240            }
241            Surface::Hardware => {
242                self.hardware.get_or_insert_with(Default::default).strategy = Some(strategy)
243            }
244            Surface::Webgpu => {
245                self.webgpu.get_or_insert_with(Default::default).strategy = Some(strategy)
246            }
247        }
248    }

261    /// Host-probed persona (sysinfo). Cached: first call probes, rest clone.
262    /// Runtime — NOT a build-script const (build host != run host).
263    ///
264    /// Reports only what it could actually read. An attribute whose probe fails
265    /// is left `None` (with a `WARN` naming it) rather than filled with a
266    /// stand-in value, so a caller never mistakes a library default for this
267    /// host. Downstream resolution supplies the fallback explicitly.
268    pub fn system() -> Persona {
269        SYSTEM.get_or_init(Persona::probe_system).clone()
270    }

272    fn probe_system() -> Persona {
273        Persona::from_host_probe(
274            crate::fingerprint::detect_platform(),
275            crate::fingerprint::clamp_cpu_count(num_cpus::get() as u32),
276            crate::fingerprint::detect_memory_gb(),
277        )
278    }

280    /// Assemble the host persona from already-probed values. Split out of
281    /// [`Persona::probe_system`] so the failure path is reachable from tests
282    /// without a host that actually fails to report its RAM (the same idiom as
283    /// `fingerprint::parse_version_banner`).
284    ///
285    /// A probe that fails leaves its field `None` — never a stand-in constant.
286    /// This persona's entire contract is "the real host", so substituting a
287    /// plausible-looking value would be the worst available failure: the caller
288    /// believes they hold a coherent host identity while every user of this
289    /// crate whose probe failed ships the *same* value, which is exactly the
290    /// cross-user correlation signal a persona exists to avoid. `None` is the
291    /// persona's "unset, resolve downstream" state, and downstream resolution
292    /// is already explicit — `patches::identity_iife` falls back to the
293    /// [`Fingerprint`](crate::fingerprint::Fingerprint)'s own probed memory,
294    /// which surfaces an error rather than inventing one if it, too, fails.
295    fn from_host_probe(
296        platform: Platform,
297        cpu: u32,
298        memory_gb: Result<u32, crate::StealthError>,
299    ) -> Persona {
300        let device_memory_gb = memory_gb
301            .inspect_err(|e| {
302                tracing::warn!(
303                    "host memory probe failed: {e}; leaving device_memory_gb unset \
304                     (resolved from the probed fingerprint downstream) rather than \
305                     reporting a fabricated value"
306                );
307            })
308            .ok();
309        Persona {
310            platform: Some(platform),
311            hardware_concurrency: Some(cpu),
312            device_memory_gb,
313            timezone: None,
314            locale: None,
315            seed: Some(Seed::random()),
316            ..Persona::default()
317        }
318    }
319}

321/// Minimal eval surface so stealth can probe a live page without depending on
322/// the `zendriver` core crate (which would be a dependency cycle).
323///
324/// The `zendriver` crate implements this for its `Tab`, mapping its own
325/// evaluation error into [`StealthError::Probe`].
326#[async_trait::async_trait]
327pub trait JsProbe {
328    /// Evaluate `js` in the page and return the result as a JSON value.
329    async fn eval_json(&self, js: &str) -> Result<serde_json::Value, crate::StealthError>;
330}
331
332/// JS run by [`Persona::from_browser`] to read the live browser's REAL
333/// fingerprint-relevant values (platform, memory, timezone, locale, WebGL
334/// vendor/renderer). Returns a single JSON object.
335const PROBE_JS: &str = r#"(() => {
336  const c = document.createElement('canvas').getContext('webgl');
337  const dbg = c && c.getExtension('WEBGL_debug_renderer_info');
338  return {
339    platform: navigator.platform,
340    deviceMemory: navigator.deviceMemory,
341    hardwareConcurrency: navigator.hardwareConcurrency,
342    timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
343    locale: navigator.language,
344    webglVendor: dbg ? c.getParameter(dbg.UNMASKED_VENDOR_WEBGL) : null,
345    webglRenderer: dbg ? c.getParameter(dbg.UNMASKED_RENDERER_WEBGL) : null,
346  };
347})()"#;
348
349impl Persona {
350    /// Probe the live Chrome for its REAL webgl/platform/memory/timezone
351    /// values, producing a maximally coherent host persona.
352    ///
353    /// Runs [`PROBE_JS`] through the supplied [`JsProbe`] (the `zendriver`
354    /// `Tab` implements it) and maps the resulting JSON onto a [`Persona`].
355    /// Fields the browser does not expose are left `None`.
356    pub async fn from_browser<P: JsProbe + Sync>(
357        probe: &P,
358    ) -> Result<Persona, crate::StealthError> {
359        let v = probe.eval_json(PROBE_JS).await?;
360        Ok(persona_from_probe(&v))
361    }
362}
363
364/// Map the [`PROBE_JS`] JSON result onto a [`Persona`].
365fn persona_from_probe(v: &serde_json::Value) -> Persona {
366    let mut p = Persona::default();
367
368    // `navigator.platform` strings match `Platform::js_string()`:
369    // "Win32" / "MacIntel" / "Linux x86_64" (any other → Linux fallback).
370    if let Some(plat) = v.get("platform").and_then(|x| x.as_str()) {
371        p.platform = Some(match plat {
372            "Win32" => Platform::Win32,
373            "MacIntel" => Platform::MacIntel,
374            _ => Platform::LinuxX86_64,
375        });
376    }
377
378    // `navigator.deviceMemory` is a JS number (gigabytes).
379    if let Some(mem) = v.get("deviceMemory").and_then(|x| x.as_u64()) {
380        p.device_memory_gb = Some(mem as u32);
381    }
382
383    if let Some(hc) = v.get("hardwareConcurrency").and_then(|x| x.as_u64()) {
384        p.hardware_concurrency = Some(hc as u32);
385    }
386
387    if let Some(tz) = v.get("timezone").and_then(|x| x.as_str()) {
388        p.timezone = Some(tz.to_string());
389    }
390
391    if let Some(loc) = v.get("locale").and_then(|x| x.as_str()) {
392        p.locale = Some(loc.to_string());
393    }
394
395    // WebGL vendor/renderer become a value-substitution WebglSpec when present.
396    let vendor = v
397        .get("webglVendor")
398        .and_then(|x| x.as_str())
399        .map(str::to_string);
400    let renderer = v
401        .get("webglRenderer")
402        .and_then(|x| x.as_str())
403        .map(str::to_string);
404    if vendor.is_some() || renderer.is_some() {
405        p.webgl = Some(WebglSpec {
406            strategy: None,
407            unmasked_vendor: vendor,
408            unmasked_renderer: renderer,
409        });
410    }
411
412    p
413}

456    /// Run `f` with a tracing subscriber capturing WARN and above, and return
457    /// what it logged. A failure warning nobody can observe is
458    /// indistinguishable from the silent fabrication it replaced. (Twin of the
459    /// helper in `patches.rs`'s test module — kept local rather than shared to
460    /// avoid a test-support module for two callers.)
461    fn captured_warnings(f: impl FnOnce()) -> String {
462        use std::io::Write;
463        use std::sync::{Arc, Mutex};
464
465        #[derive(Clone, Default)]
466        struct Sink(Arc<Mutex<Vec<u8>>>);
467        impl Write for Sink {
468            fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
469                self.0.lock().unwrap().extend_from_slice(buf);
470                Ok(buf.len())
471            }
472            fn flush(&mut self) -> std::io::Result<()> {
473                Ok(())
474            }
475        }
476        impl<'a> tracing_subscriber::fmt::MakeWriter<'a> for Sink {
477            type Writer = Self;
478            fn make_writer(&'a self) -> Self {
479                self.clone()
480            }
481        }
482
483        let sink = Sink::default();
484        let subscriber = tracing_subscriber::fmt()
485            .with_writer(sink.clone())
486            .with_max_level(tracing::Level::WARN)
487            .finish();
488        tracing::subscriber::with_default(subscriber, f);
489        let bytes = sink.0.lock().unwrap().clone();
490        String::from_utf8(bytes).expect("log output is utf-8")
491    }
