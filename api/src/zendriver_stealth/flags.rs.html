9/// Which GPU backend Chrome should render WebGL / WebGPU with.
10///
11/// Defaults to [`Disabled`](Self::Disabled), which reproduces zendriver's
12/// historical launch flags exactly. This is an explicit opt-in: zendriver
13/// never probes the host for a GPU and never switches backends on its own.
14///
15/// # Why the backend must be named explicitly
16///
17/// Removing `--disable-gpu` without also naming an ANGLE backend **hangs**
18/// headless Chrome (measured on darwin; see the design spec's Measurements
19/// section). The two decisions are therefore coupled and owned together here,
20/// rather than left to the caller to combine correctly.
21///
22/// ```
23/// use zendriver_stealth::GpuBackend;
24/// // Selecting a backend never probes the host — the caller decides.
25/// let flags = GpuBackend::Native.angle_flags();
26/// assert_eq!(flags[0], "--use-gl=angle");
27/// assert!(!GpuBackend::Native.allows_disable_gpu());
28/// ```
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
30#[serde(rename_all = "snake_case")]
31pub enum GpuBackend {
32    /// Today's behavior: `--disable-gpu` under headless and no ANGLE backend
33    /// forced. Chrome picks its own fallback.
34    #[default]
35    Disabled,
36    /// Force SwiftShader's CPU rasterizer. Guarantees a working WebGL context
37    /// on a host with no GPU, at the cost of a software-rasterizer
38    /// fingerprint that no real device produces.
39    SwiftShader,
40    /// Render on the host GPU through the platform's ANGLE backend (macOS →
41    /// Metal, Windows → D3D11, otherwise Vulkan).
42    ///
43    /// Gives a fully coherent GPU surface — real capabilities, real pixels,
44    /// real timings, a working `requestDevice()` — but reports **the host's**
45    /// GPU, not a chosen one. A fleet sharing one host shares one fingerprint.
46    ///
47    /// Requires a usable GPU, and the launch is **validated**: after the CDP
48    /// handshake, `zendriver` asks Chrome (`SystemInfo.getInfo`) what it
49    /// actually initialized and fails the launch with
50    /// `BrowserError::GpuBackendUnavailable` unless `gpu.featureStatus.webgl`
51    /// reports hardware acceleration. There is no automatic fallback — the
52    /// caller picks the software backend, `zendriver` never picks it for them.
53    ///
54    /// The check exists because Chrome *starting* proves nothing. On a
55    /// GPU-less Ubuntu 24 VM, Chrome 150 launched successfully under this
56    /// backend and then returned `null` from both `canvas.getContext('webgl')`
57    /// and `getContext('webgl2')` — no WebGL context at all, which is an older
58    /// and cheaper bot tell than the software-rasterizer fingerprint
59    /// [`SwiftShader`](Self::SwiftShader) produces, and one the stealth WebGL
60    /// patch cannot repair (it patches prototypes; a missing context has none).
61    /// Such a host now fails the launch instead of yielding that browser.
62    ///
63    /// A launch whose GPU cannot be *verified* — `SystemInfo.getInfo`
64    /// unavailable or answering with a status string zendriver does not know —
65    /// logs a warning and proceeds. A missing diagnostic API is not evidence
66    /// of a missing GPU.
67    Native,
68}
69
70/// ANGLE backend token for an OS name as reported by [`std::env::consts::OS`].
71///
72/// Unknown platforms take the Vulkan path rather than returning nothing —
73/// an absent backend is what causes the launch hang described on
74/// [`GpuBackend`].
75fn angle_backend_for_os(os: &str) -> &'static str {
76    match os {
77        "macos" => "metal",
78        "windows" => "d3d11",
79        _ => "vulkan",
80    }
81}

83impl GpuBackend {
84    /// Launch flags selecting this backend. Empty for
85    /// [`Disabled`](Self::Disabled).
86    #[must_use]
87    pub fn angle_flags(self) -> Vec<String> {
88        match self {
89            Self::Disabled => Vec::new(),
90            Self::SwiftShader => vec![
91                "--use-gl=angle".into(),
92                "--use-angle=swiftshader".into(),
93                // Chrome >= 116 refuses the SwiftShader fallback without this.
94                "--enable-unsafe-swiftshader".into(),
95            ],
96            Self::Native => vec![
97                "--use-gl=angle".into(),
98                format!("--use-angle={}", angle_backend_for_os(std::env::consts::OS)),
99            ],
100        }
101    }
102
103    /// Whether `--disable-gpu` may still be emitted under headless.
104    ///
105    /// False only for [`Native`](Self::Native), where the flag would defeat
106    /// the entire point of selecting a hardware backend.
107    #[must_use]
108    pub fn allows_disable_gpu(self) -> bool {
109        !matches!(self, Self::Native)
110    }
111}

113/// Flags ALL stealth profiles share (Native + Spoofed + Off-when-not-Off).
114/// Off profile uses an empty list (truly stock launch).
115///
116/// `native_isolation` selects the `--disable-features=...` entry: `false`
117/// (the default path) disables Chrome's render-process site isolation
118/// (`IsolateOrigins`/`site-per-process`) as today; `true` leaves those two
119/// features enabled — Chrome's stock site isolation stays on — while still
120/// disabling the unrelated `DisableLoadExtensionCommandLineSwitch` feature
121/// (needed for `--load-extension` regardless of the isolation choice). See
122/// [`StealthProfile::native_isolation`](crate::StealthProfile::native_isolation)
123/// for the caller-facing opt-in and its trade-off.
124fn shared_stealth_flags(native_isolation: bool) -> Vec<String> {
125    let disable_features = if native_isolation {
126        "--disable-features=DisableLoadExtensionCommandLineSwitch".to_string()
127    } else {
128        "--disable-features=IsolateOrigins,DisableLoadExtensionCommandLineSwitch,site-per-process"
129            .to_string()
130    };
131    vec![
132        "--no-first-run".into(),
133        "--no-service-autorun".into(),
134        "--no-default-browser-check".into(),
135        "--homepage=about:blank".into(),
136        "--no-pings".into(),
137        "--password-store=basic".into(),
138        "--disable-infobars".into(),
139        "--disable-breakpad".into(),
140        "--disable-component-update".into(),
141        "--disable-backgrounding-occluded-windows".into(),
142        "--disable-renderer-backgrounding".into(),
143        "--disable-background-networking".into(),
144        "--disable-dev-shm-usage".into(),
145        disable_features,
146        "--disable-session-crashed-bubble".into(),
147        "--disable-search-engine-choice-screen".into(),
148        "--remote-allow-origins=*".into(),
149        // Stop Blink injecting `navigator.webdriver` (native getter → `true`).
150        // py-zendriver sets this by default (config.py); ALL non-Off profiles
151        // need it. Previously it lived only in the Spoofed branch, so the
152        // Native profile leaked `navigator.webdriver === true` — an instant
153        // bot tell. The Spoofed JS webdriver shim also relies on this flag so
154        // Chrome's AutomationControlled hook can't re-inject over the shim.
155        "--disable-blink-features=AutomationControlled".into(),
156        // WebRTC IP-leak prevention (zendriver Python disable_webrtc=True default)
157        "--webrtc-ip-handling-policy=disable_non_proxied_udp".into(),
158        "--force-webrtc-ip-handling-policy".into(),
159    ]
160}

162/// Build the full flag list for a profile.
163///
164/// `native_isolation` is the opt-in from
165/// [`StealthProfile::native_isolation`](crate::StealthProfile::native_isolation):
166/// `false` (the default) is today's behavior, byte-identical for every
167/// [`ProfileKind`]. `true` omits the site-isolation-disabling feature names
168/// from `--disable-features=...`, leaving Chrome's real
169/// `IsolateOrigins`/`site-per-process` behavior in place.
170///
171/// `gpu_backend` is the opt-in from
172/// [`StealthProfile::gpu_backend`](crate::StealthProfile::gpu_backend).
173/// [`GpuBackend::Disabled`] (the default) reproduces today's behavior exactly:
174/// no ANGLE flags on `Native`, and the unconditional SwiftShader flags on
175/// `Spoofed` (see the `Spoofed` arm below for why those stay). An explicit
176/// backend adds/replaces the ANGLE flags on any non-Off profile; `Off`'s own
177/// flag contribution stays stock under every backend (this function never
178/// adds ANGLE flags to it). That is narrower than "the launch is unaffected"
179/// — a caller can still combine `StealthProfile::off()` with
180/// [`BrowserBuilder::gpu_backend`](https://docs.rs/zendriver/latest/zendriver/struct.BrowserBuilder.html#method.gpu_backend),
181/// which resolves independently of `ProfileKind` and adds its own ANGLE
182/// flags to the launch argv outside this function.
183#[must_use]
184pub fn flags_for_profile(
185    kind: ProfileKind,
186    native_isolation: bool,
187    gpu_backend: GpuBackend,
188) -> Vec<String> {
189    match kind {
190        // This function's own contribution for Off stays empty under every
191        // backend — `gpu_backend` never adds ANGLE flags to an Off profile's
192        // flag list. That does NOT mean an Off launch is unaffected by GPU
193        // backend selection end-to-end: the caller's `BrowserBuilder` resolves
194        // its effective backend independently of `ProfileKind` and can still
195        // add ANGLE flags / omit `--disable-gpu` for the overall launch argv.
196        ProfileKind::Off => Vec::new(),
197        ProfileKind::Native => {
198            let mut v = shared_stealth_flags(native_isolation);
199            v.extend(gpu_backend.angle_flags());
200            v
201        }
202        ProfileKind::Spoofed => {
203            let mut v = shared_stealth_flags(native_isolation);
204            // (`--disable-blink-features=AutomationControlled` lives in
205            // `shared_stealth_flags` — both Native and Spoofed need it.)
206            // A WebGL *context* must exist at all in headless — Chrome runs
207            // headless with `--headless=new --disable-gpu` (browser.rs), so
208            // with no software backend `canvas.getContext('webgl')` returns
209            // null, itself a bot tell (real browsers always have WebGL).
210            // This is about the context existing at all, not the
211            // vendor/renderer *identity* it reports (that's `patches.rs`'s
212            // job, gated by `native_webgl`), so it stays regardless of
213            // `native_isolation`. Historically that was guaranteed by
214            // unconditionally forcing SwiftShader here; that default is now
215            // expressed as `GpuBackend::Disabled` keeping the SwiftShader
216            // flags, while an explicit backend replaces them.
217            match gpu_backend {
218                GpuBackend::Disabled => {
219                    v.extend(GpuBackend::SwiftShader.angle_flags());
220                }
221                explicit => v.extend(explicit.angle_flags()),
222            }
223            v
224        }
225    }
226}
