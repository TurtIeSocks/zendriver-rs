1//! Locale list -> header/JS derivations. Always available (no feature gate);
2//! the observer formats `Accept-Language` from these.
3
4use crate::{Fingerprint, Persona};
5
6/// Format an ordered language list as an `Accept-Language` header value.
7///
8/// Index 0 carries implicit `q=1.0`; each later entry gets
9/// `q = (1.0 - 0.1*i)` clamped to `>= 0.1`, one decimal. Empty/duplicate
10/// entries are dropped (order preserved). Empty input -> `""`.
11pub fn accept_language(langs: &[String]) -> String {
12    let mut seen = std::collections::HashSet::new();
13    langs
14        .iter()
15        .filter(|l| !l.is_empty() && seen.insert(l.as_str()))
16        .enumerate()
17        .map(|(i, l)| {
18            if i == 0 {
19                l.clone()
20            } else {
21                let q = (1.0 - 0.1 * i as f64).max(0.1);
22                format!("{l};q={q:.1}")
23            }
24        })
25        .collect::<Vec<_>>()
26        .join(",")
27}

29/// What both language surfaces fall back to when nothing at all is configured.
30fn default_languages() -> Vec<String> {
31    vec!["en-US".to_string(), "en".to_string()]
32}
33
34/// Derive a language list from a single locale.
35///
36/// A region locale yields `[locale, base_lang]` where `base_lang` is the
37/// subtag before `-` (e.g. `"fr-FR"` -> `["fr-FR", "fr"]`); a bare locale (no
38/// `-`) yields a single entry.
39fn derive_from_locale(locale: &str) -> Vec<String> {
40    let base = locale.split('-').next().unwrap_or(locale);
41    if base == locale {
42        vec![locale.to_string()]
43    } else {
44        vec![locale.to_string(), base.to_string()]
45    }
46}

48/// The single resolution both locale-bearing CDP surfaces read: the language
49/// list a [`Fingerprint`] actually configures, or `None` when it configures
50/// neither `languages` nor `locale`.
51///
52/// Precedence is `languages` -> derived from `locale`, and the two callers
53/// differ only in what they do with `None`: the `Accept-Language` header
54/// substitutes `["en-US", "en"]` ([`fingerprint_languages`]), while
55/// `Emulation.setLocaleOverride` sends nothing at all
56/// ([`effective_locale`]) so Chrome keeps its own.
57///
58/// # Why `languages` outranks `locale`
59///
60/// In a real Chrome `navigator.language` is always `navigator.languages[0]`.
61/// A caller who sets both and lets them disagree is asking for a browser that
62/// cannot exist, so honoring both is not an option and the list has to win:
63/// `navigator.languages` is observable in full, and a `language` that is not
64/// its head is a mismatch no genuine browser produces. Resolving the header
65/// and the locale override independently is what produced exactly that — the
66/// header advertising one list while `Intl` reported a locale from outside it.
67/// An explicit `locale` with no `languages` still drives both surfaces.
68///
69/// This is the post-merge view. Once
70/// [`Fingerprint::overlay_persona`](crate::Fingerprint::overlay_persona) has
71/// folded a persona in, the fingerprint already carries whatever the persona
72/// pinned, so the observer reads that one value instead of re-deriving the
73/// persona-vs-fingerprint precedence at each CDP call site.
74fn configured_languages(fp: &Fingerprint) -> Option<Vec<String>> {
75    if let Some(langs) = fp.languages.as_ref().filter(|v| !v.is_empty()) {
76        return Some(langs.clone());
77    }
78    fp.locale.as_deref().map(derive_from_locale)
79}

81/// The language list to advertise, falling back to `["en-US", "en"]` when the
82/// fingerprint configures none. See [`configured_languages`].
83pub(crate) fn fingerprint_languages(fp: &Fingerprint) -> Vec<String> {
84    configured_languages(fp).unwrap_or_else(default_languages)
85}
86
87/// The locale to pin via `Emulation.setLocaleOverride`, or `None` to send no
88/// override at all.
89///
90/// By construction this is the head of [`fingerprint_languages`] whenever
91/// anything is configured, which is the coherence the two surfaces need — see
92/// [`configured_languages`] for why the list, not `locale`, decides.
93pub(crate) fn effective_locale(fp: &Fingerprint) -> Option<String> {
94    configured_languages(fp)?.into_iter().next()
95}
96
97/// Resolve the effective language list at apply time, persona first.
98///
99/// Precedence: `persona.languages` -> `fingerprint.languages` ->
100/// derive from the primary locale (`persona.locale` before `fp.locale`) ->
101/// `["en-US", "en"]`.
102///
103/// For a fingerprint the persona has already been folded into, this agrees
104/// with [`fingerprint_languages`] by construction — the persona's values are
105/// on `fp` by then. It stays persona-first for the public
106/// [`bootstrap_script`](crate::patches::bootstrap_script) path, whose caller
107/// may hand it an unmerged pair.
108pub(crate) fn resolve_languages(persona: &Persona, fp: &Fingerprint) -> Vec<String> {
109    if let Some(langs) = persona.languages.as_ref().filter(|v| !v.is_empty()) {
110        return langs.clone();
111    }
112    if let Some(langs) = fp.languages.as_ref().filter(|v| !v.is_empty()) {
113        return langs.clone();
114    }
115    match persona.locale.as_deref().or(fp.locale.as_deref()) {
116        Some(locale) => derive_from_locale(locale),
117        None => default_languages(),
118    }
119}

146    #[test]
147    fn resolve_precedence() {
148        let fp = bare_fingerprint();
149
150        // persona.languages wins
151        let p = crate::Persona {
152            languages: Some(vec!["es-ES".into(), "es".into()]),
153            ..Default::default()
154        };
155        assert_eq!(super::resolve_languages(&p, &fp), vec!["es-ES", "es"]);
156
157        // derive from locale: the fr-FR regression (was hardcoded "en")
158        let p = crate::Persona {
159            locale: Some("fr-FR".into()),
160            ..Default::default()
161        };
162        assert_eq!(super::resolve_languages(&p, &fp), vec!["fr-FR", "fr"]);
163
164        // bare locale -> single entry
165        let p = crate::Persona {
166            locale: Some("en".into()),
167            ..Default::default()
168        };
169        assert_eq!(super::resolve_languages(&p, &fp), vec!["en"]);
170
171        // nothing set -> default
172        let p = crate::Persona::default();
173        assert_eq!(super::resolve_languages(&p, &fp), vec!["en-US", "en"]);
174    }

176    #[test]
177    fn formats_q_weights() {
178        assert_eq!(accept_language(&v(&["en-US", "en"])), "en-US,en;q=0.9");
179        assert_eq!(
180            accept_language(&v(&["de-DE", "de", "en"])),
181            "de-DE,de;q=0.9,en;q=0.8"
182        );
183    }

185    #[test]
186    fn single_and_empty() {
187        assert_eq!(accept_language(&v(&["fr-FR"])), "fr-FR");
188        assert_eq!(accept_language(&[]), "");
189    }
190
191    #[test]
192    fn dedups_and_floors_q() {
193        assert_eq!(accept_language(&v(&["en", "en", "fr"])), "en,fr;q=0.9");
194        let many: Vec<String> = (0..12).map(|i| format!("l{i}")).collect();
195        let out = accept_language(&many);
196        assert!(out.ends_with("l11;q=0.1"), "q must floor at 0.1: {out}");
197    }
