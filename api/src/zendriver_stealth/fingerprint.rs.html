30/// Chromium's GREASE brand, derived from the major version.
31///
32/// Chromium builds the "fake" brand from a fixed 11-character alphabet indexed by the major
33/// version, and picks the version from a 3-element table the same way — see
34/// `GenerateBrandVersionList` in `components/embedder_support/user_agent_utils.cc`. It is
35/// deterministic per major, NOT random per boot, so a client that hardcodes one pair is
36/// identifiable the moment its major moves on.
37///
38/// Anchored on real captures: 142 -> `Not_A Brand`/`99`, 146 -> `Not-A.Brand`/`24`.
39fn grease_brand(chrome_major: u32) -> (String, &'static str) {
40    const CHARS: [&str; 11] = [" ", "(", ":", "-", ".", "/", ")", ";", "=", "?", "_"];
41    const VERSIONS: [&str; 3] = ["8", "99", "24"];
42    let c1 = CHARS[(chrome_major % 11) as usize];
43    let c2 = CHARS[((chrome_major + 1) % 11) as usize];
44    (
45        format!("Not{c1}A{c2}Brand"),
46        VERSIONS[(chrome_major % 3) as usize],
47    )
48}

50/// Chromium's brand-list permutation, also derived from the major version.
51///
52/// `GenerateBrandVersionList` shuffles the three brands through a fixed table of the six
53/// permutations of three elements, selected by `major % 6`, and it SCATTERS: the i-th input
54/// brand is written to slot `order[i]`. That direction matters — for the two 3-cycles
55/// (`major % 6` of 3 or 4) scattering and gathering give different results, and a real
56/// major-142 capture (`major % 6 == 4`) matches the scatter.
57///
58/// Returns the three brands in wire order given `[grease, chromium, branded]`.
59fn permute_brands<T>(chrome_major: u32, brands: [T; 3]) -> Vec<T> {
60    const ORDERS: [[usize; 3]; 6] = [
61        [0, 1, 2],
62        [0, 2, 1],
63        [1, 0, 2],
64        [1, 2, 0],
65        [2, 0, 1],
66        [2, 1, 0],
67    ];
68    let order = ORDERS[(chrome_major % 6) as usize];
69    let mut out: Vec<Option<T>> = vec![None, None, None];
70    for (brand, slot) in brands.into_iter().zip(order) {
71        out[slot] = Some(brand);
72    }
73    // Every slot is written exactly once: `order` is a permutation of 0..3.
74    out.into_iter()
75        .map(|b| b.expect("permutation covers every slot"))
76        .collect()
77}

79impl UserAgentMetadata {
80    /// Build a realistic UAM for the given platform + Chrome major version.
81    ///
82    /// Three brands, as Chrome sends them: a GREASE brand, `Chromium;v=N` and
83    /// `Google Chrome;v=N`. Both the GREASE pair and the list ORDER are derived from
84    /// `chrome_major` — see [`grease_brand`] and [`permute_brands`].
85    ///
86    /// This used to hardcode `Not_A Brand;v=8` in a fixed GREASE-first order. That pair and
87    /// that order are what Chrome **120** sends (120 % 11 == 10, 120 % 3 == 0, 120 % 6 == 0),
88    /// so every other major presented a `sec-ch-ua` no Chrome has ever sent — checkable
89    /// against a static table with no device knowledge at all.
90    pub fn realistic(platform: Platform, chrome_major: u32, chrome_full: &str) -> Self {
91        let (grease, grease_version) = grease_brand(chrome_major);
92        let brands = permute_brands(
93            chrome_major,
94            [
95                Brand {
96                    brand: grease.clone(),
97                    version: grease_version.into(),
98                },
99                Brand {
100                    brand: "Chromium".into(),
101                    version: chrome_major.to_string(),
102                },
103                Brand {
104                    brand: "Google Chrome".into(),
105                    version: chrome_major.to_string(),
106                },
107            ],
108        );
109        let full_version_list = permute_brands(
110            chrome_major,
111            [
112                Brand {
113                    brand: grease,
114                    version: format!("{grease_version}.0.0.0"),
115                },
116                Brand {
117                    brand: "Chromium".into(),
118                    version: chrome_full.to_string(),
119                },
120                Brand {
121                    brand: "Google Chrome".into(),
122                    version: chrome_full.to_string(),
123                },
124            ],
125        );
126        let (platform_version, architecture, bitness) = match platform {
127            Platform::Win32 => ("15.0.0", "x86", "64"),
128            Platform::MacIntel => ("10.15.7", "x86", "64"),
129            Platform::LinuxX86_64 => ("5.15.0", "x86", "64"),
130        };
131        Self {
132            brands,
133            full_version_list,
134            platform: platform.ch_platform().to_string(),
135            platform_version: platform_version.to_string(),
136            architecture: architecture.to_string(),
137            bitness: bitness.to_string(),
138            wow64: false,
139            mobile: false,
140            model: String::new(),
141        }
142    }
143}

156/// Probed system + Chrome facts used to compose stealth values.
157#[derive(Debug, Clone, Serialize)]
158pub struct Fingerprint {
159    pub platform: Platform,
160    pub chrome_major: u32,
161    pub chrome_full: String,
162    pub cpu_count: u32,
163    pub memory_gb: u32,
164    pub ua_string: String,
165    pub ua_metadata: UserAgentMetadata,
166    /// IANA timezone driving `Emulation.setTimezoneOverride`, from
167    /// [`StealthProfile::timezone`](crate::StealthProfile::timezone) or from a
168    /// [`Persona`](crate::Persona), which wins and is folded in when the
169    /// observer is built. `None` sends no override.
170    pub timezone: Option<String>,
171    /// Locale driving `Emulation.setLocaleOverride` when
172    /// [`languages`](Self::languages) is unset. Same two sources as
173    /// [`timezone`](Self::timezone).
174    pub locale: Option<String>,
175    /// Ordered language list driving `Accept-Language`, `navigator.languages`
176    /// and — from its first entry — `Emulation.setLocaleOverride`, which is
177    /// what keeps the JS-visible locale inside the list the header advertises.
178    /// Falls back to [`locale`](Self::locale) when unset, and the header falls
179    /// back further to `["en-US", "en"]` when that is unset too (the locale
180    /// override is then simply not sent).
181    pub languages: Option<Vec<String>>,
182    /// Screen / device-metrics override resolved from
183    /// [`StealthProfile::screen`](crate::StealthProfile::screen), or from a
184    /// [`Persona`](crate::Persona)'s own `screen`, which wins. `None` by
185    /// default (`auto_detect` never probes a screen size) — the observer's
186    /// fixed 1920x1080 default is untouched until this is explicitly set.
187    pub screen: Option<crate::persona::specs::ScreenSpec>,
188}

190impl Fingerprint {
191    /// Probe host system + installed Chrome to build a realistic fingerprint.
192    // `StealthError` is large because `PatchFailed` wraps `CallError` (~152B).
193    // Boxing it would cross the Task 5 file scope; bypass per-fn instead.
194    #[allow(clippy::result_large_err)]
195    pub fn auto_detect(chrome_executable: &Path) -> Result<Self, StealthError> {
196        let platform = detect_platform();
197        let (chrome_major, chrome_full) =
198            probe_chrome_version(chrome_executable).unwrap_or_else(|e| {
199                tracing::warn!("chrome version probe failed: {e}; using fallback");
200                (FALLBACK_CHROME_MAJOR, FALLBACK_CHROME_FULL.to_string())
201            });
202        let cpu_count = clamp_cpu_count(num_cpus::get() as u32);
203        let memory_gb = detect_memory_gb()?;
204        let ua_string = crate::ua::compose_ua_string(platform, &chrome_full);
205        let ua_metadata = UserAgentMetadata::realistic(platform, chrome_major, &chrome_full);
206        Ok(Self {
207            platform,
208            chrome_major,
209            chrome_full,
210            cpu_count,
211            memory_gb,
212            ua_string,
213            ua_metadata,
214            timezone: None,
215            locale: None,
216            languages: None,
217            screen: None,
218        })
219    }
220
221    /// Fold a [`Persona`](crate::Persona)'s explicitly-set fields into this
222    /// fingerprint.
223    ///
224    /// Four axes exist on both types — `timezone`, `locale`, `languages`,
225    /// `screen` — and three separate consumers read them: the CDP
226    /// `Emulation.set*Override` calls, the `Accept-Language` header, and the
227    /// JS patches. Merging once, at the single point where a persona and a
228    /// fingerprint meet
229    /// ([`StealthObserver::with_persona`](crate::StealthObserver::with_persona)),
230    /// is what keeps those consumers from disagreeing: each reads the merged
231    /// value rather than re-deriving the precedence for itself.
232    ///
233    /// Precedence: an explicitly-set persona field wins; `None` inherits
234    /// whatever the fingerprint already resolved — from
235    /// [`StealthProfile`](crate::StealthProfile)'s per-field setters, or from
236    /// the host probe. A [`Persona::default`](crate::Persona::default) is a
237    /// no-op, and the merge is idempotent.
238    ///
239    /// An empty `languages` list counts as unset, matching how every consumer
240    /// in [`lang`](crate::lang) already reads one: `Some(vec![])` describes a
241    /// persona that pins no languages, not one that advertises none, and
242    /// letting it overwrite would make an empty persona field destroy a
243    /// configured [`StealthProfile::languages`](crate::StealthProfile::languages).
244    pub(crate) fn overlay_persona(&mut self, persona: &crate::Persona) {
245        if let Some(timezone) = &persona.timezone {
246            self.timezone = Some(timezone.clone());
247        }
248        if let Some(locale) = &persona.locale {
249            self.locale = Some(locale.clone());
250        }
251        if let Some(languages) = persona.languages.as_ref().filter(|v| !v.is_empty()) {
252            self.languages = Some(languages.clone());
253        }
254        if let Some(screen) = persona.screen {
255            self.screen = Some(screen);
256        }
257    }
258
259    /// Recompose UA string + UAM after platform/version overrides.
260    pub fn recompose(&mut self) {
261        self.ua_string = crate::ua::compose_ua_string(self.platform, &self.chrome_full);
262        self.ua_metadata =
263            UserAgentMetadata::realistic(self.platform, self.chrome_major, &self.chrome_full);
264    }
265}

292/// Parse `Google Chrome 120.0.6099.234` / `Chromium 120.0.6099.0` into
293/// `(major, full)`. Shared by the Unix probe and its tests.
294#[cfg(not(windows))]
295#[allow(clippy::result_large_err)]
296fn parse_version_banner(stdout: &str) -> Result<(u32, String), StealthError> {
297    let full = stdout
298        .split_whitespace()
299        .find(|tok| tok.chars().next().is_some_and(|c| c.is_ascii_digit()))
300        .ok_or_else(|| StealthError::ChromeVersionDetect(format!("no version token in: {stdout}")))?
301        .to_string();
302    let major: u32 = full
303        .split('.')
304        .next()
305        .and_then(|s| s.parse().ok())
306        .ok_or_else(|| StealthError::ChromeVersionDetect(format!("bad major in: {full}")))?;
307    Ok((major, full))
308}

310/// Probe the Chrome version by running `chrome --version`.
311///
312/// Unix-only *by design*. Chrome on Windows does not implement `--version` as a
313/// print-and-exit: with no existing browser session it starts a whole browser
314/// and never exits, so reading its output blocks forever. See the `cfg(windows)`
315/// twin below for what Windows does instead, and why it must not exec.
316///
317/// Bounded even here. `--version` returns in milliseconds on a healthy Unix
318/// Chrome, but this runs *synchronously inside* `Browser::launch()`'s future:
319/// an unbounded wait would wedge the poll, and a future that never yields
320/// starves the tokio timer driver — which silently disables every
321/// `tokio::time::timeout` in the process, including the ones meant to catch
322/// exactly this. A blocking call on the async path must carry its own deadline,
323/// because no outer timeout can rescue it.
324#[cfg(not(windows))]
325#[allow(clippy::result_large_err)]
326pub(crate) fn probe_chrome_version(exe: &Path) -> Result<(u32, String), StealthError> {
327    use std::io::Read;
328    use std::process::Stdio;
329    use std::time::{Duration, Instant};
330
331    /// Generous: a healthy `chrome --version` answers in ~50ms. This is a
332    /// backstop against a wedged binary, not a performance budget.
333    const PROBE_TIMEOUT: Duration = Duration::from_secs(10);
334    const POLL_INTERVAL: Duration = Duration::from_millis(20);
335
336    let mut child = Command::new(exe)
337        .arg("--version")
338        .stdin(Stdio::null())
339        .stdout(Stdio::piped())
340        .stderr(Stdio::null())
341        .spawn()
342        .map_err(|e| StealthError::ChromeVersionDetect(format!("spawn failed: {e}")))?;
343
344    let deadline = Instant::now() + PROBE_TIMEOUT;
345    let status = loop {
346        match child.try_wait() {
347            Ok(Some(status)) => break status,
348            Ok(None) => {
349                if Instant::now() >= deadline {
350                    // Reap rather than leak: kill_on_drop is not in play here.
351                    let _ = child.kill();
352                    let _ = child.wait();
353                    return Err(StealthError::ChromeVersionDetect(format!(
354                        "`--version` did not exit within {PROBE_TIMEOUT:?}"
355                    )));
356                }
357                std::thread::sleep(POLL_INTERVAL);
358            }
359            Err(e) => {
360                return Err(StealthError::ChromeVersionDetect(format!(
361                    "wait failed: {e}"
362                )));
363            }
364        }
365    };
366    if !status.success() {
367        return Err(StealthError::ChromeVersionDetect(format!(
368            "exit {:?}",
369            status.code()
370        )));
371    }
372
373    let mut stdout = String::new();
374    child
375        .stdout
376        .take()
377        .ok_or_else(|| StealthError::ChromeVersionDetect("no stdout pipe".to_string()))?
378        .read_to_string(&mut stdout)
379        .map_err(|e| StealthError::ChromeVersionDetect(format!("read stdout: {e}")))?;
380    parse_version_banner(&stdout)
381}

383/// Probe the Chrome version by reading the binary's own PE version resource.
384///
385/// **Never exec Chrome to ask its version on Windows.** `chrome.exe --version`
386/// is not the Unix print-and-exit: with no existing browser session to hand off
387/// to, it launches a full browser (GPU + network service + crashpad children)
388/// and never exits. `Command::output()` waits for the child to exit *and* for
389/// its stdout/stderr to hit EOF, so it blocks forever.
390///
391/// That block is what made this pathological rather than merely slow. The call
392/// sits synchronously inside `Browser::launch()`'s future, so the task never
393/// yields; `tokio::time::timeout` can only cancel at an await point, leaving
394/// every guard in the launch path — `WS_ENDPOINT_TIMEOUT` (15s),
395/// `HANDSHAKE_TIMEOUT` (30s) — structurally unable to fire. The result was an
396/// infinite, *silent* hang: on windows-latest CI all three
397/// `prerelease_verification` tests timed out at 360s having emitted nothing.
398///
399/// It hides on dev machines because a developer usually has Chrome already
400/// open, and `--version` then hands off to the running instance
401/// ("Opening in existing browser session.") and exits in milliseconds. It only
402/// bites where no session exists — i.e. exactly on CI.
403///
404/// A version is a static property of the file, so read it from the file. This
405/// parses the documented `VS_FIXEDFILEINFO` block, which keeps the crate's
406/// `unsafe_code = "deny"` intact and adds no dependency.
407#[cfg(windows)]
408#[allow(clippy::result_large_err)]
409pub(crate) fn probe_chrome_version(exe: &Path) -> Result<(u32, String), StealthError> {
410    let bytes = std::fs::read(exe)
411        .map_err(|e| StealthError::ChromeVersionDetect(format!("read {}: {e}", exe.display())))?;
412    parse_pe_file_version(&bytes).ok_or_else(|| {
413        StealthError::ChromeVersionDetect(format!(
414            "no VS_FIXEDFILEINFO version resource in {}",
415            exe.display()
416        ))
417    })
418}

420/// Find the first occurrence of `needle` in `haystack`.
421#[cfg(windows)]
422fn find_bytes(haystack: &[u8], needle: &[u8]) -> Option<usize> {
423    haystack.windows(needle.len()).position(|w| w == needle)
424}

426/// Extract `(major, "a.b.c.d")` from a PE image's `VS_FIXEDFILEINFO`.
427///
428/// Anchors on the UTF-16LE `VS_VERSION_INFO` key, then reads the `0xFEEF04BD`
429/// signature that opens the fixed-info block a few bytes later. Layout
430/// (little-endian): `+0` signature, `+8` dwFileVersionMS, `+12` dwFileVersionLS,
431/// each packing two 16-bit fields as `high.low`.
432///
433/// Both halves of that search are load-bearing, and real `chrome.exe` punishes
434/// getting either wrong:
435///
436/// - **Anchor on the key, not the signature.** The signature is only four bytes
437///   and does occur by chance in a multi-megabyte image: Chrome 150 carries one
438///   at file offset ~2.79M, ~475KB *before* the resource, which decodes to
439///   nonsense (`9340.36168.8075.18720`).
440/// - **Try every key match, not just the first.** Chrome's `.rdata` contains the
441///   literal text `VS_VERSION_INFO` — error strings from Chrome's *own* resource
442///   parser ("unexpected VS_VERSIONINFO in ") — ~645KB before the genuine
443///   resource. The first match is therefore a decoy with no signature after it;
444///   only a later one is real. Scanning all matches and keeping the first that
445///   is actually followed by a signature tolerates any number of such decoys.
446#[cfg(windows)]
447fn parse_pe_file_version(bytes: &[u8]) -> Option<(u32, String)> {
448    const SIGNATURE: [u8; 4] = 0xFEEF_04BDu32.to_le_bytes();
449    /// The signature sits just past the key + NUL + 32-bit alignment padding
450    /// (34 bytes in practice). Bounding the search is what makes a decoy match
451    /// fail fast and fall through to the next candidate.
452    const SIG_SEARCH_WINDOW: usize = 64;
453
454    let key: Vec<u8> = "VS_VERSION_INFO"
455        .encode_utf16()
456        .flat_map(u16::to_le_bytes)
457        .collect();
458
459    let mut search_from = 0usize;
460    while let Some(rel) = find_bytes(bytes.get(search_from..)?, &key) {
461        let anchor = search_from + rel + key.len();
462        let window_end = anchor.saturating_add(SIG_SEARCH_WINDOW).min(bytes.len());
463
464        if let Some(sig) = bytes
465            .get(anchor..window_end)
466            .and_then(|w| find_bytes(w, &SIGNATURE))
467            .map(|sig_rel| anchor + sig_rel)
468        {
469            let word = |off: usize| -> Option<u32> {
470                Some(u32::from_le_bytes(
471                    bytes.get(sig + off..sig + off + 4)?.try_into().ok()?,
472                ))
473            };
474            let ms = word(8)?;
475            let ls = word(12)?;
476            let major = ms >> 16;
477            // A zero major means we locked onto something that is not a version
478            // block; keep looking rather than reporting `0.x.y.z`.
479            if major != 0 {
480                let full = format!("{}.{}.{}.{}", major, ms & 0xFFFF, ls >> 16, ls & 0xFFFF);
481                return Some((major, full));
482            }
483        }
484        search_from += rel + 1;
485    }
486    None
487}

493/// Detect total RAM in GB, clamped to the spec-compliant values
494/// for `navigator.deviceMemory` (capped at 8 per W3C; floor at 4 for plausibility).
495#[allow(clippy::result_large_err)]
496pub(crate) fn detect_memory_gb() -> Result<u32, StealthError> {
497    let mut sys = sysinfo::System::new();
498    sys.refresh_memory();
499    // sysinfo 0.32: total_memory() returns BYTES, not KiB. Verified against
500    // sysinfo-0.32.1/src/common/system.rs::total_memory doc comment
501    // ("Returns the RAM size in bytes.").
502    let total_bytes = sys.total_memory();
503    if total_bytes == 0 {
504        return Err(StealthError::SystemInfo("total_memory returned 0".into()));
505    }
506    let total_gb = (total_bytes / 1_073_741_824) as u32;
507    Ok(round_to_navigator_memory(total_gb))
508}
509
510fn round_to_navigator_memory(gb: u32) -> u32 {
511    // navigator.deviceMemory spec values: 0.25, 0.5, 1, 2, 4, 8. Cap at 8.
512    // We floor at 4 for plausibility (sub-4GB consumer desktops are extinct).
513    if gb >= 8 { 8 } else { 4 }
514}

627    /// Locks the `VS_FIXEDFILEINFO` field packing (`high.low` per dword) against
628    /// a synthetic resource, so the bit-twiddling is verified without depending
629    /// on whatever Chrome happens to be installed on the test host.
630    #[cfg(windows)]
631    #[test]
632    fn parses_fixed_file_info_field_packing() {
633        let mut buf: Vec<u8> = Vec::new();
634        buf.extend_from_slice(&[0xAA; 128]); // leading noise
635        buf.extend("VS_VERSION_INFO".encode_utf16().flat_map(u16::to_le_bytes));
636        buf.extend_from_slice(&[0x00, 0x00]); // NUL + padding to the signature
637        buf.extend_from_slice(&0xFEEF_04BDu32.to_le_bytes()); // dwSignature
638        buf.extend_from_slice(&0x0001_0000u32.to_le_bytes()); // dwStrucVersion
639        // Each dword packs `high.low`: MS = 150.0, LS = 7871.114.
640        buf.extend_from_slice(&(150u32 << 16).to_le_bytes());
641        buf.extend_from_slice(&((7871u32 << 16) | 114).to_le_bytes());
642
643        assert_eq!(
644            parse_pe_file_version(&buf),
645            Some((150, "150.0.7871.114".to_string()))
646        );
647    }

655    /// Real `chrome.exe` carries the literal `VS_VERSION_INFO` in `.rdata`
656    /// (error text from Chrome's own resource parser) ~645KB before the genuine
657    /// resource. Anchoring on the *first* match alone finds no signature and
658    /// silently falls back to the baked-in version — which is precisely the
659    /// wrong-version bug this probe exists to fix. Every match must be tried.
660    #[cfg(windows)]
661    #[test]
662    fn pe_version_parse_skips_decoy_key_without_signature() {
663        let key: Vec<u8> = "VS_VERSION_INFO"
664            .encode_utf16()
665            .flat_map(u16::to_le_bytes)
666            .collect();
667
668        let mut buf: Vec<u8> = Vec::new();
669        // Decoy #1: the key as plain string data, followed by prose rather than
670        // a signature (mirrors Chrome's "unexpected VS_VERSIONINFO in " text).
671        buf.extend_from_slice(&key);
672        buf.extend_from_slice(b"\0\0unexpected VS_VERSIONINFO in \0");
673        buf.extend_from_slice(&[0xCC; 96]);
674        // The genuine resource.
675        buf.extend_from_slice(&key);
676        buf.extend_from_slice(&[0x00, 0x00, 0x00, 0x00]); // NUL + alignment padding
677        buf.extend_from_slice(&0xFEEF_04BDu32.to_le_bytes());
678        buf.extend_from_slice(&0x0001_0000u32.to_le_bytes());
679        buf.extend_from_slice(&(150u32 << 16).to_le_bytes());
680        buf.extend_from_slice(&((7871u32 << 16) | 114).to_le_bytes());
681
682        assert_eq!(
683            parse_pe_file_version(&buf),
684            Some((150, "150.0.7871.114".to_string())),
685            "must skip the decoy key and find the real resource behind it"
686        );
687    }
