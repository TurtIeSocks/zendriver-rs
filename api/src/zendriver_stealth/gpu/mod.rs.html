17/// One GPU's identity and static capabilities, fully resolved.
18///
19/// Produced by flattening the shared base table, the tier's overrides, the
20/// device row, and any caller-supplied spec. Callers only ever see this
21/// flattened form, so the internal base/override split can change without
22/// breaking anyone.
23///
24/// The parameter maps hold *device capabilities* only — the values that are
25/// implementation-fixed and differ per GPU. Per-context mutable state
26/// (`VIEWPORT`, `BLEND`, the `STENCIL_*`/`PACK_*`/`UNPACK_*` families, ...)
27/// is deliberately absent so it delegates to the real backend; freezing it
28/// would contradict the page's own GL calls. A caller may still pin such a
29/// key through [`overlay`](Self::overlay) — nothing here forbids it — but the
30/// shipped tiers never do.
31///
32/// Derives `Serialize`/`Deserialize` because [`Persona`](crate::Persona) does
33/// (it round-trips through JSON as `browser_fingerprint_generate`'s return
34/// value and in `Persona::overlay`'s own tests) — every field type here
35/// already supported serde before this field existed, so this is just wiring
36/// the derive through, not adding new capability.
37#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
38pub struct GpuProfile {
39    pub provenance: Provenance,
40    pub params_webgl1: BTreeMap<String, GlParam>,
41    pub params_webgl2: BTreeMap<String, GlParam>,
42    pub precision: BTreeMap<String, ShaderPrecision>,
43    pub extensions_webgl1: Vec<String>,
44    pub extensions_webgl2: Vec<String>,
45    pub unmasked_vendor: String,
46    pub unmasked_renderer: String,
47}
48
49impl GpuProfile {
50    /// An all-empty profile, used as an overlay base.
51    #[must_use]
52    pub fn empty() -> Self {
53        Self {
54            provenance: Provenance::Derived {
55                source: "empty".into(),
56            },
57            params_webgl1: BTreeMap::new(),
58            params_webgl2: BTreeMap::new(),
59            precision: BTreeMap::new(),
60            extensions_webgl1: Vec::new(),
61            extensions_webgl2: Vec::new(),
62            unmasked_vendor: String::new(),
63            unmasked_renderer: String::new(),
64        }
65    }
66
67    /// Merge `over` on top of `self`.
68    ///
69    /// The maps merge key-wise rather than whole-field, so a caller supplying a
70    /// partial profile overrides only the keys it sets and cannot silently wipe
71    /// entries it said nothing about. The lists and strings replace wholesale,
72    /// but only when non-empty, so an empty `over` is a no-op rather than a
73    /// clear.
74    #[must_use]
75    pub fn overlay(mut self, over: GpuProfile) -> GpuProfile {
76        self.params_webgl1.extend(over.params_webgl1);
77        self.params_webgl2.extend(over.params_webgl2);
78        self.precision.extend(over.precision);
79        if !over.extensions_webgl1.is_empty() {
80            self.extensions_webgl1 = over.extensions_webgl1;
81        }
82        if !over.extensions_webgl2.is_empty() {
83            self.extensions_webgl2 = over.extensions_webgl2;
84        }
85        if !over.unmasked_vendor.is_empty() {
86            self.unmasked_vendor = over.unmasked_vendor;
87        }
88        if !over.unmasked_renderer.is_empty() {
89            self.unmasked_renderer = over.unmasked_renderer;
90        }
91        self
92    }
93}

106fn lookup<'a, V>(table: &'a [(&str, V)], key: &str) -> Option<&'a V> {
107    table.iter().find(|(k, _)| *k == key).map(|(_, v)| v)
108}

110/// Merge one context version's base table with a tier's overrides.
111///
112/// The two context versions are kept apart deliberately: a WebGL1 context
113/// serves 18 capabilities and a WebGL2 context 47, and serving the WebGL2 set
114/// to a WebGL1 context would answer enums that context has no constant for,
115/// where real Chrome returns `null` and raises `INVALID_ENUM`.
116///
117/// Those counts are the *static device capabilities* out of the 82 and up-to-132
118/// parameters the captures enumerate. The remainder is per-context mutable
119/// state and is deliberately absent from the tables, so it delegates to the
120/// real backend — see `gpu-tier-gen`'s `SERVED_CAPS` for the rule and
121/// `DELEGATED_PARAMS` for why each excluded name is excluded.
122fn flatten(
123    base: &[(&str, GlParamRef)],
124    overrides: &[(&str, &[(&str, GlParamRef)])],
125    key: &str,
126) -> BTreeMap<String, GlParam> {
127    let mut out: BTreeMap<String, GlParam> = base
128        .iter()
129        .map(|(k, v)| ((*k).to_string(), v.to_owned_param()))
130        .collect();
131    if let Some(over) = lookup(overrides, key) {
132        for (k, v) in *over {
133            out.insert((*k).to_string(), v.to_owned_param());
134        }
135    }
136    out
137}

139/// Flatten the base tables plus one tier's overrides into a profile.
140pub(crate) fn profile_for_tier(tier: Tier) -> GpuProfile {
141    let key = tier_key(tier);
142    let params = flatten(
143        tiers::BASE_PARAMS_WEBGL2,
144        tiers::PARAM_OVERRIDES_WEBGL2,
145        key,
146    );
147    let precision = lookup(tiers::PRECISION, key)
148        .map(|rows| {
149            rows.iter()
150                .map(|(k, [a, b, c])| {
151                    (
152                        (*k).to_string(),
153                        ShaderPrecision {
154                            range_min: *a,
155                            range_max: *b,
156                            precision: *c,
157                        },
158                    )
159                })
160                .collect()
161        })
162        .unwrap_or_default();
163    let exts = |t: &[(&str, &[&str])]| -> Vec<String> {
164        lookup(t, key)
165            .map(|l| l.iter().map(|s| (*s).to_string()).collect())
166            .unwrap_or_default()
167    };
168    GpuProfile {
169        provenance: Provenance::Probed {
170            chrome: "see data/gpu-tiers".into(),
171            os: "see data/gpu-tiers".into(),
172        },
173        params_webgl1: flatten(
174            tiers::BASE_PARAMS_WEBGL1,
175            tiers::PARAM_OVERRIDES_WEBGL1,
176            key,
177        ),
178        params_webgl2: params,
179        precision,
180        extensions_webgl1: exts(tiers::EXTENSIONS_WEBGL1),
181        extensions_webgl2: exts(tiers::EXTENSIONS_WEBGL2),
182        unmasked_vendor: String::new(),
183        unmasked_renderer: String::new(),
184    }
185}

187/// One tier's measured WebGPU adapter capabilities, owned so a caller's
188/// [`WebgpuSpec`](crate::WebgpuSpec) can overlay onto them.
189///
190/// Deliberately **not** a field on [`GpuProfile`]. That type is public with
191/// public fields, and `WebgpuSpec` already owns the caller-facing WebGPU
192/// override path — a second one would give the same value two places to be set
193/// from, and would make the shape of an internal table part of the public API.
194#[derive(Debug, Clone, PartialEq, Eq)]
195pub(crate) struct WebgpuAdapter {
196    /// `GPUSupportedLimits`, keyed as the WebGPU IDL spells them
197    /// (`maxTextureDimension2D`, `maxBufferSize`, ...).
198    pub limits: BTreeMap<String, u64>,
199    /// `GPUSupportedFeatures`, in Chrome's own iteration order.
200    pub features: Vec<String>,
201}
202
203/// The WebGPU adapter capabilities a tier measured, or `None` when that tier's
204/// machine has no adapter at all.
205///
206/// The sibling of [`profile_for_tier`] for the other GPU API: both read one
207/// tier key out of the generated tables, so a renderer that resolves a tier
208/// gets that tier's answer on both surfaces and cannot describe two devices.
209///
210/// **`None` is data, not a gap.** Real SwiftShader Chrome resolves
211/// `requestAdapter()` to null, so that tier has no adapter to describe, and the
212/// capture records exactly that. Callers must leave the host's adapter alone
213/// rather than substituting a neighbouring tier's numbers — a persona claiming
214/// a software rasterizer to WebGL while reporting a hardware adapter's limits
215/// to WebGPU is the cross-API contradiction this resolution exists to prevent.
216pub(crate) fn webgpu_for_tier(tier: Tier) -> Option<WebgpuAdapter> {
217    let measured = lookup(tiers::WEBGPU_ADAPTERS, tier_key(tier))?.as_ref()?;
218    Some(WebgpuAdapter {
219        limits: measured
220            .limits
221            .iter()
222            .map(|(k, v)| ((*k).to_string(), *v))
223            .collect(),
224        features: measured.features.iter().map(|f| (*f).to_string()).collect(),
225    })
226}

228/// Enums the patch must resolve that the generated [`tiers::ENUM_NAMES`] table
229/// does not carry, because the probe only walks the *core* `getParameter`
230/// enums:
231///
232/// - the two `WEBGL_debug_renderer_info` names, which exist only once that
233///   extension has been fetched, and
234/// - the shader-type and precision-type arguments `getShaderPrecisionFormat`
235///   takes, which are never passed to `getParameter` at all.
236///
237/// Both reach the page through the same number-to-name lookup as every other
238/// value, so leaving them out does not merely omit them — it makes the JS fall
239/// through to the real backend, serving the host's own unmasked pair and its
240/// own shader precision. All ten are fixed by the WebGL spec, so they are
241/// constants here rather than measurements.
242const EXTRA_ENUMS: &[(u32, &str)] = &[
243    (35632, "FRAGMENT_SHADER"),
244    (35633, "VERTEX_SHADER"),
245    (36336, "LOW_FLOAT"),
246    (36337, "MEDIUM_FLOAT"),
247    (36338, "HIGH_FLOAT"),
248    (36339, "LOW_INT"),
249    (36340, "MEDIUM_INT"),
250    (36341, "HIGH_INT"),
251    (37445, "UNMASKED_VENDOR_WEBGL"),
252    (37446, "UNMASKED_RENDERER_WEBGL"),
253];
254
255/// `DRAW_BUFFER0`, which ES 3.0 fixes at `0x8825`, with `DRAW_BUFFER1` through
256/// `DRAW_BUFFER15` following contiguously.
257const DRAW_BUFFER0: u32 = 34853;
258
259/// How many `DRAW_BUFFERn` enums the ES 3.0 spec defines.
260const DRAW_BUFFER_COUNT: u32 = 16;
261
262/// Name every `DRAW_BUFFERn` enum, whether or not a capture recorded it.
263///
264/// A capture only records the indices its own backend answered, so the
265/// SwiftShader capture stops at `DRAW_BUFFER5` (that backend has 6). Taking the
266/// names from captures alone would then leave a SwiftShader persona unable to
267/// recognise `DRAW_BUFFER6`, and `webgl.js` cannot suppress an enum it cannot
268/// name: the read would fall through to a host that has 8 and answer beside a
269/// cap saying 6 exist.
270///
271/// These are spec constants rather than measurements, the same category as
272/// [`EXTRA_ENUMS`] above, so the full range is declared here instead of being
273/// inferred from whatever hardware happened to be probed.
274fn draw_buffer_enums() -> impl Iterator<Item = (u32, String)> {
275    (0..DRAW_BUFFER_COUNT).map(|i| (DRAW_BUFFER0 + i, format!("DRAW_BUFFER{i}")))
276}

278/// Serialize a profile into the JSON object `webgl.js` consumes.
279///
280/// Each value carries a `t` tag naming its GL type so the JS side builds the
281/// right typed array; JSON alone cannot distinguish `Int32Array` from
282/// `Float32Array`.
283///
284/// The unmasked vendor/renderer strings are folded into both parameter maps
285/// (and their enums into `enumNames`, via [`EXTRA_ENUMS`]) rather than shipped
286/// as their own fields: a page reads them through `getParameter` exactly like
287/// every other value, so giving them a second path through the JS would buy
288/// nothing.
289pub(crate) fn profile_to_js(p: &GpuProfile) -> String {
290    use serde_json::{Map, Value, json};
291
292    fn val(v: &GlParam) -> Value {
293        match v {
294            GlParam::Int(i) => json!({"t": "i", "v": i}),
295            GlParam::Float(f) => json!({"t": "f", "v": f}),
296            GlParam::Bool(b) => json!({"t": "b", "v": b}),
297            GlParam::Str(s) => json!({"t": "s", "v": s}),
298            GlParam::IntPair(a) => json!({"t": "i32pair", "v": a}),
299            GlParam::IntQuad(a) => json!({"t": "i32quad", "v": a}),
300            GlParam::FloatPair(a) => json!({"t": "f32pair", "v": a}),
301            GlParam::FloatQuad(a) => json!({"t": "f32quad", "v": a}),
302            GlParam::IntList(a) => json!({"t": "u32list", "v": a}),
303        }
304    }
305    let conv = |m: &BTreeMap<String, GlParam>| -> Map<String, Value> {
306        let mut out: Map<String, Value> = m.iter().map(|(k, v)| (k.clone(), val(v))).collect();
307        out.insert(
308            "UNMASKED_VENDOR_WEBGL".to_string(),
309            json!({"t": "s", "v": p.unmasked_vendor}),
310        );
311        out.insert(
312            "UNMASKED_RENDERER_WEBGL".to_string(),
313            json!({"t": "s", "v": p.unmasked_renderer}),
314        );
315        out
316    };
317
318    let mut enums = enum_names();
319    for (num, name) in EXTRA_ENUMS {
320        enums.insert(num.to_string(), Value::from(*name));
321    }
322    for (num, name) in draw_buffer_enums() {
323        enums.insert(num.to_string(), Value::from(name));
324    }
325
326    json!({
327        "params1": conv(&p.params_webgl1),
328        "params2": conv(&p.params_webgl2),
329        "precision": p.precision.iter().map(|(k, v)| {
330            (k.clone(), json!([v.range_min, v.range_max, v.precision]))
331        }).collect::<Map<_, _>>(),
332        "extensions1": p.extensions_webgl1,
333        "extensions2": p.extensions_webgl2,
334        "enumNames": enums,
335        "inertStubs": devices::inert_stubs(),
336    })
337    .to_string()
338}

340/// Numeric GL enum to parameter name, as the JS side indexes it.
341///
342/// A JS object is keyed by number, so aliases collapse: `BLEND_EQUATION` and
343/// `BLEND_EQUATION_RGB` are both enum `32777`, and only one name survives into
344/// the emitted object. That is safe **only while aliased names carry equal
345/// values**, which is asserted below rather than assumed — if a future capture
346/// ever gives an aliased pair different values, silently keeping one would
347/// serve the wrong number for the other.
348///
349/// The comparison runs over each tier's *resolved* maps, both context
350/// versions, because that is what the page actually reads. Consulting the
351/// shared WebGL2 base alone would pass a tier override that moved one spelling
352/// of a collapsed pair and not the other, and would never look at WebGL1 at
353/// all. A present/absent split fails too: whichever spelling survives has to
354/// be the one the table carries, or the lookup misses and the enum falls
355/// through to the real backend.
356fn enum_names() -> serde_json::Map<String, serde_json::Value> {
357    let resolved: Vec<(Tier, GpuProfile)> = Tier::ALL
358        .iter()
359        .map(|tier| (*tier, profile_for_tier(*tier)))
360        .collect();
361    let mut out = serde_json::Map::new();
362    let mut chosen: BTreeMap<u32, &str> = BTreeMap::new();
363    for (num, name) in tiers::ENUM_NAMES {
364        if let Some(prev) = chosen.insert(*num, *name) {
365            // Both spellings must resolve to the same value, or collapsing
366            // them changes what the page reads.
367            for (tier, p) in &resolved {
368                for (version, params) in
369                    [("WebGL1", &p.params_webgl1), ("WebGL2", &p.params_webgl2)]
370                {
371                    assert_eq!(
372                        params.get(prev),
373                        params.get(*name),
374                        "GL enum {num} aliases {prev} and {name}, which hold different \
375                         {version} values on {tier:?}; collapsing them would serve the \
376                         wrong one"
377                    );
378                }
379            }
380            continue;
381        }
382        out.insert(num.to_string(), serde_json::Value::from(*name));
383    }
384    out
385}

544    fn per_context_mutable_state_is_never_table_served() {
545        // The C1 regression guard at the Rust layer: `getParameter` answers
546        // both static device capabilities and per-context mutable state, and
547        // only the first may come from a table. A frozen `BLEND` contradicts
548        // the `gl.enable(gl.BLEND)` the page just called, and a frozen
549        // `VIEWPORT` contradicts `gl.drawingBufferWidth` after a canvas
550        // resize. Absent from the map is all it takes: `webgl.js` falls
551        // through to the real backend for any name the table does not carry.
552        for tier in types::Tier::ALL {
553            let p = profile_for_tier(*tier);
554            for name in [
555                "BLEND",                      // gl.enable / gl.disable
556                "VIEWPORT",                   // gl.viewport, and canvas resize
557                "SCISSOR_BOX",                // gl.scissor
558                "COLOR_CLEAR_VALUE",          // gl.clearColor
559                "STENCIL_REF",                // gl.stencilFunc
560                "UNPACK_FLIP_Y_WEBGL",        // gl.pixelStorei
561                "ACTIVE_TEXTURE",             // gl.activeTexture
562                "DRAW_BUFFER0",               // gl.drawBuffers
563                "STENCIL_BITS",               // getContext({stencil: true})
564                "SAMPLES",                    // getContext({antialias: false})
565                "COMPRESSED_TEXTURE_FORMATS", // grows as extensions enable
566            ] {
567                for (version, params) in
568                    [("WebGL1", &p.params_webgl1), ("WebGL2", &p.params_webgl2)]
569                {
570                    assert!(
571                        !params.contains_key(name),
572                        "{tier:?} serves {name} to a {version} context; it is mutable state, \
573                         not a device capability, and must be delegated"
574                    );
575                }
576            }
577        }

770    #[test]
771    fn each_hardware_tier_resolves_its_own_measured_webgpu_adapter() {
772        let metal = webgpu_for_tier(types::Tier::MetalMacos).expect("Metal has an adapter");
773        let d3d11 = webgpu_for_tier(types::Tier::D3d11Fl11).expect("D3D11 has an adapter");
774
775        // Both captures enumerate the same 36 limits; what identifies the
776        // device is the values, so the assertion is that they differ rather
777        // than that the maps do.
778        assert_eq!(metal.limits.len(), 36);
779        assert_eq!(d3d11.limits.len(), 36);
780        assert_ne!(
781            metal.limits, d3d11.limits,
782            "two tiers resolving identical limits would mean one is being served the other's"
783        );
784        // The pair a script reads first, and the widest gap between the two: a
785        // Metal buffer caps at 4 GiB - 4, D3D12's at exactly 2 GiB.
786        assert_eq!(metal.limits["maxBufferSize"], 4_294_967_292);
787        assert_eq!(d3d11.limits["maxBufferSize"], 2_147_483_648);
788        // Backend-shaped rather than card-shaped, like the WebGL values:
789        // Metal's argument-buffer tier allows 10 storage buffers per stage
790        // where D3D12's binding tier allows 16.
791        assert_eq!(metal.limits["maxStorageBuffersPerShaderStage"], 10);
792        assert_eq!(d3d11.limits["maxStorageBuffersPerShaderStage"], 16);
793
794        assert_eq!(metal.features.len(), 22);
795        assert_eq!(d3d11.features.len(), 19);
796        // The three Metal has and D3D11 does not — Apple GPUs support the
797        // mobile-lineage compressed formats, desktop D3D12 does not.
798        for only_metal in [
799            "texture-compression-astc",
800            "texture-compression-astc-sliced-3d",
801            "texture-compression-etc2",
802        ] {
803            assert!(metal.features.iter().any(|f| f == only_metal));
804            assert!(!d3d11.features.iter().any(|f| f == only_metal));
805        }
806    }
