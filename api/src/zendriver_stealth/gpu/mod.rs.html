15/// One GPU's identity and static capabilities, fully resolved.
16///
17/// Produced by flattening the shared base table, the tier's overrides, the
18/// device row, and any caller-supplied spec. Callers only ever see this
19/// flattened form, so the internal base/override split can change without
20/// breaking anyone.
21///
22/// The parameter maps hold *device capabilities* only — the values that are
23/// implementation-fixed and differ per GPU. Per-context mutable state
24/// (`VIEWPORT`, `BLEND`, the `STENCIL_*`/`PACK_*`/`UNPACK_*` families, ...)
25/// is deliberately absent so it delegates to the real backend; freezing it
26/// would contradict the page's own GL calls. A caller may still pin such a
27/// key through [`overlay`](Self::overlay) — nothing here forbids it — but the
28/// shipped tiers never do.
29///
30/// Derives `Serialize`/`Deserialize` because [`Persona`](crate::Persona) does
31/// (it round-trips through JSON as `browser_fingerprint_generate`'s return
32/// value and in `Persona::overlay`'s own tests) — every field type here
33/// already supported serde before this field existed, so this is just wiring
34/// the derive through, not adding new capability.
35#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
36pub struct GpuProfile {
37    pub provenance: Provenance,
38    pub params_webgl1: BTreeMap<String, GlParam>,
39    pub params_webgl2: BTreeMap<String, GlParam>,
40    pub precision: BTreeMap<String, ShaderPrecision>,
41    pub extensions_webgl1: Vec<String>,
42    pub extensions_webgl2: Vec<String>,
43    pub unmasked_vendor: String,
44    pub unmasked_renderer: String,
45}
46
47impl GpuProfile {
48    /// An all-empty profile, used as an overlay base.
49    #[must_use]
50    pub fn empty() -> Self {
51        Self {
52            provenance: Provenance::Derived {
53                source: "empty".into(),
54            },
55            params_webgl1: BTreeMap::new(),
56            params_webgl2: BTreeMap::new(),
57            precision: BTreeMap::new(),
58            extensions_webgl1: Vec::new(),
59            extensions_webgl2: Vec::new(),
60            unmasked_vendor: String::new(),
61            unmasked_renderer: String::new(),
62        }
63    }
64
65    /// Merge `over` on top of `self`.
66    ///
67    /// The maps merge key-wise rather than whole-field, so a caller supplying a
68    /// partial profile overrides only the keys it sets and cannot silently wipe
69    /// entries it said nothing about. The lists and strings replace wholesale,
70    /// but only when non-empty, so an empty `over` is a no-op rather than a
71    /// clear.
72    #[must_use]
73    pub fn overlay(mut self, over: GpuProfile) -> GpuProfile {
74        self.params_webgl1.extend(over.params_webgl1);
75        self.params_webgl2.extend(over.params_webgl2);
76        self.precision.extend(over.precision);
77        if !over.extensions_webgl1.is_empty() {
78            self.extensions_webgl1 = over.extensions_webgl1;
79        }
80        if !over.extensions_webgl2.is_empty() {
81            self.extensions_webgl2 = over.extensions_webgl2;
82        }
83        if !over.unmasked_vendor.is_empty() {
84            self.unmasked_vendor = over.unmasked_vendor;
85        }
86        if !over.unmasked_renderer.is_empty() {
87            self.unmasked_renderer = over.unmasked_renderer;
88        }
89        self
90    }
91}

93fn tier_key(tier: Tier) -> &'static str {
94    match tier {
95        Tier::SwiftShader => "swiftshader",
96        Tier::MetalMacos => "metal-macos",
97        Tier::D3d11Fl11 => "d3d11-fl11",
98        Tier::D3d11Fl11Nvidia => "d3d11-fl11-nvidia",
99        Tier::VulkanMesaIntelIrisPro580 => "vulkan-mesa-intel-iris-pro-580",
100    }
101}

103fn lookup<'a, V>(table: &'a [(&str, V)], key: &str) -> Option<&'a V> {
104    table.iter().find(|(k, _)| *k == key).map(|(_, v)| v)
105}

107/// Merge one context version's base table with a tier's overrides.
108///
109/// The two context versions are kept apart deliberately: a WebGL1 context
110/// serves 18 capabilities and a WebGL2 context 47, and serving the WebGL2 set
111/// to a WebGL1 context would answer enums that context has no constant for,
112/// where real Chrome returns `null` and raises `INVALID_ENUM`.
113///
114/// Those counts are the *static device capabilities* out of the 82 and up-to-132
115/// parameters the captures enumerate. The remainder is per-context mutable
116/// state and is deliberately absent from the tables, so it delegates to the
117/// real backend — see `gpu-tier-gen`'s `SERVED_CAPS` for the rule and
118/// `DELEGATED_PARAMS` for why each excluded name is excluded.
119fn flatten(
120    base: &[(&str, GlParamRef)],
121    overrides: &[(&str, &[(&str, GlParamRef)])],
122    key: &str,
123) -> BTreeMap<String, GlParam> {
124    let mut out: BTreeMap<String, GlParam> = base
125        .iter()
126        .map(|(k, v)| ((*k).to_string(), v.to_owned_param()))
127        .collect();
128    if let Some(over) = lookup(overrides, key) {
129        for (k, v) in *over {
130            out.insert((*k).to_string(), v.to_owned_param());
131        }
132    }
133    out
134}

136/// Flatten the base tables plus one tier's overrides into a profile.
137pub(crate) fn profile_for_tier(tier: Tier) -> GpuProfile {
138    let key = tier_key(tier);
139    let params = flatten(
140        tiers::BASE_PARAMS_WEBGL2,
141        tiers::PARAM_OVERRIDES_WEBGL2,
142        key,
143    );
144    let precision = lookup(tiers::PRECISION, key)
145        .map(|rows| {
146            rows.iter()
147                .map(|(k, [a, b, c])| {
148                    (
149                        (*k).to_string(),
150                        ShaderPrecision {
151                            range_min: *a,
152                            range_max: *b,
153                            precision: *c,
154                        },
155                    )
156                })
157                .collect()
158        })
159        .unwrap_or_default();
160    let exts = |t: &[(&str, &[&str])]| -> Vec<String> {
161        lookup(t, key)
162            .map(|l| l.iter().map(|s| (*s).to_string()).collect())
163            .unwrap_or_default()
164    };
165    GpuProfile {
166        provenance: Provenance::Probed {
167            chrome: "see data/gpu-tiers".into(),
168            os: "see data/gpu-tiers".into(),
169        },
170        params_webgl1: flatten(
171            tiers::BASE_PARAMS_WEBGL1,
172            tiers::PARAM_OVERRIDES_WEBGL1,
173            key,
174        ),
175        params_webgl2: params,
176        precision,
177        extensions_webgl1: exts(tiers::EXTENSIONS_WEBGL1),
178        extensions_webgl2: exts(tiers::EXTENSIONS_WEBGL2),
179        unmasked_vendor: String::new(),
180        unmasked_renderer: String::new(),
181    }
182}

184/// One tier's measured WebGPU adapter capabilities, owned so a caller's
185/// [`WebgpuSpec`](crate::WebgpuSpec) can overlay onto them.
186///
187/// Deliberately **not** a field on [`GpuProfile`]. That type is public with
188/// public fields, and `WebgpuSpec` already owns the caller-facing WebGPU
189/// override path — a second one would give the same value two places to be set
190/// from, and would make the shape of an internal table part of the public API.
191#[derive(Debug, Clone, PartialEq, Eq)]
192pub(crate) struct WebgpuAdapter {
193    /// `GPUSupportedLimits`, keyed as the WebGPU IDL spells them
194    /// (`maxTextureDimension2D`, `maxBufferSize`, ...).
195    pub limits: BTreeMap<String, u64>,
196    /// `GPUSupportedFeatures`, in Chrome's own iteration order.
197    pub features: Vec<String>,
198}
199
200/// The WebGPU adapter capabilities a tier measured, or `None` when that tier's
201/// machine has no adapter at all.
202///
203/// The sibling of [`profile_for_tier`] for the other GPU API: both read one
204/// tier key out of the generated tables, so a renderer that resolves a tier
205/// gets that tier's answer on both surfaces and cannot describe two devices.
206///
207/// **`None` is data, not a gap.** Real SwiftShader Chrome resolves
208/// `requestAdapter()` to null, so that tier has no adapter to describe, and the
209/// capture records exactly that. Callers must leave the host's adapter alone
210/// rather than substituting a neighbouring tier's numbers — a persona claiming
211/// a software rasterizer to WebGL while reporting a hardware adapter's limits
212/// to WebGPU is the cross-API contradiction this resolution exists to prevent.
213pub(crate) fn webgpu_for_tier(tier: Tier) -> Option<WebgpuAdapter> {
214    let measured = lookup(tiers::WEBGPU_ADAPTERS, tier_key(tier))?.as_ref()?;
215    Some(WebgpuAdapter {
216        limits: measured
217            .limits
218            .iter()
219            .map(|(k, v)| ((*k).to_string(), *v))
220            .collect(),
221        features: measured.features.iter().map(|f| (*f).to_string()).collect(),
222    })
223}

225/// Enums the patch must resolve that the generated [`tiers::ENUM_NAMES`] table
226/// does not carry, because the probe only walks the *core* `getParameter`
227/// enums:
228///
229/// - the two `WEBGL_debug_renderer_info` names, which exist only once that
230///   extension has been fetched, and
231/// - the shader-type and precision-type arguments `getShaderPrecisionFormat`
232///   takes, which are never passed to `getParameter` at all.
233///
234/// Both reach the page through the same number-to-name lookup as every other
235/// value, so leaving them out does not merely omit them — it makes the JS fall
236/// through to the real backend, serving the host's own unmasked pair and its
237/// own shader precision. All ten are fixed by the WebGL spec, so they are
238/// constants here rather than measurements.
239const EXTRA_ENUMS: &[(u32, &str)] = &[
240    (35632, "FRAGMENT_SHADER"),
241    (35633, "VERTEX_SHADER"),
242    (36336, "LOW_FLOAT"),
243    (36337, "MEDIUM_FLOAT"),
244    (36338, "HIGH_FLOAT"),
245    (36339, "LOW_INT"),
246    (36340, "MEDIUM_INT"),
247    (36341, "HIGH_INT"),
248    (37445, "UNMASKED_VENDOR_WEBGL"),
249    (37446, "UNMASKED_RENDERER_WEBGL"),
250];
251
252/// `DRAW_BUFFER0`, which ES 3.0 fixes at `0x8825`, with `DRAW_BUFFER1` through
253/// `DRAW_BUFFER15` following contiguously.
254const DRAW_BUFFER0: u32 = 34853;
255
256/// How many `DRAW_BUFFERn` enums the ES 3.0 spec defines.
257const DRAW_BUFFER_COUNT: u32 = 16;
258
259/// Name every `DRAW_BUFFERn` enum, whether or not a capture recorded it.
260///
261/// A capture only records the indices its own backend answered, so the
262/// SwiftShader capture stops at `DRAW_BUFFER5` (that backend has 6). Taking the
263/// names from captures alone would then leave a SwiftShader persona unable to
264/// recognise `DRAW_BUFFER6`, and `webgl.js` cannot suppress an enum it cannot
265/// name: the read would fall through to a host that has 8 and answer beside a
266/// cap saying 6 exist.
267///
268/// These are spec constants rather than measurements, the same category as
269/// [`EXTRA_ENUMS`] above, so the full range is declared here instead of being
270/// inferred from whatever hardware happened to be probed.
271fn draw_buffer_enums() -> impl Iterator<Item = (u32, String)> {
272    (0..DRAW_BUFFER_COUNT).map(|i| (DRAW_BUFFER0 + i, format!("DRAW_BUFFER{i}")))
273}

275/// Serialize a profile into the JSON object `webgl.js` consumes.
276///
277/// Each value carries a `t` tag naming its GL type so the JS side builds the
278/// right typed array; JSON alone cannot distinguish `Int32Array` from
279/// `Float32Array`.
280///
281/// The unmasked vendor/renderer strings are folded into both parameter maps
282/// (and their enums into `enumNames`, via [`EXTRA_ENUMS`]) rather than shipped
283/// as their own fields: a page reads them through `getParameter` exactly like
284/// every other value, so giving them a second path through the JS would buy
285/// nothing.
286pub(crate) fn profile_to_js(p: &GpuProfile) -> String {
287    use serde_json::{Map, Value, json};
288
289    fn val(v: &GlParam) -> Value {
290        match v {
291            GlParam::Int(i) => json!({"t": "i", "v": i}),
292            GlParam::Float(f) => json!({"t": "f", "v": f}),
293            GlParam::Bool(b) => json!({"t": "b", "v": b}),
294            GlParam::Str(s) => json!({"t": "s", "v": s}),
295            GlParam::IntPair(a) => json!({"t": "i32pair", "v": a}),
296            GlParam::IntQuad(a) => json!({"t": "i32quad", "v": a}),
297            GlParam::FloatPair(a) => json!({"t": "f32pair", "v": a}),
298            GlParam::FloatQuad(a) => json!({"t": "f32quad", "v": a}),
299            GlParam::IntList(a) => json!({"t": "u32list", "v": a}),
300        }
301    }
302    let conv = |m: &BTreeMap<String, GlParam>| -> Map<String, Value> {
303        let mut out: Map<String, Value> = m.iter().map(|(k, v)| (k.clone(), val(v))).collect();
304        out.insert(
305            "UNMASKED_VENDOR_WEBGL".to_string(),
306            json!({"t": "s", "v": p.unmasked_vendor}),
307        );
308        out.insert(
309            "UNMASKED_RENDERER_WEBGL".to_string(),
310            json!({"t": "s", "v": p.unmasked_renderer}),
311        );
312        out
313    };
314
315    let mut enums = enum_names();
316    for (num, name) in EXTRA_ENUMS {
317        enums.insert(num.to_string(), Value::from(*name));
318    }
319    for (num, name) in draw_buffer_enums() {
320        enums.insert(num.to_string(), Value::from(name));
321    }
322
323    json!({
324        "params1": conv(&p.params_webgl1),
325        "params2": conv(&p.params_webgl2),
326        "precision": p.precision.iter().map(|(k, v)| {
327            (k.clone(), json!([v.range_min, v.range_max, v.precision]))
328        }).collect::<Map<_, _>>(),
329        "extensions1": p.extensions_webgl1,
330        "extensions2": p.extensions_webgl2,
331        "enumNames": enums,
332        "inertStubs": devices::inert_stubs(),
333    })
334    .to_string()
335}

337/// Numeric GL enum to parameter name, as the JS side indexes it.
338///
339/// A JS object is keyed by number, so aliases collapse: `BLEND_EQUATION` and
340/// `BLEND_EQUATION_RGB` are both enum `32777`, and only one name survives into
341/// the emitted object. That is safe **only while aliased names carry equal
342/// values**, which is asserted below rather than assumed — if a future capture
343/// ever gives an aliased pair different values, silently keeping one would
344/// serve the wrong number for the other.
345///
346/// The comparison runs over each tier's *resolved* maps, both context
347/// versions, because that is what the page actually reads. Consulting the
348/// shared WebGL2 base alone would pass a tier override that moved one spelling
349/// of a collapsed pair and not the other, and would never look at WebGL1 at
350/// all. A present/absent split fails too: whichever spelling survives has to
351/// be the one the table carries, or the lookup misses and the enum falls
352/// through to the real backend.
353fn enum_names() -> serde_json::Map<String, serde_json::Value> {
354    let resolved: Vec<(Tier, GpuProfile)> = Tier::ALL
355        .iter()
356        .map(|tier| (*tier, profile_for_tier(*tier)))
357        .collect();
358    let mut out = serde_json::Map::new();
359    let mut chosen: BTreeMap<u32, &str> = BTreeMap::new();
360    for (num, name) in tiers::ENUM_NAMES {
361        if let Some(prev) = chosen.insert(*num, *name) {
362            // Both spellings must resolve to the same value, or collapsing
363            // them changes what the page reads.
364            for (tier, p) in &resolved {
365                for (version, params) in
366                    [("WebGL1", &p.params_webgl1), ("WebGL2", &p.params_webgl2)]
367                {
368                    assert_eq!(
369                        params.get(prev),
370                        params.get(*name),
371                        "GL enum {num} aliases {prev} and {name}, which hold different \
372                         {version} values on {tier:?}; collapsing them would serve the \
373                         wrong one"
374                    );
375                }
376            }
377            continue;
378        }
379        out.insert(num.to_string(), serde_json::Value::from(*name));
380    }
381    out
382}

485    fn per_context_mutable_state_is_never_table_served() {
486        // The C1 regression guard at the Rust layer: `getParameter` answers
487        // both static device capabilities and per-context mutable state, and
488        // only the first may come from a table. A frozen `BLEND` contradicts
489        // the `gl.enable(gl.BLEND)` the page just called, and a frozen
490        // `VIEWPORT` contradicts `gl.drawingBufferWidth` after a canvas

492        // through to the real backend for any name the table does not carry.
493        for tier in types::Tier::ALL {

495            for name in [
496                "BLEND",                      // gl.enable / gl.disable
497                "VIEWPORT",                   // gl.viewport, and canvas resize
498                "SCISSOR_BOX",                // gl.scissor
499                "COLOR_CLEAR_VALUE",          // gl.clearColor
500                "STENCIL_REF",                // gl.stencilFunc
501                "UNPACK_FLIP_Y_WEBGL",        // gl.pixelStorei
502                "ACTIVE_TEXTURE",             // gl.activeTexture
503                "DRAW_BUFFER0",               // gl.drawBuffers
504                "STENCIL_BITS",               // getContext({stencil: true})
505                "SAMPLES",                    // getContext({antialias: false})
506                "COMPRESSED_TEXTURE_FORMATS", // grows as extensions enable
507            ] {
508                for (version, params) in
509                    [("WebGL1", &p.params_webgl1), ("WebGL2", &p.params_webgl2)]
510                {
511                    assert!(
512                        !params.contains_key(name),
513                        "{tier:?} serves {name} to a {version} context; it is mutable state, \
514                         not a device capability, and must be delegated"
515                    );
516                }
517            }
518        }
519    }

641        // The JS builds its precision key as
642        // `enumNames[shaderType] + '/' + enumNames[precisionType]`, and none of
643        // those eight enums is a getParameter enum, so the generated table does
644        // not carry them. Without EXTRA_ENUMS the key is "undefined/undefined",
645        // every lookup misses, and getShaderPrecisionFormat quietly serves the
646        // real backend's precision instead of the tier's.

648        let js: serde_json::Value = serde_json::from_str(&profile_to_js(&p)).unwrap();
649        let named: std::collections::BTreeSet<&str> = js["enumNames"]
650            .as_object()
651            .unwrap()
652            .values()
653            .filter_map(serde_json::Value::as_str)
654            .collect();
655        for key in p.precision.keys() {
656            let (shader, precision) = key.split_once('/').expect("precision keys are A/B");
657            assert!(named.contains(shader), "no enum number names {shader}");
658            assert!(
659                named.contains(precision),
660                "no enum number names {precision}"
661            );
662        }

666    fn every_draw_buffer_below_the_served_cap_is_named_in_the_enum_table() {
667        // `webgl.js` fills the DRAW_BUFFERn gap — the indices the served
668        // MAX_DRAW_BUFFERS claims but the host backend does not have — by
669        // walking `enumNames` for names matching DRAW_BUFFERn. A name missing
670        // from that table is an index the fill can never reach, so it would
671        // keep answering null beside a cap that says it exists.
672        for tier in types::Tier::ALL {
673            let p = profile_for_tier(*tier);
674            let GlParam::Int(max_draw) = p.params_webgl2["MAX_DRAW_BUFFERS"] else {
675                panic!("{tier:?} does not serve MAX_DRAW_BUFFERS as an integer");
676            };
677            let js: serde_json::Value = serde_json::from_str(&profile_to_js(&p)).unwrap();
678            let named: std::collections::BTreeSet<&str> = js["enumNames"]
679                .as_object()
680                .unwrap()
681                .values()
682                .filter_map(serde_json::Value::as_str)
683                .collect();
684            for i in 0..max_draw {
685                let name = format!("DRAW_BUFFER{i}");
686                assert!(
687                    named.contains(name.as_str()),
688                    "{tier:?} serves MAX_DRAW_BUFFERS {max_draw} but no enum number names \
689                     {name}; webgl.js could not fill it if the backend lacks it"
690                );
691            }
692        }

711    #[test]
712    fn each_hardware_tier_resolves_its_own_measured_webgpu_adapter() {
713        let metal = webgpu_for_tier(types::Tier::MetalMacos).expect("Metal has an adapter");
714        let d3d11 = webgpu_for_tier(types::Tier::D3d11Fl11).expect("D3D11 has an adapter");
715
716        // Both captures enumerate the same 36 limits; what identifies the
717        // device is the values, so the assertion is that they differ rather
718        // than that the maps do.
719        assert_eq!(metal.limits.len(), 36);
720        assert_eq!(d3d11.limits.len(), 36);
721        assert_ne!(
722            metal.limits, d3d11.limits,
723            "two tiers resolving identical limits would mean one is being served the other's"
724        );
725        // The pair a script reads first, and the widest gap between the two: a
726        // Metal buffer caps at 4 GiB - 4, D3D12's at exactly 2 GiB.
727        assert_eq!(metal.limits["maxBufferSize"], 4_294_967_292);
728        assert_eq!(d3d11.limits["maxBufferSize"], 2_147_483_648);
729        // Backend-shaped rather than card-shaped, like the WebGL values:
730        // Metal's argument-buffer tier allows 10 storage buffers per stage
731        // where D3D12's binding tier allows 16.
732        assert_eq!(metal.limits["maxStorageBuffersPerShaderStage"], 10);
733        assert_eq!(d3d11.limits["maxStorageBuffersPerShaderStage"], 16);
734
735        assert_eq!(metal.features.len(), 22);
736        assert_eq!(d3d11.features.len(), 19);
737        // The three Metal has and D3D11 does not — Apple GPUs support the
738        // mobile-lineage compressed formats, desktop D3D12 does not.
739        for only_metal in [
740            "texture-compression-astc",
741            "texture-compression-astc-sliced-3d",
742            "texture-compression-etc2",
743        ] {
744            assert!(metal.features.iter().any(|f| f == only_metal));
745            assert!(!d3d11.features.iter().any(|f| f == only_metal));
746        }
747    }
