1//! Value types for the GPU profile tables.
2//!
3//! `GlParam` mirrors the shapes `WebGLRenderingContext.getParameter` can
4//! return. The float variants exist even though the probe captures cannot
5//! produce them: `JSON.stringify` collapses `1.0` to `1`, so a capture shows
6//! `ALIASED_POINT_SIZE_RANGE` as `[1, 1023]` when the WebGL spec declares it
7//! `GLfloat[2]`. The generator applies the spec's declared type; emitting an
8//! `Int32Array` where Chrome returns a `Float32Array` is a one-line tell.
9
10use serde::{Deserialize, Serialize};
11
12/// One `getParameter` return value.
13#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
14#[serde(rename_all = "snake_case")]
15pub enum GlParam {
16    /// `GLint` / `GLint64` / `GLuint` scalar.
17    Int(i64),
18    /// `GLfloat` scalar (e.g. `LINE_WIDTH`, `MAX_TEXTURE_MAX_ANISOTROPY_EXT`).
19    Float(f64),
20    /// `GLboolean`.
21    Bool(bool),
22    /// `Int32Array(2)` (e.g. `MAX_VIEWPORT_DIMS`).
23    IntPair([i32; 2]),
24    /// `Float32Array(2)` (e.g. `ALIASED_POINT_SIZE_RANGE`, `DEPTH_RANGE`).
25    FloatPair([f32; 2]),
26    /// `Float32Array(4)` (e.g. `BLEND_COLOR`, `COLOR_CLEAR_VALUE`).
27    FloatQuad([f32; 4]),
28    /// `Int32Array(4)` (e.g. `VIEWPORT`, `SCISSOR_BOX`).
29    IntQuad([i32; 4]),
30    /// `Uint32Array` of variable length (`COMPRESSED_TEXTURE_FORMATS`).
31    IntList(Vec<u32>),
32    /// `DOMString` (e.g. `VERSION`, `SHADING_LANGUAGE_VERSION`).
33    Str(String),
34}
35
36/// `'static`-friendly mirror of [`GlParam`] for the generated tables.
37///
38/// `GlParam` owns its `String`/`Vec` payloads so callers can build one at
39/// runtime; a `static` table cannot. The two convert with [`GlParamRef::to_owned_param`].
40///
41/// Every shape `getParameter` can return is represented, even though the
42/// currently generated tables only construct `Int`/`Float`/`Str`/`IntPair`/
43/// `FloatPair`. The served set is static device capabilities only, and today
44/// none of those is a boolean, a quad, or a list — but the generator's
45/// emitter covers all nine, so a future capture that promotes such a
46/// capability must not need a type change here to be emittable.
47#[allow(dead_code)]
48#[derive(Debug, Clone, Copy, PartialEq)]
49pub(crate) enum GlParamRef {
50    Int(i64),
51    Float(f64),
52    Bool(bool),
53    IntPair([i32; 2]),
54    FloatPair([f32; 2]),
55    FloatQuad([f32; 4]),
56    IntQuad([i32; 4]),
57    IntList(&'static [u32]),
58    Str(&'static str),
59}
60
61impl GlParamRef {
62    /// Widen a table entry into the owned form callers see.
63    pub(crate) fn to_owned_param(self) -> GlParam {
64        match self {
65            Self::Int(i) => GlParam::Int(i),
66            Self::Float(f) => GlParam::Float(f),
67            Self::Bool(b) => GlParam::Bool(b),
68            Self::IntPair(v) => GlParam::IntPair(v),
69            Self::FloatPair(v) => GlParam::FloatPair(v),
70            Self::FloatQuad(v) => GlParam::FloatQuad(v),
71            Self::IntQuad(v) => GlParam::IntQuad(v),
72            Self::IntList(v) => GlParam::IntList(v.to_vec()),
73            Self::Str(s) => GlParam::Str(s.to_string()),
74        }
75    }
76}

99/// Where a table's values came from. Travels with the data so a reader can
100/// tell a measured value from a derived one.
101#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
102#[serde(rename_all = "snake_case")]
103pub enum Provenance {
104    /// Measured on a real browser.
105    Probed { chrome: String, os: String },
106    /// Derived from a documented source (an ANGLE constant, a spec floor).
107    /// `source` cites it precisely enough to re-check.
108    Derived { source: String },
109}

111/// A backend capability tier. Values cluster by tier, not by GPU model:
112/// ANGLE computes them from constants branched on backend and feature level.
113#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
114pub(crate) enum Tier {
115    SwiftShader,
116    /// ANGLE's Metal backend on **macOS** — every Mac, Intel and Apple silicon
117    /// alike. Probed on an Apple M4 Pro, but the values are not that machine's:
118    /// `DisplayMtl.mm:727-731` sets `max2DTextureSize = 16384`,
119    /// `maxVaryingVectors = 31 - 1` and `maxVertexOutputComponents = 124 - 4`
120    /// as compile-time constants under `#if TARGET_OS_OSX ||
121    /// TARGET_OS_MACCATALYST`, with no device query anywhere in that arm.
122    ///
123    /// It was called `MetalAppleFamily3` until the `#if` was read properly. The
124    /// `supportsAppleGPUFamily(3)` test that picks 16384 over 8192 lives in the
125    /// `#else` arm, which is **iOS**, so no Mac ever reaches it and the old name
126    /// pointed at a branch macOS cannot take. Renaming it is what stops someone
127    /// capturing an Intel Mac in the expectation of a second, lower Metal tier —
128    /// it would reproduce this one exactly.
129    ///
130    /// **Confirmed on a second generation.** An M2 Pro, on a different Chrome
131    /// patch and a different macOS, matches this capture on every value: all 82
132    /// WebGL1 parameters, all 132 WebGL2, both extension lists, every shader
133    /// precision, all 36 WebGPU limits and all 22 WebGPU features. Only the
134    /// renderer string differs, which is the identity rather than a capability.
135    /// So the generalization claimed above is measured across Apple silicon
136    /// generations, not merely read out of the `#if`.
137    MetalMacos,
138    /// Direct3D 11 at feature level 11_0 or above, as ANGLE reports it for
139    /// every vendor **except NVIDIA** and for every Intel generation except
140    /// Gen9. Probed on an AMD Radeon (Raphael integrated, `0x164E`) under
141    /// Windows.
142    ///
143    /// Named for the backend and feature level rather than the card, because
144    /// `renderer11_utils.cpp` derives almost all of these values from
145    /// `D3D11_REQ_*` constants branched on the feature level rather than from
146    /// the device. That is what lets one capture stand for many cards, and it
147    /// is measured: an Intel Gen9 part and this AMD one agree on all 82 WebGL1
148    /// parameters, on 131 of the 132 WebGL2 ones, on both extension lists in
149    /// content *and* order, on every shader precision, and on all 36 WebGPU
150    /// limits.
151    ///
152    /// **Two values are device-derived, and both were found by probing rather
153    /// than predicted.**
154    ///
155    /// - `MAX_SAMPLES` is not a `D3D11_REQ_*` constant. ANGLE fills the
156    ///   per-format caps by asking the device —
157    ///   `ID3D11Device::CheckMultisampleQualityLevels` per renderable format —
158    ///   and takes the largest count that comes back, so this one really is the
159    ///   driver's answer. Intel Gen9 reports 16 where this capture reports 8.
160    /// - The WebGPU feature list is Dawn's answer for the physical adapter, not
161    ///   a feature-level constant at all. Gen9 enumerates 16 features against
162    ///   this capture's 19, lacking `shader-f16`, `subgroups` and
163    ///   `bgra8unorm-storage`.
164    ///
165    /// Both differences live in [`D3d11Fl11IntelGen9`](Self::D3d11Fl11IntelGen9),
166    /// which is why the sentence above says "except Gen9" instead of "AMD and
167    /// Intel alike", as it did before that tier was captured.
168    ///
169    /// **It does not cover NVIDIA** either, and that exception is likewise
170    /// measured — see [`D3d11Fl11Nvidia`](Self::D3d11Fl11Nvidia).
171    D3d11Fl11,
172    /// The same backend and feature level as [`D3d11Fl11`](Self::D3d11Fl11),
173    /// with ANGLE's NVIDIA-only workaround applied. Probed on an RTX 4090,
174    /// same machine and same Chrome build as the AMD capture above.
175    ///
176    /// **Exactly one value differs at the root**: `MAX_VERTEX_UNIFORM_VECTORS`
177    /// is 4095 here against 4096 there. `MAX_VERTEX_UNIFORM_COMPONENTS`
178    /// (`× 4`) and `MAX_COMBINED_VERTEX_UNIFORM_COMPONENTS` (`+ 12 × 16384`)
179    /// follow arithmetically, so the whole difference is three reported numbers
180    /// carrying one fact. Everything else is byte-identical across the two
181    /// captures: both WebGL parameter sets otherwise, both extension lists, all
182    /// shader precisions, all 36 WebGPU limits, and all 19 WebGPU features.
183    ///
184    /// The cause is in ANGLE, not in the silicon. `renderer11_utils.cpp` sets
185    /// `ANGLE_FEATURE_CONDITION(features, skipVSConstantRegisterZero, isNvidia)`
186    /// and then, in `GenerateCaps`, does
187    /// `if (features.skipVSConstantRegisterZero.enabled) {
188    /// caps->maxVertexUniformVectors -= 1; }`. The condition is the vendor and
189    /// nothing else — no driver version, no OS check — so the split is binary
190    /// and fully predictable: NVIDIA lands here, every other vendor lands on
191    /// [`D3d11Fl11`](Self::D3d11Fl11).
192    ///
193    /// This is why the two are separate tiers rather than one tier with a
194    /// computed adjustment. Both values were measured; neither is derived at
195    /// runtime.
196    ///
197    /// **It generalizes across NVIDIA generations, measured rather than
198    /// assumed.** A Maxwell GM108 (`0x134B`) reproduces this tier exactly —
199    /// every WebGL1 and WebGL2 parameter, every shader precision, both
200    /// extension lists in content and order — seven years and three process
201    /// nodes from the Lovelace part it was probed on. The capture is
202    /// `data/gpu-confirmations/d3d11-nvidia-maxwell-gm108.json`, pinned by
203    /// `maxwell_reproduces_the_nvidia_tier_exactly`. That result is also why
204    /// [`D3d11Fl11IntelGen9`](Self::D3d11Fl11IntelGen9) is a third tier rather
205    /// than a reason to doubt the model: the generalization holds, and Gen9's
206    /// two device-derived values are the exception to it.
207    D3d11Fl11Nvidia,
208    /// The same backend and feature level as [`D3d11Fl11`](Self::D3d11Fl11),
209    /// on **Intel Gen9 graphics**. Probed on an Intel HD Graphics 520
210    /// (Skylake, `0x1916`) in a Surface Book 1, Chrome 150.0.7871.187 on
211    /// Windows 10.
212    ///
213    /// **Exactly two values differ**, and both are the ones the D3D11 tier
214    /// cannot derive from a feature level:
215    ///
216    /// - `MAX_SAMPLES` is 16 here against 8 there. ANGLE asks the device
217    ///   (`CheckMultisampleQualityLevels` per renderable format) rather than
218    ///   reading a `D3D11_REQ_*` constant, so this number belongs to the
219    ///   driver.
220    /// - The WebGPU adapter enumerates 16 features against 19, missing
221    ///   `shader-f16`, `subgroups` and `bgra8unorm-storage`. Dawn reports what
222    ///   the physical adapter supports, and Gen9 supports neither packed fp16
223    ///   nor subgroup ops.
224    ///
225    /// **`MAX_SAMPLES` follows the silicon, and that is measured across every
226    /// backend available**, which is what rules out reading it as a quirk of
227    /// one capture or one backend:
228    ///
229    /// | Architecture | D3D11 | ANGLE-GL | ANGLE-Vulkan |
230    /// |---|---|---|---|
231    /// | AMD RDNA2 | 8 | 8 | 8 |
232    /// | Intel Gen9 | 16 | — | 16 |
233    ///
234    /// Constant per architecture across three backends, and different between
235    /// the two architectures on the backend they share. A backend-derived
236    /// value would look the other way round. The RDNA2 rows come from
237    /// `data/gpu-confirmations/`, and
238    /// `max_samples_follows_the_silicon_across_every_measured_backend` pins the
239    /// whole table.
240    ///
241    /// Everything else is identical to [`D3d11Fl11`](Self::D3d11Fl11): all 82
242    /// WebGL1 parameters, the other 131 WebGL2 ones, both extension lists in
243    /// content and order, every shader precision, and all 36 WebGPU limits.
244    /// That is the measurement behind the split being two values wide rather
245    /// than a whole second table.
246    ///
247    /// **Only Gen9 routes here, and the newer Intel generations are a known
248    /// unknown rather than a decision.** Gen11 (Iris Plus G4/G7) and Gen12
249    /// (Iris Xe, and the Arc parts above it) stay on
250    /// [`D3d11Fl11`](Self::D3d11Fl11). Nothing has probed either: both values
251    /// that moved here are device-derived, so a newer part could plausibly
252    /// report a different `MAX_SAMPLES` and almost certainly does expose
253    /// `shader-f16`, which Intel added with Gen12. Sweeping all Intel into this
254    /// tier would hand those machines two numbers nobody measured on that
255    /// silicon — and Iris Xe is the single heaviest entry in the whole device
256    /// catalogue, so the blast radius of guessing is at its largest exactly
257    /// where the evidence is absent. Leaving them on the generic tier is not a
258    /// claim that they match it; it is the choice to serve a value measured on
259    /// *some* FL11 device over one measured on none. **Closing the gap needs a
260    /// Gen11 and a Gen12 capture**, taken the same way this one was.
261    ///
262    /// **What "Gen9" means here is a marketing name, not a die.** Routing goes
263    /// through `devices::intel_architecture`, the same classifier that names
264    /// the WebGPU adapter's architecture, so a renderer resolves this tier
265    /// exactly when its adapter reports `gen-9` and the two cannot contradict
266    /// each other. That takes the three-digit `HD`/`UHD Graphics` 5xx and 6xx
267    /// families — 29 of the catalogue's 482 rows, carrying about 9% of the
268    /// corpus population (13% of the catalogue's own mass; the weights are
269    /// marginal probabilities over the whole corpus and do not sum to 1 across
270    /// the catalogue).
271    ///
272    /// The digit count is load-bearing rather than incidental. Intel spelled
273    /// Broadwell (Gen8) with four digits — `HD Graphics 5300`, `5500`, `5600`,
274    /// `6000` — so matching `hd graphics 5` as a *prefix* sweeps in a
275    /// generation nobody probed, and their PCI ids say so plainly: 0x16xx is
276    /// Broadwell where this capture's own part is 0x1916, Skylake. Those models
277    /// stay on [`D3d11Fl11`](Self::D3d11Fl11) with an empty architecture token,
278    /// which reads as an ordinary unclassified adapter rather than as a device
279    /// that does not exist.
280    ///
281    /// One imprecision remains, in the safe direction: several genuine Gen9
282    /// parts are **not** matched, because Intel sold them under an `Iris` name
283    /// (`Iris Graphics 540`/`550`, `Iris Plus Graphics 640`/`655`). They stay
284    /// on the catch-all tier. There is circumstantial support for moving them —
285    /// the committed [`VulkanMesaIntelIrisPro580`](Self::VulkanMesaIntelIrisPro580)
286    /// capture is Gen9 silicon on a different backend and also reports
287    /// `MAX_SAMPLES` 16, against 8 on both non-Gen9 D3D11 captures — but a
288    /// cross-backend inference is not a measurement, so the fix for this one is
289    /// a D3D11 capture of such a part.
290    D3d11Fl11IntelGen9,
291    /// ANGLE's **Vulkan** backend on Linux, on an Intel Iris Pro Graphics 580
292    /// (Skylake GT4e) under Mesa 25.2.8. Probed on Linux Mint with Chrome
293    /// 150.0.7871.186 in a NUC6i7KYK.
294    ///
295    /// **Named for the device and the driver, because a Vulkan tier does not
296    /// generalize — now measured rather than argued.** A second Mesa/Vulkan
297    /// device (AMD RDNA2 Van Gogh under RADV, same Chrome build) differs from
298    /// this tier in 12 WebGL2 parameters: `MAX_3D_TEXTURE_SIZE` 8192 against
299    /// 2048, `UNIFORM_BUFFER_OFFSET_ALIGNMENT` 4 against 64,
300    /// `MIN`/`MAX_PROGRAM_TEXEL_OFFSET` -32/31 against -8/7, and the extension
301    /// lists disagree too. Those are `VkPhysicalDeviceLimits` entries reaching
302    /// the page unchanged. See
303    /// `data/gpu-confirmations/vulkan-amd-rdna2-vangogh.json` and
304    /// `a_vulkan_tier_does_not_generalize_across_devices`. The other two hardware tiers do, and that is why they are
305    /// named for a backend: `renderer11_utils.cpp` branches on the
306    /// `D3D_FEATURE_LEVEL`, and `DisplayMtl.mm`'s `TARGET_OS_OSX` arm uses
307    /// plain compile-time constants — neither asks the device anything. ANGLE's
308    /// Vulkan backend does the opposite: `vk_caps_utils.cpp` fills its caps
309    /// straight from `VkPhysicalDeviceLimits` (`max2DTextureSize` is
310    /// `min(limitsVk.maxFramebufferWidth, limitsVk.maxImageDimension2D)`, the
311    /// viewport bounds come from `limitsVk.maxViewportDimensions`, and that one
312    /// file reads `limitsVk.` around 99 times). The limits it reads are the
313    /// driver's answer for this physical device, so the **Mesa version is part
314    /// of what determines these numbers** as much as the silicon is: a
315    /// different Intel part, or the same part on a different Mesa release, is a
316    /// different tier and needs its own capture. Nothing here may be reused for
317    /// "Linux" or for "Vulkan" in general.
318    ///
319    /// The measurement bears that out. This capture is closest to
320    /// [`SwiftShader`](Self::SwiftShader) — 7 of 82 WebGL1 parameters differ
321    /// and 21 of 132 WebGL2, against 10/26 for [`D3d11Fl11`](Self::D3d11Fl11)
322    /// and 9/23 for [`MetalMacos`](Self::MetalMacos) — because SwiftShader's
323    /// renderer string says `Vulkan 1.3.0` and it runs through the same ANGLE
324    /// backend. Those 21 remaining WebGL2 differences between two
325    /// Vulkan-backed captures on one Chrome build are exactly the
326    /// device-derived limits, which is the empirical form of what the source
327    /// predicts.
328    ///
329    /// **No WebGPU adapter**, and that is a measurement rather than a hole:
330    /// Chrome does not enable WebGPU by default on Linux, so `navigator.gpu`
331    /// exists and `requestAdapter()` resolves null. The capture records the
332    /// explicit null and the tier serves no limits or features, exactly as the
333    /// SwiftShader tier does.
334    VulkanMesaIntelIrisPro580,
335}

337/// One catalogued GPU **identity**, layered over a measured capability
338/// [`Tier`].
339///
340/// The catalogue widens which device a persona can claim; it never widens what
341/// any device can do. Every capability value still comes from `tiers.rs`, which
342/// is what stops a catalogue entry from being able to invent one.
343///
344/// The renderer string is deliberately **not** stored. It is composed on demand
345/// from ANGLE's own format, so a Chrome format change is one fix in one place
346/// rather than a rewrite of several hundred rows — and that format has already
347/// changed once, gaining the device id current ANGLE always appends.
348#[allow(dead_code)]
349#[derive(Debug, Clone, Copy, PartialEq)]
350pub(crate) struct CatalogueEntry {
351    /// Driver-reported model text: `Description` on D3D11, `MTLDevice.name` on
352    /// Metal.
353    pub model: &'static str,
354    /// ANGLE's vendor token, the first field of the renderer string.
355    pub vendor: &'static str,
356    /// PCI device id. `None` on Metal, where Apple silicon exposes none and the
357    /// string has nowhere to put one.
358    pub device_id: Option<u32>,
359    /// Which measured tier supplies this device's capability values.
360    pub tier: Tier,
361    /// Share of the corpus population, for the share-weighted draw.
362    ///
363    /// A marginal probability, `Σ_ua P(ua) · P(device | ua)`, since the corpus
364    /// reports device frequency conditioned on user agent. These do **not** sum
365    /// to 1 across the catalogue: the categories the catalogue excludes (iOS,
366    /// Windows-on-ARM, WARP, VM adapters, unmodelled backends) hold the rest,
367    /// so a caller drawing by share renormalizes over the subset it draws from.
368    pub weight: f64,
369}

390    #[test]
391    fn gl_param_covers_every_measured_shape() {
392        // These are the six shapes the probe actually observed, plus the two
393        // float forms the capture cannot distinguish (JSON collapses 1.0 -> 1).
394        let _ = GlParam::Int(16384);
395        let _ = GlParam::Bool(false);
396        let _ = GlParam::Str("WebGL GLSL ES 3.00".into());
397        let _ = GlParam::IntPair([16384, 16384]);
398        let _ = GlParam::FloatPair([1.0, 511.0]);
399        let _ = GlParam::FloatQuad([0.0, 0.0, 0.0, 0.0]);
400        let _ = GlParam::IntQuad([0, 0, 300, 150]);
401        let _ = GlParam::IntList(vec![]);
402        let _ = GlParam::Float(1.0);
403    }

438    #[test]
439    fn gl_param_ref_widens_to_the_owned_param_for_every_shape() {
440        // GlParamRef is the 'static mirror the generated tables store; every
441        // variant must widen back to the exact owned GlParam callers see.
442        assert_eq!(GlParamRef::Int(5).to_owned_param(), GlParam::Int(5));
443        assert_eq!(GlParamRef::Float(1.5).to_owned_param(), GlParam::Float(1.5));
444        assert_eq!(GlParamRef::Bool(true).to_owned_param(), GlParam::Bool(true));
445        assert_eq!(
446            GlParamRef::IntPair([1, 2]).to_owned_param(),
447            GlParam::IntPair([1, 2])
448        );
449        assert_eq!(
450            GlParamRef::FloatPair([1.0, 2.0]).to_owned_param(),
451            GlParam::FloatPair([1.0, 2.0])
452        );
453        assert_eq!(
454            GlParamRef::FloatQuad([1.0, 2.0, 3.0, 4.0]).to_owned_param(),
455            GlParam::FloatQuad([1.0, 2.0, 3.0, 4.0])
456        );
457        assert_eq!(
458            GlParamRef::IntQuad([1, 2, 3, 4]).to_owned_param(),
459            GlParam::IntQuad([1, 2, 3, 4])
460        );
461        assert_eq!(
462            GlParamRef::IntList(&[1, 2, 3]).to_owned_param(),
463            GlParam::IntList(vec![1, 2, 3])
464        );
465        assert_eq!(
466            GlParamRef::Str("hi").to_owned_param(),
467            GlParam::Str("hi".to_string())
468        );
469    }
