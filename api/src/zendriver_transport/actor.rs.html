31/// Default broadcast bus capacity. Lagged subscribers drop frames.
32pub(crate) const EVENT_BUS_CAPACITY: usize = 1024;

34/// Depth of the command channel feeding the actor
35/// ([`Connection`] handle → [`run_actor`]).
36///
37/// Named rather than written at each `mpsc::channel` site because the two
38/// sites are far apart — the initial spawn and
39/// [`Connection::reconnect`](crate::connection::Connection::reconnect) — and
40/// tuning one without the other would silently give a reconnected connection a
41/// different queue depth than it had on first connect. Also depended on by
42/// [`Connection::call_raw_with_timeout`](crate::connection::Connection::call_raw_with_timeout)'s
43/// rustdoc, which explains why the per-call budget has to cover the enqueue.
44pub(crate) const CMD_CHANNEL_CAPACITY: usize = 64;

46/// How many dispatched commands may pass between sweeps of the pending-reply
47/// map.
48///
49/// A `call_raw` that hits its budget drops its reply receiver and walks away,
50/// but the matching `oneshot::Sender` stays keyed in `pending` until Chrome
51/// answers that id — which, for the wedged browser the budget exists to
52/// escape, never happens. Left alone the map grows for the life of the
53/// connection.
54///
55/// Sweeping every 256 dispatches bounds the debris at a few hundred small heap
56/// entries — tens of kilobytes at the bound, since each `oneshot::Sender` is
57/// one pointer wide but retains a heap cell sized for
58/// `Result<Value, CdpRpcError>` — for one `HashMap::retain` per 256 commands.
59/// Deliberately **not** a capacity cap on `pending`: refusing to enqueue
60/// commands would shed real work on a busy connection, which is worse than a
61/// slow leak.
62///
63/// The counter advances on dispatch rather than on a clock, so this bounds the
64/// debris by command count, not by wall time: an idle connection holds whatever
65/// it has accumulated until traffic resumes.
66const PENDING_SWEEP_DISPATCH_INTERVAL: u64 = 256;
67
68/// Drop every pending entry whose caller has gone away — its reply receiver
69/// dropped, i.e. a `call_raw` that timed out or was cancelled. Returns how
70/// many entries were removed.
71///
72/// Dropping the sender early is invisible to the caller (it is already gone);
73/// a late reply for a swept id lands on the existing "response for unknown id"
74/// branch, exactly as it already does when a caller drops mid-flight.
75fn sweep_pending(pending: &mut HashMap<u64, oneshot::Sender<Result<Value, CdpRpcError>>>) -> usize {
76    let before = pending.len();
77    pending.retain(|_id, reply| !reply.is_closed());
78    before - pending.len()
79}

81/// Accounted-bus plumbing for one actor run: the second, opt-in broadcast
82/// sender plus this run's generation number. Bundled into a struct (rather
83/// than two more [`run_actor`] parameters) to keep the function signature
84/// under clippy's `too_many_arguments` threshold and to make the bus/
85/// generation pairing explicit at each call site — mirrors the
86/// [`crate::connection`]-side rationale for bundling wide handshake inputs
87/// into a struct.
88pub(crate) struct AccountedBus {
89    /// Second broadcast bus behind
90    /// [`Connection::subscribe_raw_accounted`](crate::connection::Connection::subscribe_raw_accounted).
91    pub(crate) tx: broadcast::Sender<AccountedRawEvent>,
92    /// This actor run's generation number, stamped onto every
93    /// [`AccountedRawEvent`] it produces.
94    pub(crate) generation: u64,
95}

113/// Runs the actor loop until `shutdown` is cancelled or the WS dies.
114///
115/// Generic over the WS sink + stream so tests can drive in-memory streams
116/// instead of real WebSockets.
117///
118/// `observers` are invoked serially (in registration order) on each new
119/// `Target.attachedToTarget`. The handler runs in a spawned task so the actor
120/// loop stays responsive; `weak_inner` lets the handler reconstruct a
121/// `Connection` (without forming a strong cycle) to talk back through this
122/// same actor.
123///
124/// `accounted` bundles the second, opt-in broadcast bus behind
125/// [`Connection::subscribe_raw_accounted`](crate::connection::Connection::subscribe_raw_accounted)
126/// with this actor run's generation number, stamped onto every
127/// [`AccountedRawEvent`] it produces. Mirroring onto `accounted.tx` — the
128/// per-event clone of [`RawEvent`] and the send itself — is gated behind
129/// `accounted.tx.receiver_count() > 0` so a connection with no accounted
130/// subscribers pays nothing beyond the existing `event_tx` broadcast.
131pub(crate) async fn run_actor<S>(
132    mut ws: S,
133    mut cmd_rx: mpsc::Receiver<OutboundCmd>,
134    event_tx: broadcast::Sender<RawEvent>,
135    accounted: AccountedBus,
136    shutdown: CancellationToken,
137    observers: Vec<Arc<dyn TargetObserver>>,
138    weak_inner: Weak<ConnectionInner>,
139) where
140    S: futures::Sink<Message, Error = tokio_tungstenite::tungstenite::Error>
141        + futures::Stream<Item = Result<Message, tokio_tungstenite::tungstenite::Error>>
142        + Unpin,
143{
144    let mut pending: HashMap<u64, oneshot::Sender<Result<Value, CdpRpcError>>> = HashMap::new();
145    let mut next_id: u64 = 1;
146    // Per-generation monotonic counter for `AccountedRawEvent::Event::sequence`.
147    // Reset to 1 on every actor run (a fresh call to `run_actor`, i.e. every
148    // reconnect); only advances when `accounted_tx` actually has a
149    // subscriber, since it counts positions *on the accounted bus*.
150    let mut next_accounted_sequence: u64 = 1;
151    // Commands dispatched since the last `sweep_pending`. See
152    // [`PENDING_SWEEP_DISPATCH_INTERVAL`].
153    let mut dispatched_since_sweep: u64 = 0;
154
155    // Sentinel stamped onto drained pendings when the loop exits. Defaults to
156    // the clean-shutdown code; the unexpected-ws-death branches below flip it
157    // to `DISCONNECTED_CODE` so callers awaiting a CDP call during a dropped
158    // connection can distinguish "Chrome died / socket dropped" from a
159    // caller-requested `shutdown()`.
160    let mut drain_code = crate::connection::SHUTDOWN_DRAIN_CODE;
161
162    loop {
163        tokio::select! {
164            biased;
165            () = shutdown.cancelled() => {
166                debug!("actor shutdown received; draining {} pending", pending.len());
167                break;
168            }
169            cmd = cmd_rx.recv() => {
170                let Some(cmd) = cmd else {
171                    debug!("cmd channel closed; shutting down");
172                    break;
173                };
174                let id = next_id;
175                next_id = next_id.wrapping_add(1);
176                let frame = CdpCommand {
177                    id,
178                    method: &cmd.method,
179                    params: cmd.params,
180                    session_id: cmd.session_id.as_deref(),
181                };
182                match serde_json::to_string(&frame) {
183                    Ok(s) => {
184                        trace!(id, method = %cmd.method, "send");
185                        if let Err(e) = ws.send(Message::text(s)).await {
186                            error!("ws send failed: {e}");
187                            // Same sentinel the remaining pendings get below:
188                            // a failed write means the socket died, and this
189                            // command is the one that hit the death first. A
190                            // generic `-32000` here would surface as an
191                            // ordinary `CallError::Rpc` — a code Chrome itself
192                            // emits constantly — so the caller's
193                            // `Disconnected` recovery would never fire for the
194                            // very call that discovered the disconnect.
195                            let _ = cmd.reply.send(Err(CdpRpcError {
196                                code: crate::connection::DISCONNECTED_CODE,
197                                message: format!("ws send failed: {e}"),
198                                data: None,
199                            }));
200                            // A failed write means the socket died — treat the
201                            // remaining pendings as an unexpected disconnect,
202                            // consistent with the read-side death branches.
203                            drain_code = crate::connection::DISCONNECTED_CODE;
204                            break;
205                        }
206                        pending.insert(id, cmd.reply);
207                        dispatched_since_sweep += 1;
208                        if dispatched_since_sweep >= PENDING_SWEEP_DISPATCH_INTERVAL {
209                            dispatched_since_sweep = 0;
210                            let swept = sweep_pending(&mut pending);
211                            if swept > 0 {
212                                debug!(
213                                    swept,
214                                    remaining = pending.len(),
215                                    "swept pendings whose caller had gone away"
216                                );
217                                // Publish the count so the sweep is observable
218                                // from outside this loop: `pending` is a local,
219                                // and a `debug!` nothing reads is not a fact a
220                                // test can assert on. The upgrade costs a
221                                // refcount bump at most once per
222                                // `PENDING_SWEEP_DISPATCH_INTERVAL` commands,
223                                // and only when the sweep actually reaped
224                                // something.
225                                if let Some(inner) = weak_inner.upgrade() {
226                                    inner.swept_total.fetch_add(
227                                        swept as u64,
228                                        std::sync::atomic::Ordering::Relaxed,
229                                    );
230                                }
231                            }
232                        }
233                    }
234                    Err(e) => {
235                        let _ = cmd.reply.send(Err(CdpRpcError {
236                            code: -32700,
237                            message: format!("serialize: {e}"),
238                            data: None,
239                        }));
240                    }
241                }
242            }
243            frame = ws.next() => {
244                match frame {
245                    None => {
246                        debug!("ws stream ended");
247                        // Stream vanished without a caller-requested shutdown:
248                        // an unexpected disconnect, not a clean close.
249                        drain_code = crate::connection::DISCONNECTED_CODE;
250                        break;
251                    }
252                    Some(Err(e)) => {
253                        error!("ws read failed: {e}");
254                        drain_code = crate::connection::DISCONNECTED_CODE;
255                        break;
256                    }
257                    Some(Ok(Message::Text(text))) => {
258                        match serde_json::from_str::<CdpInbound>(&text) {
259                            Ok(CdpInbound::Response { id, result, error, .. }) => {
260                                match pending.remove(&id) { Some(reply) => {
261                                    let res = match error {
262                                        Some(e) => Err(e),
263                                        None    => Ok(result.unwrap_or(Value::Null)),
264                                    };
265                                    let _ = reply.send(res);
266                                } _ => {
267                                    warn!(id, "response for unknown id (caller dropped?)");
268                                }}
269                            }
270                            Ok(CdpInbound::Event { method, params, session_id }) => {
271                                // Branch on target-lifecycle events before broadcasting.
272                                if method == "Target.attachedToTarget" && !observers.is_empty() {
273                                    match serde_json::from_value::<TargetAttached>(params.clone()) {
274                                        Ok(ev) => {
275                                            match weak_inner.upgrade() { Some(strong) => {
276                                                let conn = Connection { inner: strong };
277                                                let timeout_dur = conn.observer_timeout();
278                                                let observers_clone = observers.clone();
279                                                tokio::spawn(async move {
280                                                    handle_target_attached(
281                                                        conn,
282                                                        ev,
283                                                        observers_clone,
284                                                        timeout_dur,
285                                                    )
286                                                    .await;
287                                                });
288                                            } _ => {
289                                                warn!(
290                                                    "Target.attachedToTarget arrived but \
291                                                     Connection has dropped; skipping observers"
292                                                );
293                                            }}
294                                        }
295                                        Err(e) => error!(
296                                            "bad Target.attachedToTarget payload: {e}"
297                                        ),
298                                    }
299                                } else if method == "Target.detachedFromTarget"
300                                    && !observers.is_empty()
301                                {
302                                    if let Ok(ev) =
303                                        serde_json::from_value::<TargetDetached>(params.clone())
304                                    {
305                                        // Mirror the `attachedToTarget` path:
306                                        // observers can panic in user code, so
307                                        // wrap each call in AssertUnwindSafe +
308                                        // catch_unwind and a soft timeout. We
309                                        // can't tear down the session on
310                                        // failure (it's already detaching),
311                                        // but logging gives users a fighting
312                                        // chance to find a regression instead
313                                        // of dropping the panic silently.
314                                        let timeout_dur = match weak_inner.upgrade()
315                                        { Some(strong) => {
316                                            Connection { inner: strong }.observer_timeout()
317                                        } _ => {
318                                            // Connection has dropped — pick a
319                                            // short default so the task can't
320                                            // hang forever.
321                                            Duration::from_secs(5)
322                                        }};
323                                        for obs in &observers {
324                                            let obs2 = obs.clone();
325                                            let sid = ev.session_id.clone();
326                                            let name = obs2.name();
327                                            tokio::spawn(async move {
328                                                let fut = obs2.on_target_detached(&sid);
329                                                match tokio::time::timeout(
330                                                    timeout_dur,
331                                                    AssertUnwindSafe(fut).catch_unwind(),
332                                                )
333                                                .await
334                                                {
335                                                    Ok(Ok(())) => {}
336                                                    Ok(Err(panic)) => {
337                                                        let msg = panic_payload(&panic);
338                                                        error!(
339                                                            observer = name,
340                                                            session_id = %sid,
341                                                            panic = %msg,
342                                                            "detached-target observer panicked",
343                                                        );
344                                                    }
345                                                    Err(_) => warn!(
346                                                        observer = name,
347                                                        session_id = %sid,
348                                                        "detached-target observer timed out",
349                                                    ),
350                                                }
351                                            });
352                                        }
353                                    } else {
354                                        warn!("bad Target.detachedFromTarget payload");
355                                    }
356                                }
357                                let ev = RawEvent { method, params, session_id };
358                                // Gate the per-event clone + accounted-bus send
359                                // behind an actual subscriber: cloning `ev`
360                                // (method String + params Value — potentially
361                                // large Network.* JSON) is wasted work when
362                                // nobody is listening on the accounted bus.
363                                if accounted.tx.receiver_count() > 0 {
364                                    let sequence = next_accounted_sequence;
365                                    next_accounted_sequence =
366                                        next_accounted_sequence.wrapping_add(1);
367                                    let _ = accounted.tx.send(AccountedRawEvent::Event {
368                                        generation: accounted.generation,
369                                        sequence,
370                                        event: ev.clone(),
371                                    });
372                                }
373                                // Ignore SendError: zero subscribers is fine.
374                                let _ = event_tx.send(ev);
375                            }
376                            Err(e) => warn!("frame parse failed: {e} (text: {text})"),
377                        }
378                    }
379                    Some(Ok(Message::Close(_))) => {
380                        debug!("ws close frame; disconnecting");
381                        // Chrome closed the socket on us — treat as an
382                        // unexpected disconnect so in-flight callers see
383                        // `Disconnected` rather than the clean-shutdown code.
384                        drain_code = crate::connection::DISCONNECTED_CODE;
385                        break;
386                    }
387                    Some(Ok(_)) => { /* ignore binary, ping, pong, frame */ }
388                }
389            }
390        }
391    }
392
393    // Latch *why* this actor stopped, so calls that arrive after it is gone
394    // can still tell "Chrome died" from "I closed it". Without this, only
395    // in-flight calls (drained just below with `drain_code`) ever see
396    // `Disconnected`: every later call finds a closed command channel, whose
397    // `SendError` carries no reason, and reports `Shutdown`.
398    //
399    // The **channel** is what makes the flag trustworthy, not the atomic
400    // ordering. `cmd_rx` is dropped when this function returns, which is
401    // strictly after this store; the drop synchronizes with a sender observing
402    // the channel closed; happens-before is transitive. So any sender that
403    // observes the channel closed necessarily observes the flag too — there is
404    // no window where a caller sees the death but not its cause, and a
405    // `Relaxed` store would be visible just the same.
406    //
407    // `Release`/`Acquire` is kept anyway (rather than the `Relaxed` used for
408    // the advisory counters on `ConnectionInner`) because it costs nothing on
409    // this once-per-actor-exit path and keeps the flag correct if a future read
410    // path ever reaches it without going through the channel. Note the pairing
411    // only orders *this* location: adding a second flag would not give an
412    // ordering guarantee between the two.
413    if let Some(inner) = weak_inner.upgrade() {
414        inner.socket_died.store(
415            drain_code == crate::connection::DISCONNECTED_CODE,
416            std::sync::atomic::Ordering::Release,
417        );
418    }
419
420    // Drain pending into transport errors so callers don't hang. `drain_code`
421    // is a reserved sentinel matched by `Connection::call_raw`: it surfaces
422    // drained pendings as `TransportError::Shutdown` (clean shutdown) or
423    // `TransportError::Disconnected` (unexpected ws death) rather than a
424    // generic `Rpc` error.
425    let message = if drain_code == crate::connection::DISCONNECTED_CODE {
426        "connection lost"
427    } else {
428        "connection shut down"
429    };
430    // A genuine ws death (not a caller-requested `shutdown()` and not a
431    // `reconnect()` — both cancel via `shutdown.cancelled()`, which never
432    // sets `DISCONNECTED_CODE`) gets exactly one `Disconnected` on the
433    // accounted bus, gated the same as the per-event mirroring above.
434    if drain_code == crate::connection::DISCONNECTED_CODE && accounted.tx.receiver_count() > 0 {
435        let _ = accounted.tx.send(AccountedRawEvent::Disconnected {
436            generation: accounted.generation,
437        });
438    }
439    for (_id, reply) in pending.drain() {
440        let _ = reply.send(Err(CdpRpcError {
441            code: drain_code,
442            message: message.into(),
443            data: None,
444        }));
445    }
446    debug!(drain_code, "actor exit");
447}

449/// Run all `observers` serially against the just-attached session. Returns
450/// after either: (a) every observer succeeds — in which case we release the
451/// debugger via `Runtime.runIfWaitingForDebugger`; (b) one observer errors or
452/// panics — we detach via `Target.detachFromTarget` and return without
453/// releasing; or (c) an observer exceeds `observer_timeout` — behavior then
454/// depends on [`TargetObserver::failure_policy`]:
455/// [`ObserverFailurePolicy::Required`] (the default) detaches the same as
456/// (b), fail closed, so a hung observer can never leave the page running
457/// without its setup; [`ObserverFailurePolicy::BestEffort`] logs a warning
458/// and **skips only the timed-out observer**, then continues on to the
459/// remaining observers in the chain (if any) exactly as if this one had
460/// returned `Ok`. Once every remaining observer has run (or there were none
461/// left), the debugger is released the same as the success path. This
462/// matters because zendriver registers its tab-registrar observer last as a
463/// ready barrier (see `assemble_observer_chain` in `zendriver::browser`): a
464/// `BestEffort` timeout earlier in the chain must not skip the observers
465/// that come after it, or the registrar would never run and the target
466/// would leak — live and released, but absent from `BrowserInner::tabs`.
467///
468/// ## Cancellation
469///
470/// `AssertUnwindSafe + catch_unwind` covers panics inside the observer
471/// future, not cancellation of the outer task. In practice this function
472/// runs inside a `tokio::spawn`'d task that has no `JoinHandle` retained
473/// by the actor loop — the spawned task therefore cannot be cancelled by
474/// the actor dropping. The task owns the [`Connection`] via the moved
475/// `conn` field below, so the underlying transport stays alive at least
476/// until the task completes naturally. The only way for in-flight CDP
477/// calls to orphan is a runtime-wide shutdown, in which case the entire
478/// process is tearing down and the calls are moot.
479async fn handle_target_attached(
480    conn: Connection,
481    ev: TargetAttached,
482    observers: Vec<Arc<dyn TargetObserver>>,
483    observer_timeout: Duration,
484) {
485    let session_id = ev.session_id.clone();
486    for obs in &observers {
487        let paused = PausedSession {
488            session_id: &session_id,
489            target_info: &ev.target_info,
490            conn: &conn,
491        };
492        let name = obs.name();
493        let fut = obs.on_target_attached(paused);
494        match tokio::time::timeout(observer_timeout, AssertUnwindSafe(fut).catch_unwind()).await {
495            Ok(Ok(Ok(()))) => continue,
496            Ok(Ok(Err(e))) => {
497                error!(observer = name, %session_id, error = %e, "observer failed; detaching");
498                detach_target(&conn, &session_id).await;
499                return;
500            }
501            Ok(Err(panic)) => {
502                let msg = panic_payload(&panic);
503                error!(observer = name, %session_id, panic = %msg, "observer panicked; detaching");
504                detach_target(&conn, &session_id).await;
505                return;
506            }
507            Err(_) => match obs.failure_policy() {
508                ObserverFailurePolicy::Required => {
509                    // Same severity as the Err/panic branches above: a Required
510                    // observer timing out is a fail-closed detach, not a
511                    // best-effort skip — log it at error!, not warn!.
512                    error!(
513                        observer = name,
514                        %session_id,
515                        "observer timed out; detaching (fail-closed)"
516                    );
517                    detach_target(&conn, &session_id).await;
518                    return;
519                }
520                ObserverFailurePolicy::BestEffort => {
521                    warn!(
522                        observer = name,
523                        %session_id,
524                        "observer timed out; skipping (best-effort) and continuing the chain"
525                    );
526                    continue;
527                }
528            },
529        }
530    }
531    let _ = conn
532        .call_raw(
533            "Runtime.runIfWaitingForDebugger",
534            json!({}),
535            Some(session_id),
536        )
537        .await;
538}

540/// Shared detach path for every observer failure mode (`Err`, panic, or a
541/// `Required`-policy timeout): force-detach the session via
542/// `Target.detachFromTarget` so the target is never handed back to Chrome
543/// with `Runtime.runIfWaitingForDebugger`.
544async fn detach_target(conn: &Connection, session_id: &str) {
545    let _ = conn
546        .call_raw(
547            "Target.detachFromTarget",
548            json!({ "sessionId": session_id }),
549            None,
550        )
551        .await;
552}

554/// Best-effort extraction of a textual panic message from a `catch_unwind`
555/// payload. The standard library only guarantees a `Box<dyn Any + Send>`; we
556/// downcast to `&str` and `String` (the two cases the macros produce) and
557/// fall back to a placeholder for everything else.
558fn panic_payload(payload: &Box<dyn Any + Send>) -> String {
559    if let Some(s) = payload.downcast_ref::<&str>() {
560        (*s).to_string()
561    } else if let Some(s) = payload.downcast_ref::<String>() {
562        s.clone()
563    } else {
564        "<unknown panic payload>".to_string()
565    }
566}

977    /// The sweep has two halves and this test pins both.
978    ///
979    /// It must **run**: abandoned callers that cross the dispatch boundary get
980    /// reaped, observed through `ConnectionInner::swept_total` because
981    /// `pending` is a local inside [`run_actor`] and nothing else about the
982    /// sweep reaches the outside world. Without that assertion the entire leak
983    /// fix could be deleted from the actor loop with the suite still green —
984    /// which was the case until this test grew the counter check.
985    ///
986    /// And it must **reap only the abandoned**: a call still waiting has to
987    /// survive crossing the boundary and still get routed when Chrome finally
988    /// answers.
989    ///
990    /// Driven through the real `spawn_actor_with_observers` rather than a bare
991    /// [`run_actor`] with a `Weak::new()`, so the actor has a live
992    /// `ConnectionInner` to publish into; commands go in through the
993    /// connection's own `cmd_tx` so abandoned replies can be staged directly.
994    #[tokio::test]
995    async fn the_sweep_reaps_abandoned_pendings_and_spares_live_ones() {
996        const ABANDONED: u64 = PENDING_SWEEP_DISPATCH_INTERVAL * 2;
997
998        // +8 for the live command and headroom: the driver's sink is
999        // `try_send`-based and nothing drains it here, so every dispatched
1000        // frame has to fit or the write kills the actor.
1001        let (ws, test_tx, _test_rx) = duplex_pair_with_capacity(ABANDONED as usize + 8);
1002        let conn = spawn_actor_with_observers(ws, Vec::new());
1003        let cmd_tx = conn
1004            .inner
1005            .cmd_tx
1006            .lock()
1007            .unwrap_or_else(std::sync::PoisonError::into_inner)
1008            .clone();
1009
1010        // The one caller that stays alive. First command dispatched, so id 1.
1011        let (live_reply_tx, live_reply_rx) = oneshot::channel();
1012        cmd_tx
1013            .send(OutboundCmd {
1014                method: "Page.navigate".into(),
1015                params: json!({ "url": "https://x.test" }),
1016                session_id: None,
1017                reply: live_reply_tx,
1018            })
1019            .await
1020            .unwrap();
1021
1022        // Enough abandoned callers to cross the sweep interval twice over.
1023        // Dropping the receiver first is what a `call_raw` that hit its budget
1024        // leaves behind.
1025        for _ in 0..ABANDONED {
1026            let (abandoned_tx, abandoned_rx) = oneshot::channel();
1027            drop(abandoned_rx);
1028            cmd_tx
1029                .send(OutboundCmd {
1030                    method: "Abandoned.call".into(),
1031                    params: json!({}),
1032                    session_id: None,
1033                    reply: abandoned_tx,
1034                })
1035                .await
1036                .unwrap();
1037        }
1038
1039        // The actor's select is `biased` with `cmd_rx` ahead of the socket
1040        // read, so every one of those commands (and therefore both sweeps) is
1041        // processed before this reply is routed. Awaiting the reply below is
1042        // therefore also the barrier that makes `swept_total` readable without
1043        // sleeping.
1044        test_tx
1045            .send(Ok(Message::text(
1046                json!({ "id": 1, "result": { "frameId": "F1" } }).to_string(),
1047            )))
1048            .await
1049            .unwrap();
1050
1051        let res = tokio::time::timeout(Duration::from_secs(5), live_reply_rx)
1052            .await
1053            .expect("a live pending must still be routable after a sweep")
1054            .unwrap()
1055            .unwrap();
1056        assert_eq!(res["frameId"], "F1");
1057
1058        // Both sweeps have run by now. The first reaps the abandoned entries
1059        // dispatched before it, the second the rest, so the total lands just
1060        // under `ABANDONED` — assert the conservative lower bound rather than
1061        // an exact figure that would pin the boundary arithmetic.
1062        let swept = conn
1063            .inner
1064            .swept_total
1065            .load(std::sync::atomic::Ordering::Relaxed);
1066        assert!(
1067            swept >= PENDING_SWEEP_DISPATCH_INTERVAL,
1068            "the actor must actually sweep abandoned pendings; swept {swept} of {ABANDONED} \
1069             abandoned callers across {} dispatches",
1070            ABANDONED + 1
1071        );
1072
1073        conn.shutdown();
1074    }

1089        name: &'static str,
1090        calls: Arc<Mutex<Vec<(&'static str, String)>>>,
1091        behavior: ObserverBehavior,
1092        policy: ObserverFailurePolicy,
1093    }

1097        fn name(&self) -> &'static str {
1098            self.name
1099        }

1101        async fn on_target_attached(
1102            &self,
1103            session: PausedSession<'_>,
1104        ) -> Result<(), ObserverError> {
1105            self.calls
1106                .lock()
1107                .unwrap()
1108                .push((self.name, session.session_id.to_string()));

1113                ObserverBehavior::Sleep(d) => {
1114                    tokio::time::sleep(*d).await;
1115                    Ok(())
1116                }
1117            }
1118        }

1121            self.policy
1122        }

1251    /// Default (`Required`) policy: a timeout must fail closed, exactly like
1252    /// an `Err` or a panic — detach, and never hand the target out via
1253    /// `Runtime.runIfWaitingForDebugger`. This replaces the old fail-open
1254    /// `observer_timeout_releases_debugger_anyway` test — that behavior is
1255    /// now `BestEffort`-only, covered by
1256    /// `best_effort_observer_timeout_releases_debugger_anyway` below.

1259        let (ws, test_tx, mut test_rx) = duplex_pair();
1260        let calls = Arc::new(Mutex::new(Vec::new()));
1261        let obs: Arc<dyn TargetObserver> = Arc::new(RecordingObserver {
1262            name: "slow-required",
1263            calls: calls.clone(),
1264            behavior: ObserverBehavior::Sleep(Duration::from_secs(10)),
1265            policy: ObserverFailurePolicy::Required,

1269

1271

1273        // detach the target instead of releasing it.

1322    /// A `BestEffort` observer that times out must not abandon the rest of
1323    /// the chain: the observer(s) registered after it must still run. This
1324    /// is what keeps a registrar-last chain (zendriver's `TabRegistrar`)
1325    /// from being skipped when an earlier, non-critical observer hangs —
1326    /// see the `handle_target_attached` doc comment above.
1327    #[tokio::test]
1328    async fn best_effort_observer_timeout_does_not_abandon_later_observers() {
1329        let (ws, test_tx, mut test_rx) = duplex_pair();
1330        let calls = Arc::new(Mutex::new(Vec::new()));
1331        let slow: Arc<dyn TargetObserver> = Arc::new(RecordingObserver {
1332            name: "slow-best-effort",
1333            calls: calls.clone(),
1334            behavior: ObserverBehavior::Sleep(Duration::from_secs(10)),
1335            policy: ObserverFailurePolicy::BestEffort,
1336        });
1337        let after: Arc<dyn TargetObserver> = Arc::new(RecordingObserver {
1338            name: "after",
1339            calls: calls.clone(),
1340            behavior: ObserverBehavior::Ok,
1341            policy: ObserverFailurePolicy::Required,
1342        });
1343        let conn = spawn_actor_with_observers_and_timeout(
1344            ws,
1345            vec![slow, after],
1346            Duration::from_millis(100),
1347        );
1348
1349        emit_attached(&test_tx, "S-chain-continues").await;
1350
1351        // The chain must still reach the end and release the debugger, which
1352        // only happens once every observer (including the one after the
1353        // timed-out `BestEffort` observer) has run.
1354        let frame = tokio::time::timeout(Duration::from_millis(500), next_frame(&mut test_rx))
1355            .await
1356            .expect("timeout waiting for runIfWaitingForDebugger");
1357        assert_eq!(frame["method"], "Runtime.runIfWaitingForDebugger");
1358        assert_eq!(frame["sessionId"], "S-chain-continues");
1359
1360        // Prove the second observer actually ran (not just that a release
1361        // frame appeared for some other reason).
1362        let recorded = calls.lock().unwrap().clone();
1363        assert_eq!(
1364            recorded,
1365            vec![
1366                ("slow-best-effort", "S-chain-continues".to_string()),
1367                ("after", "S-chain-continues".to_string()),
1368            ]
1369        );
1370
1371        conn.shutdown();
1372    }
